-- ── Pre-load config BEFORE building UI so CurrentValue is correct from the start ──
local HttpService = game:GetService("HttpService")
local CONFIG_FILE = "TAT_Configs.json"

local function loadAllConfigs()
    local ok, data = pcall(function() return readfile(CONFIG_FILE) end)
    if not ok or not data or data == "" then return {} end
    local ok2, parsed = pcall(function() return HttpService:JSONDecode(data) end)
    if ok2 and type(parsed) == "table" then return parsed end
    return {}
end

local function saveAllConfigs(configs)
    local ok, encoded = pcall(function() return HttpService:JSONEncode(configs) end)
    if not ok then return false end
    local ok2 = pcall(function() writefile(CONFIG_FILE, encoded) end)
    return ok2
end

local startFlags = {
    AutoLaunchCycle      = false,
    AutoPerfectLaunch    = false,
    AutoBuyBurger        = false,
    AutoTrashFarm        = false,
    AutoTrashFarmBlatant = false,
    MeteorESP            = false,
    PlayerESP            = false,
    AutoStateKeeper      = false,
    AntiAFK              = false,
}
local autoLoadConfigName = nil
do
    local configs = loadAllConfigs()
    for name, cfg in pairs(configs) do
        if cfg.autoLoad and cfg.flags then
            autoLoadConfigName = name
            for flag, val in pairs(cfg.flags) do
                if startFlags[flag] ~= nil then
                    startFlags[flag] = val
                end
            end
            break
        end
    end
end

local function isPlayerNearby()
    local char = LocalPlayer.Character
    local hrp  = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local oh = p.Character:FindFirstChild("HumanoidRootPart")
            if oh and (hrp.Position - oh.Position).Magnitude <= PLAYER_RADIUS then
                return true
            end
        end
    end
    return false
end

local function startNoclip()
    if noclipConnection then noclipConnection:Disconnect() end
    noclipConnection = RunService.Stepped:Connect(function()
        local char = LocalPlayer.Character
        if not char then return end
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") then part.CanCollide = false end
        end
    end)
end

local function stopNoclip()
    if noclipConnection then noclipConnection:Disconnect() noclipConnection = nil end
    local char = LocalPlayer.Character
    if char then
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") then part.CanCollide = true end
        end
    end
end

local function randomWait(mn, mx)
    task.wait(mn + math.random() * (mx - mn))
end

local function maybeJump(humanoid)
    if math.random(1,3) == 1 then humanoid.Jump = true task.wait(0.1) humanoid.Jump = false end
end

local function checkStuck(hrp, humanoid, target)
    local cur = hrp.Position
    if lastPos then
        if (cur - lastPos).Magnitude < 0.5 then
            stuckCount += 1
            if stuckCount >= 2 then
                humanoid.Jump = true task.wait(0.2) humanoid.Jump = false
                humanoid:MoveTo(cur + Vector3.new(math.random(-4,4), 0, math.random(-4,4)))
                task.wait(0.3) humanoid:MoveTo(target) stuckCount = 0
            end
        else stuckCount = 0 end
    end
    lastPos = cur
end

local function walkTo(hrp, humanoid, target, timeout, blacklistKey, farmCheck)
    stuckCount = 0 lastPos = nil
    humanoid:MoveTo(target)
    local t = tick() + (timeout or 20)
    local startPos = hrp.Position
    repeat
        task.wait(0.3)
        if farmCheck and not farmCheck() then return false end
        checkStuck(hrp, humanoid, target) maybeJump(humanoid)
    until (hrp.Position - target).Magnitude < 8 or tick() > t
    local arrived = (hrp.Position - target).Magnitude < 8
    if not arrived and blacklistKey and (hrp.Position - startPos).Magnitude < 15 then
        trashBlacklist[blacklistKey] = true
    end
    return arrived
end

local function goUnderground(hrp)
    if isUnderground then return end
    local dest = Vector3.new(hrp.Position.X, UNDERGROUND_Y, hrp.Position.Z)
    local dist = (dest - hrp.Position).Magnitude
    local dur = math.max(dist / SAFE_SPEED, 0.05)
    local tween = TweenService:Create(hrp, TweenInfo.new(dur, Enum.EasingStyle.Linear), {CFrame = CFrame.new(dest)})
    tween:Play() tween.Completed:Wait()
    isUnderground = true
end

local function surfaceUp(hrp)
    if not isUnderground then return end
    local dest = Vector3.new(hrp.Position.X, UNDERGROUND_Y + 8, hrp.Position.Z)
    local dist = (dest - hrp.Position).Magnitude
    local dur = math.max(dist / SAFE_SPEED, 0.05)
    local tween = TweenService:Create(hrp, TweenInfo.new(dur, Enum.EasingStyle.Linear), {CFrame = CFrame.new(dest)})
    tween:Play() tween.Completed:Wait()
    isUnderground = false
end

local function touchGround(hrp)
    if isTouchingGround then return end
    isTouchingGround = true
    local downDest = Vector3.new(hrp.Position.X, GROUND_Y, hrp.Position.Z)
    local downDur = math.max((downDest - hrp.Position).Magnitude / SAFE_SPEED, 0.05)
    local downTween = TweenService:Create(hrp, TweenInfo.new(downDur, Enum.EasingStyle.Linear), {CFrame = CFrame.new(downDest)})
    downTween:Play() downTween.Completed:Wait() task.wait(0.1)
    local upDest = Vector3.new(hrp.Position.X, UNDERGROUND_Y, hrp.Position.Z)
    local upDur = math.max((upDest - hrp.Position).Magnitude / SAFE_SPEED, 0.05)
    local upTween = TweenService:Create(hrp, TweenInfo.new(upDur, Enum.EasingStyle.Linear), {CFrame = CFrame.new(upDest)})
    upTween:Play() upTween.Completed:Wait()
    lastGroundTouch = tick() isTouchingGround = false
end

local function blatantTweenTo(hrp, targetPos, farmCheck)
    local function makeTween()
        local dest = Vector3.new(targetPos.X, UNDERGROUND_Y, targetPos.Z)
        local dist = (dest - hrp.Position).Magnitude
        if dist < 1 then return nil end
        local dur = math.max(dist / SAFE_SPEED, 0.05)
        return TweenService:Create(hrp, TweenInfo.new(dur, Enum.EasingStyle.Linear), {CFrame = CFrame.new(dest)})
    end
    local tween = makeTween() if not tween then return end
    local tweenDone = false
    tween.Completed:Connect(function() tweenDone = true end) tween:Play()
    while not tweenDone do
        task.wait(0.5)
        if not farmCheck() then tween:Cancel() return end
        if tick() - lastGroundTouch >= GROUND_TOUCH_INTERVAL and not isTouchingGround then
            tween:Cancel() touchGround(hrp)
            tween = makeTween() if not tween then return end
            tweenDone = false
            tween.Completed:Connect(function() tweenDone = true end) tween:Play()
        end
    end
end

local function blatantWalkTo(hrp, humanoid, target, timeout, blacklistKey, farmCheck)
    if not isUnderground then goUnderground(hrp) end
    blatantTweenTo(hrp, target, farmCheck)
    local arrived = (Vector3.new(hrp.Position.X, 0, hrp.Position.Z) - Vector3.new(target.X, 0, target.Z)).Magnitude < 8
    if not arrived and blacklistKey then trashBlacklist[blacklistKey] = true end
    return arrived
end

local function getTrashCount(waitForValid)
    local function read()
        local ok, label = pcall(function() return playerGui.HUD["Main HUD"].Corner.Trash.Label end)
        if not ok or not label then return 0, 0 end
        local cur, max = label.Text:match("(%d+)%s*/%s*(%d+)")
        return tonumber(cur) or 0, tonumber(max) or 0
    end
    if waitForValid then
        local deadline = tick() + 3
        local c, m = read()
        while m == 0 and tick() < deadline do task.wait(0.2) c, m = read() end
        return c, m
    end
    return read()
end

local function isDumpsterOnCooldown(part)
    local t = part:FindFirstChild("TimerGUI")
    return t and t.Enabled == true
end

local function getAvailableDumpster()
    local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local best, bestDist = nil, math.huge
    for _, part in ipairs(workspace.Map.Environment.Dumpsters:GetChildren()) do
        if part:IsA("Part") and not isDumpsterOnCooldown(part) then
            local prompt = part:FindFirstChild("Prompt")
            if prompt and prompt.Enabled then
                local d = (hrp.Position - part.Position).Magnitude
                if d < bestDist then best = { part = part, pos = part.Position } bestDist = d end
            end
        end
    end
    return best
end

local function getNextAvailableDumpsterTime()
    local soonest = nil
    for _, part in ipairs(workspace.Map.Environment.Dumpsters:GetChildren()) do
        if part:IsA("Part") then
            local tg = part:FindFirstChild("TimerGUI")
            if tg and tg.Enabled then
                local lbl = tg:FindFirstChild("Label")
                if lbl and lbl.Text ~= "" then
                    local m, s = lbl.Text:match("(%d+):(%d+)")
                    if m and s then
                        local secs = tonumber(m)*60 + tonumber(s)
                        if not soonest or secs < soonest then soonest = secs end
                    end
                end
            else return 0 end
        end
    end
    return soonest or 0
end

local function runFarmLoop(farmCheck, blatant)
    local function getChar()
        local c = LocalPlayer.Character
        if not c then return nil, nil end
        return c:FindFirstChild("HumanoidRootPart"), c:FindFirstChild("Humanoid")
    end
    local lastTrashCount = -1
    local lastIncreaseTime = tick()
    local hasPickedUpAtLeastOne = false
    startAutoSprint(farmCheck)
    if blatant then startNoclip() lastGroundTouch = tick() end
    while farmCheck() do
        local hrp, humanoid = getChar()
        if not hrp or not humanoid then task.wait(1) continue end
        local cur, max = getTrashCount(true)
        if cur ~= lastTrashCount then lastTrashCount = cur lastIncreaseTime = tick() end
        if tick() - lastIncreaseTime > 10 then
            Rayfield:Notify({ Title = "Trash Farm", Content = "Stuck! Resetting...", Duration = 3 })
            if isUnderground then surfaceUp(hrp) end
            LocalPlayer.Character:BreakJoints()
            task.wait(4) isUnderground = false
            startAutoSprint(farmCheck)
            if blatant then startNoclip() lastGroundTouch = tick() end
            lastIncreaseTime = tick() lastTrashCount = -1 hasPickedUpAtLeastOne = false
            continue
        end
        if hasPickedUpAtLeastOne and max > 0 and cur >= max then
            if isUnderground then local h, _ = getChar() if h then surfaceUp(h) end end
            local dumpster = getAvailableDumpster()
            if dumpster then
                local arrived = walkTo(hrp, humanoid, dumpster.pos, 30, nil, farmCheck)
                if arrived then
                    randomWait(0.3, 0.7)
                    local prompt = dumpster.part:FindFirstChild("Prompt")
                    if prompt then
                        fireproximityprompt(prompt, 2) task.wait(2)
                        trashBlacklist = {} lastTrashCount = -1
                        lastIncreaseTime = tick() hasPickedUpAtLeastOne = false
                    end
                end
            else
                local waitSecs = math.max(getNextAvailableDumpsterTime() or 60, 5)
                Rayfield:Notify({ Title = "Trash Farm", Content = "Dumpsters on cooldown! Waiting " .. math.ceil(waitSecs) .. "s", Duration = 5 })
                local waitEnd = tick() + waitSecs + 2
                repeat
                    task.wait(2)
                    if getAvailableDumpster() then break end
                until tick() > waitEnd or not farmCheck()
                lastIncreaseTime = tick()
            end
        else
            local hrp2, humanoid2 = getChar()
            if not hrp2 or not humanoid2 then task.wait(1) continue end
            local nearest, nearestDist, nearestPrompt, nearestKey = nil, math.huge, nil, nil
            for _, paper in ipairs(workspace.Map.Environment.Trash:GetChildren()) do
                if trashBlacklist[paper.Name] then continue end
                local prompt = paper:FindFirstChild("Prompt")
                if prompt and prompt.Enabled then
                    local d = (hrp2.Position - paper.Position).Magnitude
                    if d < nearestDist then
                        nearest = paper nearestDist = d
                        nearestPrompt = prompt nearestKey = paper.Name
                    end
                end
            end
            if nearest then
                local arrived
                if blatant then
                    arrived = blatantWalkTo(hrp2, humanoid2, nearest.Position, 20, nearestKey, farmCheck)
                else
                    arrived = walkTo(hrp2, humanoid2, nearest.Position, 20, nearestKey, farmCheck)
                end
                if arrived and nearestPrompt and nearestPrompt.Enabled then
                    fireproximityprompt(nearestPrompt)
                    hasPickedUpAtLeastOne = true
                    lastIncreaseTime = tick()
                    task.wait(0.3)
                elseif not arrived then
                    lastIncreaseTime = tick()
                end
            else
                task.wait(3)
            end
        end
        task.wait(0.1)
    end
    stopAutoSprint()
    if blatant then
        stopNoclip()
        local h = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if h and isUnderground then surfaceUp(h) end
        isUnderground = false
    end
end

local function topNeedsPickup()
    local top = getMyTop()
    if not top then return false end
    local target = top:FindFirstChild("Target")
    return target ~= nil and target.Value == nil
end

local function getTopPosition()
    local top = getMyTop()
    if not top then return nil end
    return top:GetPivot().Position
end

end

local function doAutoLaunchCycle()
    if isDoingLaunchCycle then return end
    isDoingLaunchCycle = true
    local character = LocalPlayer.Character
    local hrp = character and character:FindFirstChild("HumanoidRootPart")
    local humanoid = character and character:FindFirstChild("Humanoid")
    if not hrp or not humanoid then isDoingLaunchCycle = false return end
    local standCFrame = hrp.CFrame
    print("Deploying bey...")
    pressKey(Enum.KeyCode.E)
    task.wait(0.5)
    print("Triggering launch...")
    VirtualInputManager:SendMouseButtonEvent(0, 0, 0, true, game, 1)
    task.wait(0.05)
    VirtualInputManager:SendMouseButtonEvent(0, 0, 0, false, game, 1)
    task.wait(2)
    print("Waiting for bey to need pickup...")
    local timeout = tick() + 120
    repeat
        task.wait(0.5)
        if not autoLaunchRunning then isDoingLaunchCycle = false return end
    until topNeedsPickup() or tick() > timeout
    if not autoLaunchRunning then isDoingLaunchCycle = false return end
    if not topNeedsPickup() then print("Timed out waiting for bey") isDoingLaunchCycle = false return end
    local topPos = getTopPosition()
    if not topPos then isDoingLaunchCycle = false return end
    print("Walking to bey...")
    stuckCount = 0 lastPos = nil
    humanoid:MoveTo(topPos)
    local walkTimeout = tick() + 25
    repeat
        task.wait(0.3)
        if not autoLaunchRunning then isDoingLaunchCycle = false return end
        local newPos = getTopPosition()
        if newPos then topPos = newPos humanoid:MoveTo(topPos) end
        checkStuck(hrp, humanoid, topPos) maybeJump(humanoid)
    until (hrp.Position - (getTopPosition() or topPos)).Magnitude < 5 or tick() > walkTimeout
    task.wait(0.3)
    print("Picking up bey...")
    pressKey(Enum.KeyCode.E)
    task.wait(0.5)
    print("Walking back...")
    stuckCount = 0 lastPos = nil
    humanoid:MoveTo(standCFrame.Position)
    local returnTimeout = tick() + 20
    repeat
        task.wait(0.3)
        if not autoLaunchRunning then isDoingLaunchCycle = false return end
        checkStuck(hrp, humanoid, standCFrame.Position) maybeJump(humanoid)
    until (hrp.Position - standCFrame.Position).Magnitude < 3 or tick() > returnTimeout
    hrp.CFrame = standCFrame
    print("Cycle complete, repeating...")
    isDoingLaunchCycle = false
    if autoLaunchRunning then task.wait(0.3) task.spawn(doAutoLaunchCycle) end
end

    end)
end

    if not ok or not label then return false end
    return label.Visible and string.find(label.Text, "out of energy") ~= nil
end

local isBurgerBuying = false

local function doBurgerRun(manual)
    if isBurgerBuying then return end
    isBurgerBuying = true isManualRun = manual or false manualCancelled = false
    local character = LocalPlayer.Character
    local hrp = character and character:FindFirstChild("HumanoidRootPart")
    local humanoid = character and character:FindFirstChild("Humanoid")
    if not hrp or not humanoid then isBurgerBuying = false return end
    local originalPos = hrp.Position
    local standPos = Vector3.new(573.88, 118, -157.43)
    local keyConnection
    if isManualRun then
        keyConnection = UserInputService.InputBegan:Connect(function(input, gp)
            if gp then return end
            if input.UserInputType == Enum.UserInputType.Keyboard then
                manualCancelled = true humanoid:MoveTo(hrp.Position)
            end
        end)
    end
    stuckCount = 0 lastPos = nil
    humanoid:MoveTo(standPos)
    local timeout = tick() + 12
    repeat
        task.wait(0.3)
        if manualCancelled then break end
        checkStuck(hrp, humanoid, standPos) maybeJump(humanoid)
    until (hrp.Position - standPos).Magnitude < 3 or tick() > timeout or manualCancelled
    if not manualCancelled then
        randomWait(0.4, 0.9)
        local bought = false
        local ok, vendor = pcall(function() return workspace.Map["Food Vendors"]:GetChildren()[11] end)
        if ok and vendor then
            local prompt = vendor.Counter["Prompt Host"].Prompt
            if prompt and not bought then bought = true fireproximityprompt(prompt) end
        end
        if not isManualRun then
            randomWait(2.8, 3.4)
            local returnPos = originalPos + Vector3.new(math.random(-2,2), 0, math.random(-2,2))
            stuckCount = 0 lastPos = nil
            humanoid:MoveTo(returnPos)
            local returnTimeout = tick() + 12
            repeat
                task.wait(0.3) checkStuck(hrp, humanoid, returnPos) maybeJump(humanoid)
            until (hrp.Position - returnPos).Magnitude < 3 or tick() > returnTimeout
        end
    end
    if keyConnection then keyConnection:Disconnect() end
    isBurgerBuying = false isManualRun = false manualCancelled = false
end

local function onAutoLaunchCycle(value)
    toggleStates.AutoLaunchCycle = value
    autoLaunchRunning = value
    if value then
        task.spawn(doAutoLaunchCycle)
        Rayfield:Notify({ Title = "Auto Launch", Content = "Auto launch cycle started!", Duration = 3 })
    else
        isDoingLaunchCycle = false
        Rayfield:Notify({ Title = "Auto Launch", Content = "Auto launch stopped.", Duration = 3 })
    end
end
toggleCallbacks["AutoLaunchCycle"] = onAutoLaunchCycle
MainTab:CreateToggle({ Name = "Auto Launch Cycle", CurrentValue = startFlags.AutoLaunchCycle, Flag = "AutoLaunchCycle", Callback = onAutoLaunchCycle })
if startFlags.AutoLaunchCycle then onAutoLaunchCycle(true) end

local function onAutoPerfectLaunch(value)
    toggleStates.AutoPerfectLaunch = value
    if perfectConnection then perfectConnection:Disconnect() perfectConnection = nil end
    if value then
        local passCount = 0
        local wasInZone = false
        perfectConnection = RunService.Heartbeat:Connect(function()
            local now = tick()
            if now - lastClick < COOLDOWN then return end
            local bar = getBar() if not bar then return end
            local goal    = bar:FindFirstChild("Goal")
            local tracker = bar:FindFirstChild("Tracker")
            if not goal or not tracker then return end
            local tc = tracker.AbsolutePosition.X + tracker.AbsoluteSize.X / 2
            local goalCenter = goal.AbsolutePosition.X + goal.AbsoluteSize.X / 2
            local goalHalf = goal.AbsoluteSize.X / 2
            local threshold = goalHalf * 0.4
            local inZone = math.abs(tc - goalCenter) < threshold
            if inZone and not wasInZone then passCount = passCount + 1 end
            wasInZone = inZone
            if inZone and passCount >= 2 then
                passCount = 0 lastClick = now doClick()
            end
        end)
    end
end
toggleCallbacks["AutoPerfectLaunch"] = onAutoPerfectLaunch
MainTab:CreateToggle({ Name = "Auto Perfect Launch", CurrentValue = startFlags.AutoPerfectLaunch, Flag = "AutoPerfectLaunch", Callback = onAutoPerfectLaunch })
if startFlags.AutoPerfectLaunch then onAutoPerfectLaunch(true) end

-- ── Auto State Keeper ─────────────────────────────────────────────────────────
-- Cycle: Attacking -> Guarding -> Idle -> Attacking
-- From Idle: 1x G = Attacking, 2x G = Guarding
local preferredState = "Attacking"
local stateKeeperConnection = nil
local isFixingState = false

local function getStateKeeperLabel()
    local ok, label = pcall(function()
        return playerGui.HUD["Top UI"].Info.Top.State
    end)
    if ok and label then return label end
    return nil
end

local function pressG()
    VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.G, false, game)
    task.wait(0.05)
    VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.G, false, game)
end

local function onAutoStateKeeper(value)
    toggleStates.AutoStateKeeper = value
    if value then
        stateKeeperConnection = RunService.Heartbeat:Connect(function()
            if isFixingState then return end
            local label = getStateKeeperLabel()
            if not label then return end
            local state = label.Text
            if state:find("Recalling") then return end
            if not state:find("Idle") then return end
            isFixingState = true
            task.spawn(function()
                -- From Idle: press G once to reach Attacking, twice to reach Guarding
                local presses = preferredState == "Guarding" and 2 or 1
                for _ = 1, presses do
                    pressG()
                    task.wait(0.3)
                end
                isFixingState = false
            end)
        end)
        Rayfield:Notify({ Title = "State Keeper", Content = "Keeping state: " .. preferredState, Duration = 3 })
    else
        if stateKeeperConnection then stateKeeperConnection:Disconnect() stateKeeperConnection = nil end
        isFixingState = false
        Rayfield:Notify({ Title = "State Keeper", Content = "State keeper stopped.", Duration = 3 })
    end
end

local ESPTab = Window:CreateTab("ESP", 4483362458)

local playerESPEnabled = false
local playerESPConnections = {}
local playerESPHighlights = {}

local function removePlayerESP(player)
    if playerESPHighlights[player] then playerESPHighlights[player]:Destroy() playerESPHighlights[player] = nil end
    if player.Character then
        local hrp = player.Character:FindFirstChild("HumanoidRootPart")
        if hrp then local bb = hrp:FindFirstChild("PlayerESPBB") if bb then bb:Destroy() end end
    end
end

local function applyPlayerESP(player)
    if player == LocalPlayer then return end
    local char = player.Character if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then
        local deadline = tick() + 3
        repeat task.wait(0.1) hrp = char:FindFirstChild("HumanoidRootPart") until hrp or tick() > deadline
    end
    if not hrp then return end
    local old = char:FindFirstChild("PlayerESPHighlight")
    if old then old:Destroy() end
    if playerESPHighlights[player] then
        pcall(function() playerESPHighlights[player]:Destroy() end)
        playerESPHighlights[player] = nil
    end
    local oldBB = hrp:FindFirstChild("PlayerESPBB")
    if oldBB then oldBB:Destroy() end
    local highlight = Instance.new("Highlight")
    highlight.Name = "PlayerESPHighlight"
    highlight.FillColor = Color3.fromRGB(255, 0, 0)
    highlight.OutlineColor = Color3.fromRGB(255, 0, 0)
    highlight.FillTransparency = 0.3
    highlight.OutlineTransparency = 0
    highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    highlight.Adornee = char
    highlight.Parent = char
    playerESPHighlights[player] = highlight
    local bb = Instance.new("BillboardGui")
    bb.Name = "PlayerESPBB" bb.AlwaysOnTop = true
    bb.Size = UDim2.new(0, 150, 0, 50) bb.StudsOffset = Vector3.new(0, 3.5, 0)
    bb.Adornee = hrp bb.Parent = hrp
    local label = Instance.new("TextLabel")
    label.BackgroundTransparency = 1 label.Size = UDim2.new(1, 0, 1, 0)
    label.Font = Enum.Font.GothamBold label.TextSize = 16
    label.TextColor3 = Color3.fromRGB(255, 80, 80)
    label.TextStrokeTransparency = 0 label.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
    label.Text = player.DisplayName .. "\n(@" .. player.Name .. ")"
    label.Parent = bb
end

local function setupPlayerESP(player)
    if player == LocalPlayer then return end
    if playerESPConnections[player] then
        playerESPConnections[player]:Disconnect()
        playerESPConnections[player] = nil
    end
    if player.Character then task.spawn(function() applyPlayerESP(player) end) end
    local conn = player.CharacterAdded:Connect(function()
        task.wait(0.5) applyPlayerESP(player)
    end)
    playerESPConnections[player] = conn
end

local function startPlayerESP()
    for _, p in ipairs(Players:GetPlayers()) do setupPlayerESP(p) end
    playerESPConnections["PlayerAdded"] = Players.PlayerAdded:Connect(function(p)
        task.wait(0.5) setupPlayerESP(p)
    end)
end

local function stopPlayerESP()
    for _, conn in pairs(playerESPConnections) do conn:Disconnect() end
    playerESPConnections = {}
    for _, p in ipairs(Players:GetPlayers()) do removePlayerESP(p) end
    playerESPHighlights = {}
end

        AutoLaunchCycle      = toggleStates.AutoLaunchCycle,

local function applyFlags(flags)
    for flag, value in pairs(flags) do
        local cb = toggleCallbacks[flag]
        if cb then
            toggleStates[flag] = value
            pcall(cb, value)
        end
    end
end

local function getConfigNames()
    local configs = loadAllConfigs()
    local names = {}
    for name in pairs(configs) do table.insert(names, name) end
    table.sort(names)
    if #names == 0 then table.insert(names, "No Configs Saved") end
    return names
end

local function refreshDropdown()
    local names = getConfigNames()
    local current = selectedConfig ~= "" and selectedConfig or names[1]
    local found = false
    for _, n in ipairs(names) do if n == current then found = true break end end
    if not found then current = names[1] selectedConfig = current ~= "No Configs Saved" and current or "" end
    pcall(function() configDropdown:Refresh(names, { current }) end)
end
