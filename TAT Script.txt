-- ── Noclip ────────────────────────────────────────────────────────────────────
local function startNoclip()
    if noclipConnection then noclipConnection:Disconnect() end
    noclipConnection = RunService.Stepped:Connect(function()
        local char = Players.LocalPlayer.Character
        if not char then return end
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") then
                part.CanCollide = false
            end
        end
    end)
end

local function stopNoclip()
    if noclipConnection then
        noclipConnection:Disconnect()
        noclipConnection = nil
    end
    local char = Players.LocalPlayer.Character
    if char then
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") then
                part.CanCollide = true
            end
        end
    end
end

local function pressCtrl()
    pcall(function()
        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.LeftControl, false, game)
    end)
end

local function releaseCtrl()
    pcall(function()
        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.LeftControl, false, game)

end

local function startAutoSprint(farmRunningRef)
    if sprintConnection then sprintConnection:Disconnect() end
    pressCtrl()
    sprintConnection = RunService.Heartbeat:Connect(function()
        if not farmRunningRef() then
            releaseCtrl()
            sprintConnection:Disconnect()
            sprintConnection = nil
            return
        end
        pressCtrl()
    end)
end

local function stopAutoSprint()
    if sprintConnection then
        sprintConnection:Disconnect()
        sprintConnection = nil
    end
    releaseCtrl()
end

local function randomWait(min, max)
    task.wait(min + math.random() * (max - min))
end
 
local function maybeJump(humanoid)
    if math.random(1, 3) == 1 then
        humanoid.Jump = true
        task.wait(0.1)
        humanoid.Jump = false
    end
end
 
local function checkStuck(hrp, humanoid, target)
    local currentPos = hrp.Position
    if lastPos then
        local moved = (currentPos - lastPos).Magnitude
        if moved < 0.5 then
            stuckCount += 1
            if stuckCount >= 2 then
                humanoid.Jump = true
                task.wait(0.2)
                humanoid.Jump = false
                local offset = Vector3.new(math.random(-4, 4), 0, math.random(-4, 4))
                humanoid:MoveTo(currentPos + offset)
                task.wait(0.3)
                humanoid:MoveTo(target)
                stuckCount = 0
            end
        else
            stuckCount = 0
        end
    end
    lastPos = currentPos
end
 
local function walkTo(hrp, humanoid, target, timeout, blacklistKey, farmCheck)
    stuckCount = 0
    lastPos = nil
    humanoid:MoveTo(target)
    local t = tick() + (timeout or 20)
    local startPos = hrp.Position

    repeat
        task.wait(0.3)
        if not farmCheck() then return false end
        checkStuck(hrp, humanoid, target)
        maybeJump(humanoid)
    until (hrp.Position - target).Magnitude < 8 or tick() > t

    local arrived = (hrp.Position - target).Magnitude < 8

    if not arrived and blacklistKey then
        local totalMoved = (hrp.Position - startPos).Magnitude
        if totalMoved < 15 then
            trashBlacklist[blacklistKey] = true
        end
    end

    return arrived
end
 
local function getTrashCount(waitForValid)
    local function read()
        local ok, label = pcall(function()
            return playerGui.HUD["Main HUD"].Corner.Trash.Label
        end)
        if not ok or not label then return 0, 0 end
        local cur, max = label.Text:match("(%d+)%s*/%s*(%d+)")
        local c = tonumber(cur) or 0
        local m = tonumber(max) or 0
        return c, m
    end

    if waitForValid then
        local deadline = tick() + 3
        local c, m = read()
        while m == 0 and tick() < deadline do
            task.wait(0.2)
            c, m = read()
        end
        return c, m
    end
    return read()
end
 
local function isDumpsterOnCooldown(dumpsterPart)
    local timerGui = dumpsterPart:FindFirstChild("TimerGUI")
    if not timerGui then return false end
    return timerGui.Enabled == true
end

local function getAvailableDumpster()
    local dumpsters = workspace.Map.Environment.Dumpsters
    local char = Players.LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local best, bestDist = nil, math.huge
    for _, part in ipairs(dumpsters:GetChildren()) do
        if part:IsA("Part") and not isDumpsterOnCooldown(part) then
            local prompt = part:FindFirstChild("Prompt")
            if prompt and prompt.Enabled then
                local pos = part.Position
                local dist = (hrp.Position - pos).Magnitude
                if dist < bestDist then
                    best = {name = part.Name, part = part, pos = pos}
                    bestDist = dist
                end
            end
        end
    end
    return best
end

local function getNextAvailableDumpsterTime()
    local dumpsters = workspace.Map.Environment.Dumpsters
    local soonest = nil
    for _, part in ipairs(dumpsters:GetChildren()) do
        if part:IsA("Part") then
            local timerGui = part:FindFirstChild("TimerGUI")
            if timerGui and timerGui.Enabled then
                local label = timerGui:FindFirstChild("Label")
                if label and label.Text ~= "" then
                    local m, s = label.Text:match("(%d+):(%d+)")
                    if m and s then
                        local secs = tonumber(m) * 60 + tonumber(s)
                        if not soonest or secs < soonest then
                            soonest = secs
                        end
                    end
                end
            else
                return 0
            end
        end
    end
    return soonest or 0
end

-- ── Shared farm loop (used by both normal and blatant) ───────────────────────
local function runFarmLoop(farmCheck, blatant)
    local function getChar()
        local c = Players.LocalPlayer.Character
        if not c then return nil, nil end
        return c:FindFirstChild("HumanoidRootPart"), c:FindFirstChild("Humanoid")
    end

    local lastTrashCount = -1
    local lastIncreaseTime = tick()
    local hasPickedUpAtLeastOne = false

    startAutoSprint(farmCheck)
    if blatant then startNoclip() end

    while farmCheck() do
        local hrp, humanoid = getChar()
        if not hrp or not humanoid then task.wait(1) continue end

        local cur, max = getTrashCount(true)

        if cur ~= lastTrashCount then
            lastTrashCount = cur
            lastIncreaseTime = tick()
        end

        if tick() - lastIncreaseTime > 10 then
            Rayfield:Notify({ Title = "Trash Farm", Content = "Stuck! Resetting character...", Duration = 3 })
            Players.LocalPlayer.Character:BreakJoints()
            task.wait(4)
            startAutoSprint(farmCheck)
            if blatant then startNoclip() end
            lastIncreaseTime = tick()
            lastTrashCount = -1
            hasPickedUpAtLeastOne = false
            continue
        end

        if hasPickedUpAtLeastOne and max > 0 and cur >= max then
            local dumpster = getAvailableDumpster()
            if dumpster then
                local arrived = walkTo(hrp, humanoid, dumpster.pos, 30, nil, farmCheck)
                if arrived then
                    randomWait(0.3, 0.7)
                    local prompt = dumpster.part:FindFirstChild("Prompt")
                    if prompt then
                        fireproximityprompt(prompt, 2)
                        task.wait(2)
                        trashBlacklist = {}
                        lastTrashCount = -1
                        lastIncreaseTime = tick()
                        hasPickedUpAtLeastOne = false
                    end
                end
            else
                local waitSecs = getNextAvailableDumpsterTime()
                waitSecs = math.max(waitSecs or 60, 5)
                Rayfield:Notify({
                    Title = "Trash Farm",
                    Content = "All dumpsters on cooldown! Waiting " .. math.ceil(waitSecs) .. "s",
                    Duration = 5,
                })
                local waitEnd = tick() + waitSecs + 2
                repeat
                    task.wait(2)
                    if getAvailableDumpster() then break end
                until tick() > waitEnd or not farmCheck()
                lastIncreaseTime = tick()
            end
        else
            local hrp2, humanoid2 = getChar()
            if not hrp2 or not humanoid2 then task.wait(1) continue end

            local trashFolder = workspace.Map.Environment.Trash
            local nearest, nearestDist, nearestPrompt, nearestKey = nil, math.huge, nil, nil

            for _, paper in ipairs(trashFolder:GetChildren()) do
                if trashBlacklist[paper.Name] then continue end
                local prompt = paper:FindFirstChild("Prompt")
                if prompt and prompt.Enabled then
                    local dist = (hrp2.Position - paper.Position).Magnitude
                    if dist < nearestDist then
                        nearest = paper
                        nearestDist = dist
                        nearestPrompt = prompt
                        nearestKey = paper.Name
                    end
                end
            end

            if nearest then
                local arrived = walkTo(hrp2, humanoid2, nearest.Position, 20, nearestKey, farmCheck)
                if arrived and nearestPrompt and nearestPrompt.Enabled then
                    fireproximityprompt(nearestPrompt)
                    hasPickedUpAtLeastOne = true
                    lastIncreaseTime = tick()
                    task.wait(0.3)
                elseif not arrived then
                    lastIncreaseTime = tick()
                end
            else
                task.wait(3)
            end
        end

        task.wait(0.1)
    end

    stopAutoSprint()
    if blatant then stopNoclip() end
end

local function getBar()
    local hud = playerGui:FindFirstChild("HUD")
    if not hud then return end
    local launch = hud:FindFirstChild("Launch")
    if not launch or not launch.Visible then return end
    local bar = launch:FindFirstChild("Bar")
    if not bar or not bar.Visible then return end
    return bar
end
 
local function isTrackerInPerfectMiddle(zone, tracker)
    local zoneCenter = zone.AbsolutePosition.X + zone.AbsoluteSize.X / 2
    local trackerCenter = tracker.AbsolutePosition.X + tracker.AbsoluteSize.X / 2
    local threshold = zone.AbsoluteSize.X * 0.15
    return math.abs(trackerCenter - zoneCenter) < threshold
end
 
local function isOutOfEnergy()
    local ok, label = pcall(function()
        return playerGui["Player Notifications"].Notifications.Message.Label
    end)
    if not ok or not label then return false end
    return label.Visible and string.find(label.Text, "out of energy") ~= nil
end
 
local isBurgerBuying = false
 
local function doBurgerRun(manual)
    if isBurgerBuying then return end
    isBurgerBuying = true
    isManualRun = manual or false
    manualCancelled = false
 
    local character = Players.LocalPlayer.Character
    local hrp = character and character:FindFirstChild("HumanoidRootPart")
    local humanoid = character and character:FindFirstChild("Humanoid")
    if not hrp or not humanoid then isBurgerBuying = false return end
 
    local originalPos = hrp.Position
    local standPos = Vector3.new(573.88, 118, -157.43)
 
    local keyConnection
    if isManualRun then
        keyConnection = UserInputService.InputBegan:Connect(function(input, gp)
            if gp then return end
            if input.UserInputType == Enum.UserInputType.Keyboard then
                manualCancelled = true
                humanoid:MoveTo(hrp.Position)
            end
        end)
    end
 
    stuckCount = 0
    lastPos = nil
    humanoid:MoveTo(standPos)
    local timeout = tick() + 12
    repeat
        task.wait(0.3)
        if manualCancelled then break end
        checkStuck(hrp, humanoid, standPos)
        maybeJump(humanoid)
    until (hrp.Position - standPos).Magnitude < 3 or tick() > timeout or manualCancelled
 
    if not manualCancelled then
        randomWait(0.4, 0.9)
        local bought = false
        local ok, vendor = pcall(function()
            return workspace.Map["Food Vendors"]:GetChildren()[11]
        end)
        if ok and vendor then
            local prompt = vendor.Counter["Prompt Host"].Prompt
            if prompt and not bought then
                bought = true
                fireproximityprompt(prompt)
            end
        end
 
        if not isManualRun then
            randomWait(2.8, 3.4)
            local returnPos = originalPos + Vector3.new(math.random(-2,2), 0, math.random(-2,2))
            stuckCount = 0
            lastPos = nil
            humanoid:MoveTo(returnPos)
            local returnTimeout = tick() + 12
            repeat
                task.wait(0.3)
                checkStuck(hrp, humanoid, returnPos)
                maybeJump(humanoid)
            until (hrp.Position - returnPos).Magnitude < 3 or tick() > returnTimeout
        end
    end
 
    if keyConnection then keyConnection:Disconnect() end
    isBurgerBuying = false
    isManualRun = false
    manualCancelled = false
end
 
local function onAutoPerfectLaunch(value)
    toggleStates.AutoPerfectLaunch = value
    if value then
        perfectConnection = RunService.Heartbeat:Connect(function()
            local now = tick()
            if now - lastClick < COOLDOWN then return end
            local bar = getBar()
            if not bar then return end
            local goal = bar:FindFirstChild("Goal")
            local tracker = bar:FindFirstChild("Tracker")
            if not goal or not tracker then return end
            if isTrackerInPerfectMiddle(goal, tracker) then
                lastClick = now
                doClick()
            end
        end)
    else
        if perfectConnection then perfectConnection:Disconnect() perfectConnection = nil end
    end
end

local meteorESPEnabled = false
local meteorESPConnection = nil
local meteorAddedConnection = nil
local meteorRemovedConnections = {}

local function createMeteorESP(meteor)
    local core = meteor:FindFirstChild("Core")
    if not core then return end

    if core:FindFirstChild("MeteorESP") then return end

    local bb = Instance.new("BillboardGui")
    bb.Name = "MeteorESP"
    bb.AlwaysOnTop = true
    bb.Size = UDim2.new(0, 120, 0, 40)
    bb.StudsOffset = Vector3.new(0, 4, 0)
    bb.Adornee = core
    bb.Parent = core

    local frame = Instance.new("Frame")
    frame.BackgroundTransparency = 1
    frame.Size = UDim2.new(1, 0, 1, 0)
    frame.Parent = bb

    local label = Instance.new("TextLabel")
    label.BackgroundTransparency = 1
    label.Size = UDim2.new(1, 0, 1, 0)
    label.Font = Enum.Font.GothamBold
    label.TextSize = 14
    label.TextColor3 = Color3.fromRGB(180, 120, 255)
    label.TextStrokeTransparency = 0
    label.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
    label.Text = "☄ Meteor"
    label.Parent = frame

    local conn = RunService.Heartbeat:Connect(function()
        if not meteorESPEnabled or not core or not core.Parent then return end
        local char = Players.LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then
            local dist = math.floor((hrp.Position - core.Position).Magnitude)
            label.Text = "☄ Meteor\n" .. dist .. " studs"
        end
    end)

    local key = meteor:GetFullName()
    meteorRemovedConnections[key] = meteor.AncestryChanged:Connect(function()
        if not meteor.Parent then
            conn:Disconnect()
            if meteorRemovedConnections[key] then
                meteorRemovedConnections[key]:Disconnect()
                meteorRemovedConnections[key] = nil
            end
        end
    end)
end

local function removeAllMeteorESP()
    for _, meteor in ipairs(workspace.Map.Meteors:GetChildren()) do
        local core = meteor:FindFirstChild("Core")
        if core then
            local bb = core:FindFirstChild("MeteorESP")
            if bb then bb:Destroy() end
        end
    end
    for _, conn in pairs(meteorRemovedConnections) do
        conn:Disconnect()
    end
    meteorRemovedConnections = {}
end

local function startMeteorESP()
    for _, meteor in ipairs(workspace.Map.Meteors:GetChildren()) do
        createMeteorESP(meteor)
    end
    meteorAddedConnection = workspace.Map.Meteors.ChildAdded:Connect(function(meteor)
        if meteorESPEnabled then
            task.wait(0.2)
            createMeteorESP(meteor)
        end
    end)
end

local function stopMeteorESP()
    if meteorAddedConnection then
        meteorAddedConnection:Disconnect()
        meteorAddedConnection = nil
    end
    removeAllMeteorESP()
end

-- ── Config System ─────────────────────────────────────────────────────────────
local HttpService = game:GetService("HttpService")

-- Use a folder inside the executor's workspace folder for reliability
local CONFIG_FILE = "TAT_Configs.json"

local function fileExists(path)
    local ok, result = pcall(function() return isfile(path) end)
    return ok and result
end

local function loadAllConfigs()
    if not fileExists(CONFIG_FILE) then return {} end
    local ok, data = pcall(function() return readfile(CONFIG_FILE) end)
    if not ok or not data or data == "" then return {} end
    local ok2, parsed = pcall(function() return HttpService:JSONDecode(data) end)
    if ok2 and type(parsed) == "table" then return parsed end
    return {}
end

local function saveAllConfigs(configs)
    local ok, encoded = pcall(function() return HttpService:JSONEncode(configs) end)
    if not ok then
        Rayfield:Notify({ Title = "Config", Content = "Failed to encode config!", Duration = 4 })
        return false
    end
    local ok2, err = pcall(function() writefile(CONFIG_FILE, encoded) end)
    if not ok2 then
        Rayfield:Notify({ Title = "Config", Content = "Failed to write file! Check executor permissions.", Duration = 5 })
        return false
    end
    -- Verify the write actually worked by reading back
    local ok3, verify = pcall(function() return readfile(CONFIG_FILE) end)
    if not ok3 or verify ~= encoded then
        Rayfield:Notify({ Title = "Config", Content = "Save verify failed! File may not persist.", Duration = 5 })
        return false
    end
    return true
end

local function getCurrentFlags()
    return {
        AutoPerfectLaunch    = toggleStates.AutoPerfectLaunch,
        AutoBuyBurger        = toggleStates.AutoBuyBurger,
        AutoTrashFarm        = toggleStates.AutoTrashFarm,
        AutoTrashFarmBlatant = toggleStates.AutoTrashFarmBlatant,
        MeteorESP            = toggleStates.MeteorESP,
    }
end

local function applyFlags(flags)
    for flag, value in pairs(flags) do
        local cb = toggleCallbacks[flag]
        if cb then
            -- Always apply, even if value matches, to ensure visual + functional sync
            pcall(function()
                if Rayfield.Flags[flag] then
                    Rayfield.Flags[flag]:Set(value)
                end
            end)
            if value ~= toggleStates[flag] then
                cb(value)
            end
        end
    end
end

local function getConfigNames()
    local configs = loadAllConfigs()
    local names = {}
    for name, _ in pairs(configs) do
        table.insert(names, name)
    end
    if #names == 0 then
        table.insert(names, "No Configs Saved")
    end
    return names
end

    CurrentOption = {"No Configs Saved"},
    Flag = "SelectedConfig",
    MultipleOptions = false,
    Callback = function(selected)
        selectedConfig = type(selected) == "table" and selected[1] or selected
    end,
})

ConfigTab:CreateButton({
    Name = "Load Selected Config",
    Callback = function()
        if selectedConfig == "" or selectedConfig == "No Configs Saved" then
            Rayfield:Notify({ Title = "Config", Content = "Select a config first!", Duration = 3 })
            return
        end
        local configs = loadAllConfigs()
        local cfg = configs[selectedConfig]
        if cfg and cfg.flags then
            applyFlags(cfg.flags)
            -- Show what was loaded for confirmation
            local summary = ""
            for flag, val in pairs(cfg.flags) do
                if val then summary = summary .. flag .. " " end
            end
            Rayfield:Notify({
                Title = "Config Loaded: " .. selectedConfig,
                Content = summary ~= "" and ("ON: " .. summary) or "All toggles off",
                Duration = 5,
            })
        else
            Rayfield:Notify({ Title = "Config", Content = "Config not found or empty!", Duration = 3 })
        end
    end,
})

ConfigTab:CreateButton({
    Name = "Delete Selected Config",
    Callback = function()
        if selectedConfig == "" or selectedConfig == "No Configs Saved" then
            Rayfield:Notify({ Title = "Config", Content = "Select a config to delete!", Duration = 3 })
            return
        end
        local configs = loadAllConfigs()
        if configs[selectedConfig] then
            configs[selectedConfig] = nil
            saveAllConfigs(configs)
            Rayfield:Notify({ Title = "Config", Content = "Deleted: " .. selectedConfig, Duration = 3 })
            selectedConfig = ""
            configDropdown:Refresh(getConfigNames(), {"No Configs Saved"})
        else
            Rayfield:Notify({ Title = "Config", Content = "Config not found!", Duration = 3 })
        end
    end,
})

local configNameInput = ""
ConfigTab:CreateInput({

    PlaceholderText = "Enter config name...",
    RemoveTextAfterFocusLost = false,

        configNameInput = text
    end,

local autoLoadOnSave = false
ConfigTab:CreateToggle({
    Name = "Auto Load on Execute",
    CurrentValue = false,
    Flag = "ConfigAutoLoad",
    Callback = function(value)
        autoLoadOnSave = value
    end,
})

ConfigTab:CreateButton({
    Name = "Save Config",
    Callback = function()
        local name = configNameInput
        if not name or name == "" then
            Rayfield:Notify({ Title = "Config", Content = "Enter a config name first!", Duration = 3 })
            return
        end
        local configs = loadAllConfigs()
        if autoLoadOnSave then
            for _, v in pairs(configs) do v.autoLoad = false end
        end
        local flags = getCurrentFlags()
        configs[name] = { flags = flags, autoLoad = autoLoadOnSave }
        local saved = saveAllConfigs(configs)
        if saved then
            -- Show exactly what got saved
            local summary = ""
            for flag, val in pairs(flags) do
                if val then summary = summary .. flag .. " " end
            end
            Rayfield:Notify({
                Title = "Saved: " .. name,
                Content = (summary ~= "" and ("ON: " .. summary) or "All toggles off") .. (autoLoadOnSave and " | Auto Load ON" or ""),
                Duration = 5,
            })
            configDropdown:Refresh(getConfigNames(), {name})
            selectedConfig = name
        end
    end,
})

-- On load: populate dropdown and auto-load if flagged
task.spawn(function()
    task.wait(2)
    local configs = loadAllConfigs()
    local names = getConfigNames()
    configDropdown:Refresh(names, {names[1]})
    selectedConfig = names[1] ~= "No Configs Saved" and names[1] or ""
    for name, cfg in pairs(configs) do
        if cfg.autoLoad and cfg.flags then
            task.wait(0.5)
            applyFlags(cfg.flags)
            Rayfield:Notify({ Title = "Config", Content = "Auto loaded: " .. name, Duration = 4 })
            break
        end
    end
end)
