-- ── Pre-load config BEFORE building UI so CurrentValue is correct from the start ──
local HttpService = game:GetService("HttpService")
local CONFIG_FILE = "TAT_Configs.json"

local function loadAllConfigs()
    local ok, data = pcall(function() return readfile(CONFIG_FILE) end)
    if not ok or not data or data == "" then return {} end
    local ok2, parsed = pcall(function() return HttpService:JSONDecode(data) end)
    if ok2 and type(parsed) == "table" then return parsed end
    return {}
end

local function saveAllConfigs(configs)
    local ok, encoded = pcall(function() return HttpService:JSONEncode(configs) end)
    if not ok then return false end
    local ok2 = pcall(function() writefile(CONFIG_FILE, encoded) end)
    return ok2
end

-- ── Meteor time persistence ───────────────────────────────────────────────────
local METEOR_TIME_FILE = "TAT_MeteorTime.json"

local function loadLastMeteorTime()
    local ok, data = pcall(function() return readfile(METEOR_TIME_FILE) end)
    if not ok or not data or data == "" then return nil end
    local ok2, parsed = pcall(function() return HttpService:JSONDecode(data) end)
    if ok2 and type(parsed) == "table" and parsed.lastSpawn then return parsed.lastSpawn end
    return nil
end

local function saveLastMeteorTime(t)
    pcall(function() writefile(METEOR_TIME_FILE, HttpService:JSONEncode({ lastSpawn = t })) end)
end

local function getMoveCost(moveName)
    if not moveName or moveName == "?" or moveName == "none" or moveName == "None" then return nil end
    -- Try exact match first
    local cost = MOVE_COST_EXACT[moveName]
    if cost then return cost end
    -- Try case-insensitive / stripped match
    local stripped = moveName:gsub("[%s%p]", ""):lower()
    for k, v in pairs(MOVE_COST_EXACT) do
        if k:gsub("[%s%p]", ""):lower() == stripped then return v end
    end
    return nil
end

local function randomWait(mn,mx) task.wait(mn+math.random()*(mx-mn)) end
local function maybeJump(humanoid)
    if math.random(1,3)==1 then humanoid.Jump=true task.wait(0.1) humanoid.Jump=false end
end
local function checkStuck(hrp,humanoid,target)
    local cur=hrp.Position
    if farmState.lastPos then
        if (cur-farmState.lastPos).Magnitude<0.5 then
            farmState.stuckCount+=1
            if farmState.stuckCount>=2 then
                humanoid.Jump=true task.wait(0.2) humanoid.Jump=false
                humanoid:MoveTo(cur+Vector3.new(math.random(-4,4),0,math.random(-4,4)))
                task.wait(0.3) humanoid:MoveTo(target) farmState.stuckCount=0
            end
        else farmState.stuckCount=0 end
    end
    farmState.lastPos=cur
end
local function walkTo(hrp,humanoid,target,timeout,blacklistKey,farmCheck)
    farmState.stuckCount=0 farmState.lastPos=nil
    humanoid:MoveTo(target)
    local t=tick()+(timeout or 20) local startPos=hrp.Position
    repeat
        task.wait(0.3)
        if farmCheck and not farmCheck() then return false end
        checkStuck(hrp,humanoid,target) maybeJump(humanoid)
    until (hrp.Position-target).Magnitude<8 or tick()>t
    local arrived=(hrp.Position-target).Magnitude<8
    if not arrived and blacklistKey and (hrp.Position-startPos).Magnitude<15 then
        farmState.trashBlacklist[blacklistKey]=true
    end
    return arrived
end

-- ── Underground movement ──────────────────────────────────────────────────────
local function goUnderground(hrp)
    if moveState.isUnderground then return end
    local dest=Vector3.new(hrp.Position.X,moveState.UNDERGROUND_Y,hrp.Position.Z)
    local dur=math.max((dest-hrp.Position).Magnitude/moveState.SAFE_SPEED,0.05)
    local tw=TweenService:Create(hrp,TweenInfo.new(dur,Enum.EasingStyle.Linear),{CFrame=CFrame.new(dest)})
    tw:Play() tw.Completed:Wait() moveState.isUnderground=true
end
local function surfaceUp(hrp)
    if not moveState.isUnderground then return end
    local dest=Vector3.new(hrp.Position.X,moveState.UNDERGROUND_Y+8,hrp.Position.Z)
    local dur=math.max((dest-hrp.Position).Magnitude/moveState.SAFE_SPEED,0.05)
    local tw=TweenService:Create(hrp,TweenInfo.new(dur,Enum.EasingStyle.Linear),{CFrame=CFrame.new(dest)})
    tw:Play() tw.Completed:Wait() moveState.isUnderground=false
end
local function touchGround(hrp)
    if moveState.isTouchingGround then return end
    moveState.isTouchingGround=true
    local d1=Vector3.new(hrp.Position.X,moveState.GROUND_Y,hrp.Position.Z)
    local t1=TweenService:Create(hrp,TweenInfo.new(math.max((d1-hrp.Position).Magnitude/moveState.SAFE_SPEED,0.05),Enum.EasingStyle.Linear),{CFrame=CFrame.new(d1)})
    t1:Play() t1.Completed:Wait() task.wait(0.1)
    local d2=Vector3.new(hrp.Position.X,moveState.UNDERGROUND_Y,hrp.Position.Z)
    local t2=TweenService:Create(hrp,TweenInfo.new(math.max((d2-hrp.Position).Magnitude/moveState.SAFE_SPEED,0.05),Enum.EasingStyle.Linear),{CFrame=CFrame.new(d2)})
    t2:Play() t2.Completed:Wait()
    moveState.lastGroundTouch=tick() moveState.isTouchingGround=false
end
local function blatantTweenTo(hrp,targetPos,farmCheck)
    local function makeTween()
        local dest=Vector3.new(targetPos.X,moveState.UNDERGROUND_Y,targetPos.Z)
        local dist=(dest-hrp.Position).Magnitude
        if dist<1 then return nil end
        return TweenService:Create(hrp,TweenInfo.new(math.max(dist/moveState.SAFE_SPEED,0.05),Enum.EasingStyle.Linear),{CFrame=CFrame.new(dest)})
    end
    local tween=makeTween() if not tween then return end
    local tweenDone=false
    tween.Completed:Connect(function() tweenDone=true end) tween:Play()
    while not tweenDone do
        task.wait(0.5)
        if not farmCheck() then tween:Cancel() return end
        if tick()-moveState.lastGroundTouch>=moveState.GROUND_TOUCH_INTERVAL and not moveState.isTouchingGround then
            tween:Cancel() touchGround(hrp)
            tween=makeTween() if not tween then return end
            tweenDone=false
            tween.Completed:Connect(function() tweenDone=true end) tween:Play()
        end
    end
end
local function blatantWalkTo(hrp,humanoid,target,timeout,blacklistKey,farmCheck)
    if not moveState.isUnderground then goUnderground(hrp) end
    blatantTweenTo(hrp,target,farmCheck)
    local arrived=(Vector3.new(hrp.Position.X,0,hrp.Position.Z)-Vector3.new(target.X,0,target.Z)).Magnitude<8
    if not arrived and blacklistKey then farmState.trashBlacklist[blacklistKey]=true end
    return arrived
end

-- ── Trash/Dumpster helpers ────────────────────────────────────────────────────
local function getTrashCount(waitForValid)
    local function read()
        local ok,label=pcall(function() return playerGui.HUD["Main HUD"].Corner.Trash.Label end)
        if not ok or not label then return 0,0 end
        local cur,max=label.Text:match("(%d+)%s*/%s*(%d+)")
        return tonumber(cur) or 0, tonumber(max) or 0
    end
    if waitForValid then
        local deadline=tick()+3 local c,m=read()
        while m==0 and tick()<deadline do task.wait(0.2) c,m=read() end
        return c,m
    end
    return read()
end
local function isDumpsterOnCooldown(part)
    local t=part:FindFirstChild("TimerGUI") return t and t.Enabled==true
end
local function getAvailableDumpster()
    local hrp=LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local best,bestDist=nil,math.huge
    for _,part in ipairs(workspace.Map.Environment.Dumpsters:GetChildren()) do
        if part:IsA("Part") and not isDumpsterOnCooldown(part) then
            local prompt=part:FindFirstChild("Prompt")
            if prompt and prompt.Enabled then
                local d=(hrp.Position-part.Position).Magnitude
                if d<bestDist then best={part=part,pos=part.Position} bestDist=d end
            end
        end
    end
    return best
end
local function getNextAvailableDumpsterTime()
    local soonest=nil
    for _,part in ipairs(workspace.Map.Environment.Dumpsters:GetChildren()) do
        if part:IsA("Part") then
            local tg=part:FindFirstChild("TimerGUI")
            if tg and tg.Enabled then
                local lbl=tg:FindFirstChild("Label")
                if lbl and lbl.Text~="" then
                    local m,s=lbl.Text:match("(%d+):(%d+)")
                    if m and s then
                        local secs=tonumber(m)*60+tonumber(s)
                        if not soonest or secs<soonest then soonest=secs end
                    end
                end
            else return 0 end
        end
    end
    return soonest or 0
end

-- ── Farm Loop ─────────────────────────────────────────────────────────────────
local function runFarmLoop(farmCheck,blatant)
    local function getChar()
        local c=LocalPlayer.Character if not c then return nil,nil end
        return c:FindFirstChild("HumanoidRootPart"),c:FindFirstChild("Humanoid")
    end
    local lastTrashCount=-1 local lastIncreaseTime=tick() local hasPickedUpAtLeastOne=false
    startAutoSprint(farmCheck)
    if blatant then startNoclip() moveState.lastGroundTouch=tick() end
    while farmCheck() do
        local hrp,humanoid=getChar()
        if not hrp or not humanoid then task.wait(1) continue end
        local cur,max=getTrashCount(true)
        if cur~=lastTrashCount then lastTrashCount=cur lastIncreaseTime=tick() end
        if tick()-lastIncreaseTime>10 then
            Notify("Trash Farm","Stuck! Resetting...",3)
            if moveState.isUnderground then surfaceUp(hrp) end
            LocalPlayer.Character:BreakJoints()
            task.wait(4) moveState.isUnderground=false
            startAutoSprint(farmCheck)
            if blatant then startNoclip() moveState.lastGroundTouch=tick() end
            lastIncreaseTime=tick() lastTrashCount=-1 hasPickedUpAtLeastOne=false
            continue
        end
        if hasPickedUpAtLeastOne and max>0 and cur>=max then
            if moveState.isUnderground then local h=getChar() if h then surfaceUp(h) end end
            local dumpster=getAvailableDumpster()
            if dumpster then
                local arrived=walkTo(hrp,humanoid,dumpster.pos,30,nil,farmCheck)
                if arrived then
                    randomWait(0.3,0.7)
                    local prompt=dumpster.part:FindFirstChild("Prompt")
                    if prompt then
                        fireproximityprompt(prompt,2) task.wait(2)
                        farmState.trashBlacklist={} lastTrashCount=-1
                        lastIncreaseTime=tick() hasPickedUpAtLeastOne=false
                    end
                end
            else
                local waitSecs=math.max(getNextAvailableDumpsterTime() or 60,5)
                Notify("Trash Farm","Dumpsters on cooldown! Waiting "..math.ceil(waitSecs).."s",5)
                local waitEnd=tick()+waitSecs+2
                repeat task.wait(2) if getAvailableDumpster() then break end
                until tick()>waitEnd or not farmCheck()
                lastIncreaseTime=tick()
            end
        else
            local hrp2,humanoid2=getChar()
            if not hrp2 or not humanoid2 then task.wait(1) continue end
            local nearest,nearestDist,nearestPrompt,nearestKey=nil,math.huge,nil,nil
            for _,paper in ipairs(workspace.Map.Environment.Trash:GetChildren()) do
                if farmState.trashBlacklist[paper.Name] then continue end
                local prompt=paper:FindFirstChild("Prompt")
                if prompt and prompt.Enabled then
                    local d=(hrp2.Position-paper.Position).Magnitude
                    if d<nearestDist then nearest=paper nearestDist=d nearestPrompt=prompt nearestKey=paper.Name end
                end
            end
            if nearest then
                local arrived
                if blatant then arrived=blatantWalkTo(hrp2,humanoid2,nearest.Position,20,nearestKey,farmCheck)
                else arrived=walkTo(hrp2,humanoid2,nearest.Position,20,nearestKey,farmCheck) end
                if arrived and nearestPrompt and nearestPrompt.Enabled then
                    fireproximityprompt(nearestPrompt) hasPickedUpAtLeastOne=true lastIncreaseTime=tick() task.wait(0.3)
                elseif not arrived then lastIncreaseTime=tick() end
            else task.wait(3) end
        end
        task.wait(0.1)
    end
    stopAutoSprint()
    if blatant then
        stopNoclip()
        local h=LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if h and moveState.isUnderground then surfaceUp(h) end
        moveState.isUnderground=false
    end
end

    local launch=hud:FindFirstChild("Launch") if not launch or not launch.Visible then return end
    local bar=launch:FindFirstChild("Bar") if not bar or not bar.Visible then return end
    return bar
end
local function isOutOfEnergy()
    local ok,label=pcall(function() return playerGui["Player Notifications"].Notifications.Message.Label end)
    if not ok or not label then return false end
    return label.Visible and string.find(label.Text,"out of energy")~=nil
end

-- ── Auto Launch Cycle ─────────────────────────────────────────────────────────
local function doAutoLaunchCycle()
    if launchState.isDoingCycle then return end
    launchState.isDoingCycle=true
    local character=LocalPlayer.Character
    local hrp=character and character:FindFirstChild("HumanoidRootPart")
    local humanoid=character and character:FindFirstChild("Humanoid")
    if not hrp or not humanoid then launchState.isDoingCycle=false return end
    if not launchState.savedCFrame then launchState.savedCFrame=hrp.CFrame end
    if isTopInArena() then
        local battleTimeout=tick()+180
        repeat task.wait(0.5) if not launchState.running then launchState.isDoingCycle=false return end
        until not isTopInArena() or tick()>battleTimeout
        if not launchState.running then launchState.isDoingCycle=false return end
        task.wait(1.5)
    end
    if getMyTop()~=nil then
        local pickupTimeout=tick()+30
        repeat task.wait(0.3) until topNeedsPickup() or topIsInHand() or tick()>pickupTimeout
        if topNeedsPickup() then
            pressKey(Enum.KeyCode.E) task.wait(0.8)
            if not topIsInHand() and topNeedsPickup() then
                Notify("Auto Launch","Too far! Walking to bey...",3)
                local topPos=getTopPosition()
                if topPos then
                    farmState.stuckCount=0 farmState.lastPos=nil
                    humanoid:MoveTo(topPos)
                    local wt=tick()+25
                    repeat
                        task.wait(0.3)
                        if not launchState.running then launchState.isDoingCycle=false return end
                        local np=getTopPosition() if np then topPos=np humanoid:MoveTo(topPos) end
                        checkStuck(hrp,humanoid,topPos) maybeJump(humanoid)
                    until (hrp.Position-(getTopPosition() or topPos)).Magnitude<5 or tick()>wt
                    task.wait(0.3) pressKey(Enum.KeyCode.E) task.wait(0.8)
                end
            end
        end
    end
    local handTimeout=tick()+10
    repeat task.wait(0.3) until topIsInHand() or tick()>handTimeout
    if not topIsInHand() then
        Notify("Auto Launch","Pickup failed, retrying...",3)
        launchState.isDoingCycle=false
        if launchState.running then task.wait(2) task.spawn(doAutoLaunchCycle) end
        return
    end
    if not launchState.running then launchState.isDoingCycle=false return end
    local standPos=launchState.savedCFrame.Position
    farmState.stuckCount=0 farmState.lastPos=nil humanoid:MoveTo(standPos)
    local returnTimeout=tick()+20
    repeat
        task.wait(0.3)
        if not launchState.running then launchState.isDoingCycle=false return end
        checkStuck(hrp,humanoid,standPos) maybeJump(humanoid)
    until (hrp.Position-standPos).Magnitude<4 or tick()>returnTimeout
    hrp.CFrame=launchState.savedCFrame task.wait(2)
    if not launchState.running then launchState.isDoingCycle=false return end
    pressKey(Enum.KeyCode.E) task.wait(1.5)
    if not launchState.running then launchState.isDoingCycle=false return end
    VirtualInputManager:SendMouseButtonEvent(0,0,0,true,game,1) task.wait(0.05)
    VirtualInputManager:SendMouseButtonEvent(0,0,0,false,game,1)
    launchState.isDoingCycle=false
    if launchState.running then task.wait(2) task.spawn(doAutoLaunchCycle) end
end

-- ── Burger Run ────────────────────────────────────────────────────────────────
local function doBurgerRun(manual)
    if farmState.isBurgerBuying then return end
    farmState.isBurgerBuying=true farmState.isManualRun=manual or false farmState.manualCancelled=false
    local character=LocalPlayer.Character
    local hrp=character and character:FindFirstChild("HumanoidRootPart")
    local humanoid=character and character:FindFirstChild("Humanoid")
    if not hrp or not humanoid then farmState.isBurgerBuying=false return end
    local originalPos=hrp.Position local standPos=Vector3.new(573.88,118,-157.43)
    local keyConnection
    if farmState.isManualRun then
        keyConnection=UserInputService.InputBegan:Connect(function(input,gp)
            if gp then return end
            if input.UserInputType==Enum.UserInputType.Keyboard then
                farmState.manualCancelled=true humanoid:MoveTo(hrp.Position)
            end
        end)
    end
    farmState.stuckCount=0 farmState.lastPos=nil humanoid:MoveTo(standPos)
    local timeout=tick()+12
    repeat
        task.wait(0.3) if farmState.manualCancelled then break end
        checkStuck(hrp,humanoid,standPos) maybeJump(humanoid)
    until (hrp.Position-standPos).Magnitude<3 or tick()>timeout or farmState.manualCancelled
    if not farmState.manualCancelled then
        randomWait(0.4,0.9)
        local ok,vendor=pcall(function() return workspace.Map["Food Vendors"]:GetChildren()[11] end)
        if ok and vendor then
            local prompt=vendor.Counter["Prompt Host"].Prompt
            if prompt then fireproximityprompt(prompt) end
        end
        if not farmState.isManualRun then
            randomWait(2.8,3.4)
            local returnPos=originalPos+Vector3.new(math.random(-2,2),0,math.random(-2,2))
            farmState.stuckCount=0 farmState.lastPos=nil humanoid:MoveTo(returnPos)
            local returnTimeout=tick()+12
            repeat task.wait(0.3) checkStuck(hrp,humanoid,returnPos) maybeJump(humanoid)
            until (hrp.Position-returnPos).Magnitude<3 or tick()>returnTimeout
        end
    end
    if keyConnection then keyConnection:Disconnect() end
    farmState.isBurgerBuying=false farmState.isManualRun=false farmState.manualCancelled=false
end

local function runACBypassTimer()

    local char=LocalPlayer.Character
    local humanoid=char and char:FindFirstChild("Humanoid")
    if humanoid then humanoid.WalkSpeed=0 humanoid.UseJumpPower=true humanoid.JumpPower=0 end
    local freezeConn=RunService.Heartbeat:Connect(function()
        local c=LocalPlayer.Character local h=c and c:FindFirstChild("Humanoid")
        if h then h.WalkSpeed=0 h.JumpPower=0 end
    end)
    local startTick=tick()
    if connections.acBypassTimer then connections.acBypassTimer:Disconnect() end
    connections.acBypassTimer=RunService.Heartbeat:Connect(function()
        if not acBypassGui then return end
        local elapsed=tick()-startTick local left=math.max(0,10-elapsed)
        acBypassGui.countdown.Text=tostring(math.ceil(left))
        local pct=left/10

        else acBypassGui.countdown.TextColor3=Color3.fromRGB(255,60,60) end

            connections.acBypassTimer:Disconnect() connections.acBypassTimer=nil
            freezeConn:Disconnect() charState.isFreezing=false stopNoclip()

        end

        VirtualInputManager:SendKeyEvent(true,Enum.KeyCode.E,false,game) task.wait(0.15)

local function onAutoReEngage(value)
    toggleStates.AutoReEngage=value
    if connections.autoReEngage then connections.autoReEngage:Disconnect() connections.autoReEngage=nil end
    reEngageState.cooldown=false
    if not value then Notify("Auto Re-Engage","Stopped.",3) return end
    reEngageState.wasInBattle=false reEngageState.watchedTop=nil
    local function hookTopHealth(top)
        if reEngageState.topHealthConn then reEngageState.topHealthConn:Disconnect() reEngageState.topHealthConn=nil end
        reEngageState.watchedTop=top if not top then return end
        reEngageState.topHealthConn=top.AttributeChanged:Connect(function(attr)
            if attr~="Health" then return end
            local health=top:GetAttribute("Health")
            if reEngageState.wasInBattle and (not health or health<=0) then
                reEngageState.wasInBattle=false doReEngage()
            elseif health and health>0 then reEngageState.wasInBattle=true end
        end)
    end
    connections.autoReEngage=RunService.Heartbeat:Connect(function()
        local top=getMyTop()
        if top~=reEngageState.watchedTop then
            hookTopHealth(top)
            if not top and reEngageState.wasInBattle then reEngageState.wasInBattle=false doReEngage() end
        end
        if top then local health=top:GetAttribute("Health") if health and health>0 then reEngageState.wasInBattle=true end end
    end)
    Notify("Auto Re-Engage","Started! Will press E×2 after each 1v1.",4)
end
toggleCallbacks["AutoReEngage"]=onAutoReEngage

-- ── Auto Switch State At RPM % ────────────────────────────────────────────────
local autoSwitchState = {
    enabled = false,
    threshold = 0.100,       -- 100% default
    switchTo = "Attacking", -- state to switch TO when threshold is crossed
    blacklist = {},         -- set of type strings to ignore
    hasSwitched = false,    -- one-shot per match: don't keep re-triggering
    wasInBattle = false,    -- tracks if we entered a battle this session
}

local function getOpponentTypeForSwitch()
    -- Primary: use the auto viewer's locked-in opponent top
    local oppTop = viewer.currentTop
    -- Fallback: scan workspace.Tops for any non-self top with health
    if not oppTop or not oppTop.Parent then
        local myId = tostring(LocalPlayer.UserId)
        local myTop = workspace.Tops:FindFirstChild(myId)
        local myHealth = myTop and myTop:GetAttribute("Health")
        if myTop and myHealth and myHealth > 0 then
            local ok0, myPivot = pcall(function() return myTop:GetPivot().Position end)
            if ok0 then
                local closestDist = math.huge
                for _, top in ipairs(workspace.Tops:GetChildren()) do
                    if top.Name == myId then continue end
                    local h = top:GetAttribute("Health")
                    if not h or h <= 0 then continue end
                    local ok, pivot = pcall(function() return top:GetPivot().Position end)
                    if ok then
                        local d = (myPivot - pivot).Magnitude
                        if d < 50 and d < closestDist then
                            closestDist = d
                            oppTop = top
                        end
                    end
                end
            end
        end
    end
    if not oppTop or not oppTop.Parent then return nil end
    local rawType = oppTop:GetAttribute("Type") or oppTop:GetAttribute("Style") or ""
    return tostring(rawType)
end

local function getCurrentStateLabel()
    local ok, label = pcall(function() return playerGui.HUD["Top UI"].Info.Top.State end)
    if not ok or not label then return nil end
    return label.Text
end

local function pressGToState(targetState)
    -- Figure out current state first
    local current = getCurrentStateLabel()
    if not current then return end
    -- State cycle order: Attacking -> Guarding -> Attacking (G toggles)
    -- Pressing G once: Attacking->Guarding, Guarding->Attacking
    -- So we just need to know if we're already there
    if current:find(targetState) then return end -- already in target state
    -- Press G once to toggle
    task.spawn(function()
        VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.G, false, game)
        task.wait(0.05)
        VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.G, false, game)
        task.wait(0.35)
        -- Verify and press again if needed (sometimes first press misses)
        local newState = getCurrentStateLabel()
        if newState and not newState:find(targetState) then
            VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.G, false, game)
            task.wait(0.05)
            VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.G, false, game)
        end
    end)
end

local autoSwitchConn = nil
local function startAutoSwitchState()
    if autoSwitchConn then autoSwitchConn:Disconnect() autoSwitchConn = nil end
    autoSwitchState.hasSwitched = false
    autoSwitchState.wasInBattle = false

    autoSwitchConn = RunService.Heartbeat:Connect(function()
        if not autoSwitchState.enabled then return end
        local myId = tostring(LocalPlayer.UserId)
        local myTop = workspace.Tops:FindFirstChild(myId)
        if not myTop then
            -- Top left arena — reset so next match works
            if autoSwitchState.wasInBattle then
                autoSwitchState.hasSwitched = false
                autoSwitchState.wasInBattle = false
            end
            return
        end
        local health = myTop:GetAttribute("Health") or 0
        local maxHealth = myTop:GetAttribute("MaxHealth") or 0
        if maxHealth <= 0 or health <= 0 then
            if autoSwitchState.wasInBattle then
                autoSwitchState.hasSwitched = false
                autoSwitchState.wasInBattle = false
            end
            return
        end
        -- We're in battle
        autoSwitchState.wasInBattle = true
        -- Already switched this match — don't trigger again until next match
        if autoSwitchState.hasSwitched then return end
        local myRPMPct = health / maxHealth  -- Health is proportional to RPM
        -- Only trigger when RPM drops TO or BELOW the threshold (falling through it)
        if myRPMPct > autoSwitchState.threshold then return end
        -- Check opponent type against blacklist
        local oppType = getOpponentTypeForSwitch()
        if oppType and oppType ~= "" and oppType ~= "?" then
            -- Normalize: trim and lower for comparison
            local oppLower = oppType:lower():gsub("%s+", "")
            for blacklisted, _ in pairs(autoSwitchState.blacklist) do
                local bLower = blacklisted:lower():gsub("%s+", "")
                if oppLower:find(bLower, 1, true) or bLower:find(oppLower, 1, true) then
                    -- Opponent type is blacklisted — don't switch, mark as "handled" for this match
                    autoSwitchState.hasSwitched = true
                    Notify("Auto Switch State", "Skipped — opponent is "..oppType.." (blacklisted)", 4)
                    return
                end
            end
        end
        -- Switch!
        autoSwitchState.hasSwitched = true
        local pctDisplay = math.floor(myRPMPct * 100)
        Notify("Auto Switch State", "Switching to "..autoSwitchState.switchTo.." at "..pctDisplay.."% RPM", 4)
        pressGToState(autoSwitchState.switchTo)
    end)
end

local function stopAutoSwitchState()
    if autoSwitchConn then autoSwitchConn:Disconnect() autoSwitchConn = nil end
    autoSwitchState.hasSwitched = false
    autoSwitchState.wasInBattle = false
end

MainLeft:AddToggle('AutoSwitchState', {
    Text = 'Auto Switch State At RPM %',
    Default = false,
    Callback = function(value)
        autoSwitchState.enabled = value
        if value then
            startAutoSwitchState()
            Notify("Auto Switch State", "Enabled! Will switch to "..autoSwitchState.switchTo.." at "..math.floor(autoSwitchState.threshold*100).."%", 4)
        else
            stopAutoSwitchState()
            Notify("Auto Switch State", "Disabled.", 3)
        end
    end,
})

    Text = 'Switch At RPM %',
    Min = 0, Max = 110, Default = 100,
    Rounding = 0, Suffix = '%',
    Callback = function(value)
        autoSwitchState.threshold = value / 100

local function formatTime(secs)
    secs=math.max(0,math.floor(secs))
    return string.format("%02d : %02d",math.floor(secs/60),secs%60)
end
local function isRealMeteor(child) return child:IsA("Model") or child:IsA("Part") or child:IsA("BasePart") end
local function getLiveMeteorCount()
    local count=0
    for _,child in ipairs(workspace.Map.Meteors:GetChildren()) do if isRealMeteor(child) then count+=1 end end
    return count
end
local function startMeteorTimer()
    buildMeteorTimerGui()
    local waitingTaskActive=false
    local function scheduleWaitingStatus()
        if waitingTaskActive then return end waitingTaskActive=true
        task.spawn(function()
            task.wait(30) if not waitingTaskActive then return end
            if not meteorESP.enabled then waitingTaskActive=false return end
            if getLiveMeteorCount()==0 and meteorESP.overrideText=="" then
                setMeteorStatus("Waiting for meteors to spawn...",Color3.fromRGB(180,180,180))
            end
            waitingTaskActive=false
        end)
    end
    if not meteorESP.lastCycleStart then
        local savedUnix=loadLastMeteorTime()
        if savedUnix then
            local secondsSince=(os.time()-savedUnix)%meteorESP.INTERVAL
            meteorESP.lastCycleStart=tick()-secondsSince meteorESP.cycleResultShown=false
            local remaining=math.floor(meteorESP.INTERVAL-secondsSince)
            Notify("Meteor Timer","Synced! ~"..math.floor(remaining/60).."m "..(remaining%60).."s remaining.",5)
        else meteorESP.lastCycleStart=tick() meteorESP.cycleResultShown=false end
    end
    local existingCount=getLiveMeteorCount()
    if existingCount>0 then meteorESP.overrideText="Spawned! ("..existingCount..")" meteorESP.overrideColor=Color3.fromRGB(100,255,100)
    else meteorESP.overrideText="" end
    meteorESP.isPlayerCountLow=getServerPlayerCount()<meteorESP.MIN_PLAYERS
    if meteorESP.isPlayerCountLow then
        setMeteorStatus("Paused – need "..meteorESP.MIN_PLAYERS.."+ players ("..getServerPlayerCount().." now)",Color3.fromRGB(200,180,0))
    end
    updateStatusLabel()
    connections.meteorTimerAdded=workspace.Map.Meteors.ChildAdded:Connect(function(child)
        if not isRealMeteor(child) then return end
        if not meteorESP.enabled then return end
        waitingTaskActive=false
        local count=getLiveMeteorCount()
        meteorESP.overrideText="Spawned! ("..count..")" meteorESP.overrideColor=Color3.fromRGB(100,255,100)
        setMeteorStatus(meteorESP.overrideText,meteorESP.overrideColor)
        meteorESP.lastCycleStart=tick() meteorESP.cycleResultShown=false saveLastMeteorTime(os.time())
    end)
    connections.meteorTimerRemoved=workspace.Map.Meteors.ChildRemoved:Connect(function(child)
        if not isRealMeteor(child) then return end
        if not meteorESP.enabled then return end
        local remaining=getLiveMeteorCount()
        if remaining>0 then
            meteorESP.overrideText="Spawned! ("..remaining..")" meteorESP.overrideColor=Color3.fromRGB(100,255,100)
            setMeteorStatus(meteorESP.overrideText,meteorESP.overrideColor)
        else
            meteorESP.overrideText=""
            if not meteorESP.isPlayerCountLow then setMeteorStatus("All meteors picked up",Color3.fromRGB(255,140,0)) scheduleWaitingStatus()
            else updateStatusLabel() end
        end
    end)
    connections.meteorTimer=RunService.Heartbeat:Connect(function()
        if not meteorESP.timerGui then return end
        local playerCount=getServerPlayerCount() local lowNow=playerCount<meteorESP.MIN_PLAYERS
        if lowNow~=meteorESP.isPlayerCountLow then
            meteorESP.isPlayerCountLow=lowNow
            if lowNow then Notify("Meteor Timer","Paused – only "..playerCount.." players.",4)
            else Notify("Meteor Timer","Resumed – "..playerCount.." players.",4) end
            updateStatusLabel()
        end
        if meteorESP.isPlayerCountLow then
            meteorESP.timerGui.countdown.Text="-- : -- ⏸" meteorESP.timerGui.countdown.TextColor3=Color3.fromRGB(200,180,0)
        else
            if not meteorESP.lastCycleStart then return end
            local elapsed=tick()-meteorESP.lastCycleStart local remaining=meteorESP.INTERVAL-elapsed
            if remaining<=0 then
                if not meteorESP.cycleResultShown then
                    meteorESP.cycleResultShown=true
                    if getLiveMeteorCount()==0 and meteorESP.overrideText=="" then setMeteorStatus("Not Spawned",Color3.fromRGB(255,80,80)) end
                    meteorESP.lastCycleStart=tick() meteorESP.cycleResultShown=false
                end
                meteorESP.timerGui.countdown.Text="00 : 00" meteorESP.timerGui.countdown.TextColor3=Color3.fromRGB(255,60,60)
            else
                meteorESP.cycleResultShown=false meteorESP.timerGui.countdown.Text=formatTime(remaining)
                local pct=remaining/meteorESP.INTERVAL
                if pct>0.5 then meteorESP.timerGui.countdown.TextColor3=Color3.fromRGB(180,120,255)
                elseif pct>0.2 then meteorESP.timerGui.countdown.TextColor3=Color3.fromRGB(255,165,0)
                else meteorESP.timerGui.countdown.TextColor3=Color3.fromRGB(255,60,60) end
            end
        end
        if meteorESP.timerGui.status.Text~=meteorESP.statusText then meteorESP.timerGui.status.Text=meteorESP.statusText end
    end)
end

local function getPartName(top,partName)
    local part=top:FindFirstChild(partName) if not part then return "?" end
    local n=part:GetAttribute("Name")
    if n and tostring(n)~="" then return tostring(n) end
    local iname=part.Name

    return "?"
end
local function getRPMBarColor(pct)
    if pct>=1.0 then return Color3.fromRGB(180,80,255)
    elseif pct>=0.50 then return Color3.fromRGB(0,200,255)
    elseif pct>=0.25 then return Color3.fromRGB(255,165,0)
    else return Color3.fromRGB(255,50,50) end
end
local function getLiveRPM(top)
    local state=tostring(top:GetAttribute("State") or "")
    if not LIVE_STATES[state] then return nil end
    local health=top:GetAttribute("Health")
    if health and health>0 then return math.floor(health*20) end return 0
end
local function getLiveMaxRPM(top)
    local mh=top:GetAttribute("MaxHealth")

    local vc=tonumber(top:GetAttribute("VictoryCents")) or 60
    return math.max(vc*RPM_MULT,1)
end

local PART_MODE_ATTRS = {"BeltMode","BodyMode","Mode","PartMode","CurrentMode","ActiveMode","ModeState"}

local function getTopLevelMode(top, partName)
    if not top or not partName then return nil end
    local attrName = partName.."Mode" -- e.g. "BeltMode", "BodyMode"
    local v = top:GetAttribute(attrName)
    if v ~= nil then return tostring(v) end
    return nil
end

local function getPartModeDisplay(part, baseName)
    if not part or not baseName then return nil end
    -- Find the mode attribute value
    local rawVal=nil
    for _,attr in ipairs(PART_MODE_ATTRS) do
        local v=part:GetAttribute(attr)
        if v~=nil then rawVal=tostring(v) break end
    end
    if not rawVal then return nil end
    -- Find matching display table
    for partKey,modeMap in pairs(PART_MODE_DISPLAY) do
        if baseName:lower():find(partKey:lower(),1,true) then
            local display=modeMap[rawVal]
            if display then return display end
        end
    end
    return nil
end

    local function getPartMode(part)
        if not part then return nil end
        -- Common attribute names the game might use for mode
        local modeAttrs={"Mode","PartMode","CurrentMode","State","ModeState","ActiveMode","Selected"}
        for _,attr in ipairs(modeAttrs) do
            local v=part:GetAttribute(attr)
            if v and tostring(v)~="" and tostring(v)~="nil" then return tostring(v) end
        end
        -- Check children for StringValues that look like mode indicators
        for _,child in ipairs(part:GetChildren()) do
            if child:IsA("StringValue") and (child.Name=="Mode" or child.Name=="PartMode" or child.Name=="CurrentMode" or child.Name=="ActiveMode") then
                if child.Value~="" then return child.Value end
            end
            if child:IsA("BoolValue") then
                -- e.g. "BladesOn" = true means Blades On mode
                local n=child.Name:lower()
                if n:find("on") or n:find("off") or n:find("mode") or n:find("lock") or n:find("active") then
                    return child.Name.."="..(child.Value and "true" or "false")
                end
            end
        end
        return nil
    end

    local function resolveMode(partInstanceName, rawMode)
        if not rawMode then return nil end
        -- Try to find a matching display table by checking if partInstanceName contains a known key
        for partKey, modeMap in pairs(PART_MODE_DISPLAY) do
            if partInstanceName:lower():find(partKey:lower(), 1, true) then
                local display=modeMap[rawMode]
                if display then return display end
                -- Try case-insensitive
                for k,v in pairs(modeMap) do
                    if k:lower()==rawMode:lower() then return v end
                end
                -- BoolValue format: "BladesOn=true"
                if rawMode:find("=") then
                    local boolVal=rawMode:match("=(.+)$")
                    if boolVal then
                        local d=modeMap[boolVal] or modeMap[boolVal:lower()]
                        if d then return d end
                    end
                end
                -- Return raw if we know this part has modes but can't map it
                return rawMode
            end
        end
        return nil
    end

    local function partAttr(pName)
        local part=top:FindFirstChild(pName) if not part then return "?" end
        -- Try Name attribute first (player tops)
        local n=part:GetAttribute("Name")
        if n and tostring(n)~="" and tostring(n)~="?" then return tostring(n) end
        -- Try instance name (NPC tops store name as instance name)
        local iname=part.Name
        if iname and iname~=pName and iname~="" then return iname end
        -- Try every attribute
        for attrName,attrVal in pairs(part:GetAttributes()) do
            local v=tostring(attrVal)
            local al=attrName:lower()
            if v~="" and v~="?" and al~="id" and al~="rarity" and al~="health"
            and al~="maxhealth" and al~="attack" and al~="defense" and al~="stamina"
            and al~="mode" and al~="partmode" and al~="currentmode" then
                return v
            end
        end
        -- Check children StringValues
        for _,child in ipairs(part:GetChildren()) do
            if child:IsA("StringValue") and (child.Name=="Name" or child.Name=="PartName") then
                if child.Value~="" and child.Value~="?" then return child.Value end
            end
        end
        task.wait(0.3)
        local n2=part:GetAttribute("Name")
        if n2 and tostring(n2)~="" and tostring(n2)~="?" then return tostring(n2) end
        local iname2=part.Name
        if iname2 and iname2~=pName and iname2~="" then return iname2 end
        return "?"
    end

    local function partAttrWithMode(pName)
        local part=top:FindFirstChild(pName) if not part then return "?" end
        local baseName=partAttr(pName)
        if baseName=="?" then return "?" end
        -- Check top-level attribute first (e.g. BeltMode, BodyMode stored on the top itself)
        local rawMode
        local topLevelAttr = pName.."Mode"
        local topLevelVal = top:GetAttribute(topLevelAttr)
        if topLevelVal ~= nil then rawMode = tostring(topLevelVal) end
        -- Fall back to checking the part's own attributes
        if not rawMode then rawMode = getPartMode(part) end
        if rawMode then
            local display = resolveMode(baseName, rawMode)
            if display then return baseName.." ("..display..")" end
        end
        return baseName
    end

-- ── Auto Opponent Viewer ──────────────────────────────────────────────────────
local function startAutoOpponentViewer()
    if viewer.autoConn then viewer.autoConn:Disconnect() viewer.autoConn=nil end
    if viewer._extraConns then
        for _,c in ipairs(viewer._extraConns) do pcall(function() c:Disconnect() end) end
        viewer._extraConns=nil
    end
    local lastOpponentName=nil
    local lastOpponentTop=nil
    local targetWatchConn=nil
    local myId=tostring(LocalPlayer.UserId)

    -- Returns the Name string to display for a top (handles NPC tops too)
    local function getOpponentNameFromTop(oppTop)
        if not oppTop then return nil end
        local uid=tonumber(oppTop.Name)
        if uid then
            local p=Players:GetPlayerByUserId(uid)
            if p then return p.Name end
            -- NPC with numeric ID — use the ID as display key
            return "NPC:"..oppTop.Name
        end
        local ov=oppTop:FindFirstChild("Owner")
        if ov and ov.Value then return ov.Value.Name end
        -- Non-numeric name top (NPC or named top)
        return oppTop.Name
    end

    -- Check if a top belongs to an NPC (not a real player)
    local function isNPCTop(oppTop)
        if not oppTop then return false end
        local uid=tonumber(oppTop.Name)
        if uid then
            return Players:GetPlayerByUserId(uid)==nil
        end
        local ov=oppTop:FindFirstChild("Owner")
        if ov and ov.Value then
            -- Owner exists and is a player instance — not NPC
            return false
        end
        -- No numeric name and no Owner player reference = NPC
        return true
    end

    local oppStoppedTime=nil
    local oppCycleOverride=false -- true when cycling is active, bypass the name lock
    local function openForOpponent(oppTop, oppName)
        local name = oppName or getOpponentNameFromTop(oppTop)
        if not name or name=="" then return end
        if name==lastOpponentName and oppTop==lastOpponentTop then return end
        -- If we already have a real player locked, don't replace with an NPC
        if lastOpponentName and lastOpponentTop then
            local currentIsNPC=isNPCTop(lastOpponentTop)
            local newIsNPC=oppTop and isNPCTop(oppTop) or true
            if not currentIsNPC and newIsNPC then return end
        end
        -- If new top is NPC but a real player top exists nearby, skip the NPC
        if oppTop and isNPCTop(oppTop) then
            local myTop=workspace.Tops:FindFirstChild(myId)
            if myTop then
                local ok0,myPivot=pcall(function() return myTop:GetPivot().Position end)
                if ok0 then
                    for _,top in ipairs(workspace.Tops:GetChildren()) do
                        if top.Name==myId then continue end
                        if isNPCTop(top) then continue end
                        local h=top:GetAttribute("Health")
                        if not h or h<=0 then continue end
                        local ok2,pivot=pcall(function() return top:GetPivot().Position end)
                        if ok2 and (myPivot-pivot).Magnitude<45 then
                            -- Real player top nearby, show that instead
                            local realName=getOpponentNameFromTop(top)
                            lastOpponentName=realName lastOpponentTop=top oppStoppedTime=nil
                            buildViewerGui(realName, top)
                            Notify("Auto Viewer","Opponent: "..realName, 4)
                            return
                        end
                    end
                end
            end
        end
        lastOpponentName=name lastOpponentTop=oppTop oppStoppedTime=nil
        if oppTop and oppTop.Parent then
            buildViewerGui(name, oppTop)
            Notify("Auto Viewer","Opponent: "..name, 4)
        else
            buildViewerGui("Waiting for beyblade...", nil)
        end
    end

    local function clearOpponent()
        lastOpponentName=nil
        lastOpponentTop=nil
        if viewer.autoEnabled then
            buildViewerGui("Waiting for beyblade...", nil)
        end
    end

    -- Given my top, hook its Target child to detect opponent
    local function hookMyTop(myTop)
        if targetWatchConn then targetWatchConn:Disconnect() targetWatchConn=nil end
        if not myTop then clearOpponent() return end

        local function handleTargetValue(targetChild)
            if not targetChild then return end
            local oppTop=targetChild.Value
            if oppTop and oppTop.Parent then
                openForOpponent(oppTop, nil)
            end
            -- Watch for value change (target assigned after child added)
            targetChild:GetPropertyChangedSignal("Value"):Connect(function()
                local opp=targetChild.Value
                if opp and opp.Parent then
                    openForOpponent(opp, nil)
                end
            end)
        end

        -- Check if Target already exists
        local existingTarget=myTop:FindFirstChild("Target")
        if existingTarget then handleTargetValue(existingTarget) end

        -- Watch for Target being added
        targetWatchConn=myTop.ChildAdded:Connect(function(child)
            if child.Name~="Target" then return end
            handleTargetValue(child)
        end)

        -- Watch for Target removed (match ended)
        myTop.ChildRemoved:Connect(function(child)
            if child.Name~="Target" then return end
            task.delay(2, function()
                if not viewer.autoEnabled then return end
                local t2=workspace.Tops:FindFirstChild(myId)
                if not t2 or not t2:FindFirstChild("Target") then
                    clearOpponent()
                end
            end)
        end)
    end

    -- Watch for my top appearing/disappearing
    local topsAddedConn=workspace.Tops.ChildAdded:Connect(function(top)
        if not viewer.autoEnabled then return end
        if top.Name==myId then task.wait(0.3) hookMyTop(top) end
    end)
    local topsRemovedConn=workspace.Tops.ChildRemoved:Connect(function(top)
        if not viewer.autoEnabled then return end
        if top.Name==myId then hookMyTop(nil) end
    end)

    -- Hook existing top immediately
    local existingTop=workspace.Tops:FindFirstChild(myId)
    if existingTop then hookMyTop(existingTop) end

    -- Watch for IsStadiumFight flipping true on my top so we scan immediately
    local stadiumFlagConn=nil
    local function hookStadiumFlag(myTop)
        if stadiumFlagConn then stadiumFlagConn:Disconnect() stadiumFlagConn=nil end
        if not myTop then return end
        stadiumFlagConn=myTop.AttributeChanged:Connect(function(attr)
            if attr~="IsStadiumFight" then return end
            if myTop:GetAttribute("IsStadiumFight")==true then
                proximityInterval=0.5 -- force immediate scan next heartbeat
                buildCycleList()
                if #cycleList>0 then
                    local entry=cycleList[1]
                    lastOpponentName=entry.name
                    lastOpponentTop=entry.top
                    oppStoppedTime=nil
                    buildViewerGui(entry.name,entry.top)
                end
            end
        end)
    end
    if existingTop then hookStadiumFlag(existingTop) end
    local oldTopsAddedConn=topsAddedConn
    workspace.Tops.ChildAdded:Connect(function(top)
        if top.Name==myId then task.wait(0.3) hookStadiumFlag(top) end
    end)

    -- VS label scanner: only fires when no current opponent locked in
    local myName=LocalPlayer.Name
    local myDisplay=LocalPlayer.DisplayName
    local function extractVsLabel(text)
        if not text or text=="" then return nil end
        local tl=text:lower()
        if not tl:find(" vs ") then return nil end
        local left,right=tl:match("^(.-)%s+vs%s+(.+)$")
        if not left or not right then return nil end
        left=left:gsub("^%s+",""):gsub("%s+$","")
        right=right:gsub("^%s+",""):gsub("%s+$","")
        local myL=myName:lower()
        local myDL=myDisplay:lower()
        -- Only proceed if one side is us (use contains check too in case of truncation)
        local oppL
        if left==myL or left==myDL or myL:find(left,1,true) or left:find(myL,1,true) or myDL:find(left,1,true) or left:find(myDL,1,true) then
            oppL=right
        elseif right==myL or right==myDL or myL:find(right,1,true) or right:find(myL,1,true) or myDL:find(right,1,true) or right:find(myDL,1,true) then
            oppL=left
        else return nil end
        -- Try to resolve to a real player name
        for _,p in ipairs(Players:GetPlayers()) do
            if p.Name:lower()==oppL or p.DisplayName:lower()==oppL then return p.Name end
        end
        -- Could be NPC — return as-is for top search
        return oppL
    end

    -- Try to find an NPC top matching a name string
    local function resolveTopByName(oppL)
        local lower=oppL:lower()
        -- Real player first
        for _,p in ipairs(Players:GetPlayers()) do
            if p.Name:lower()==lower or p.DisplayName:lower()==lower then
                return workspace.Tops:FindFirstChild(tostring(p.UserId))
            end
        end
        -- NPC tops — try every possible match
        for _,top in ipairs(workspace.Tops:GetChildren()) do
            if isNPCTop(top) then
                local topLower=top.Name:lower()
                -- Direct contains match either way
                if topLower:find(lower,1,true) or lower:find(topLower,1,true) then
                    return top
                end
                -- Word-token overlap (e.g. "park computer" vs "parkcomputer")
                local function tokens(s)
                    local t={} for w in s:gmatch("%a+") do t[#t+1]=w end return t
                end
                local oppTokens=tokens(lower)
                local topTokens=tokens(topLower)
                local matched=0
                for _,ot in ipairs(oppTokens) do
                    for _,tt in ipairs(topTokens) do
                        if ot==tt then matched+=1 end
                    end
                end
                if matched>0 then return top end
                -- Check Owner StringValue or ObjectValue name
                local ov=top:FindFirstChild("Owner")
                if ov then
                    local ovStr=""
                    if ov:IsA("StringValue") then ovStr=ov.Value:lower()
                    elseif ov:IsA("ObjectValue") and ov.Value then
                        ovStr=tostring(ov.Value.Name or ""):lower()
                    end
                    if ovStr~="" and (ovStr:find(lower,1,true) or lower:find(ovStr,1,true)) then
                        return top
                    end
                end
                -- Check NPC name attribute on the top itself
                local npcName=top:GetAttribute("Name") or top:GetAttribute("NPCName") or top:GetAttribute("ComputerName")
                if npcName then
                    local nl=tostring(npcName):lower()
                    if nl:find(lower,1,true) or lower:find(nl,1,true) then return top end
                end
            end
        end
        return nil
    end

    -- Find the closest non-self top in the same arena as MY top (not player position)
    local function getClosestOpponentTopInArena()
        local myTop=workspace.Tops:FindFirstChild(myId)
        if not myTop then return nil end
        local myHealth=myTop:GetAttribute("Health")
        if not myHealth or myHealth<=0 then return nil end
        local ok0,myPivot=pcall(function() return myTop:GetPivot().Position end)
        if not ok0 then return nil end
        -- Try to get actual arena radius from workspace
        local ARENA_RADIUS=35 -- conservative default
        pcall(function()
            -- Check if there's an arena model we can measure
            local stadiums=workspace.Map:FindFirstChild("Stadiums") or workspace.Map:FindFirstChild("Arenas") or workspace.Map:FindFirstChild("Stadium")
            if stadiums then
                for _,stadium in ipairs(stadiums:GetChildren()) do
                    local ok2,pos=pcall(function() return stadium:GetPivot().Position end)
                    if ok2 then
                        local d=(myPivot-pos).Magnitude
                        if d<60 then
                            -- Find bounding size of this stadium
                            local ok3,cf,size=pcall(function() return stadium:GetBoundingBox() end)
                            if ok3 and size then
                                ARENA_RADIUS=math.min(size.X,size.Z)/2*0.6
                            end
                            break
                        end
                    end
                end
            end
        end)
        ARENA_RADIUS=math.clamp(ARENA_RADIUS,20,45)
        local closest=nil local closestDist=math.huge
        -- Prefer real players over NPCs
        local closestReal=nil local closestRealDist=math.huge
        for _,top in ipairs(workspace.Tops:GetChildren()) do
            if top.Name==myId then continue end
            local health=top:GetAttribute("Health")
            if not health or health<=0 then continue end
            local ok,pivot=pcall(function() return top:GetPivot().Position end)
            if not ok then continue end
            local dist=(myPivot-pivot).Magnitude
            if dist>ARENA_RADIUS then continue end
            if not isNPCTop(top) then
                if dist<closestRealDist then closestReal=top closestRealDist=dist end
            else
                if dist<closestDist then closest=top closestDist=dist end
            end
        end
        -- Always prefer real player over NPC
        return closestReal or closest
    end

    local STOPPED_CLEAR_DELAY=4 -- seconds after 0 RPM before we unlock
    local scanInterval=0
    local guiScanConn=RunService.Heartbeat:Connect(function(dt)
        if not viewer.autoEnabled then return end
        -- Re-clear opponent only after they've been at 0 RPM for a few seconds
        -- Re-clear opponent only after they've been at 0 RPM for a few seconds
        if lastOpponentName and lastOpponentTop then
            local h=lastOpponentTop:GetAttribute("Health")
            local topGone=not lastOpponentTop.Parent
            local topStopped=(h ~= nil and h<=0)
            if topGone or topStopped then
                if not oppStoppedTime then oppStoppedTime=tick() end
                if tick()-oppStoppedTime >= STOPPED_CLEAR_DELAY then
                    lastOpponentName=nil lastOpponentTop=nil oppStoppedTime=nil
                end
                -- While in stopped-delay, block everything else from running
                return
            else
                oppStoppedTime=nil
            end
        else
            oppStoppedTime=nil
        end
        -- Only use VS scan as a fallback when we have no locked opponent
        if lastOpponentName then return end
        scanInterval+=dt
        if scanInterval<0.5 then return end
        scanInterval=0
        for _,gui in ipairs(playerGui:GetChildren()) do
            if not gui:IsA("ScreenGui") or not gui.Enabled then continue end
            local function collectLabels(inst, out)
                for _,child in ipairs(inst:GetChildren()) do
                    if child:IsA("TextLabel") and child.Visible then table.insert(out,child.Text) end
                    collectLabels(child,out)
                end
            end
            local labels={}
            collectLabels(gui,labels)
            for _,text in ipairs(labels) do
                local opp=extractVsLabel(text)
                if opp and opp~=lastOpponentName then
                    -- Try to find their top
                    local oppTop=resolveTopByName(opp)
                    openForOpponent(oppTop, oppTop and nil or opp)
                    return
                end
            end
        end
    end)

    -- Multi-opponent cycling state
    local cycleList={}
    local cycleIndex=1
    local cycleTimer=0
    local CYCLE_INTERVAL=3

    local function getAllOpponentsInArena()
        local myTop=workspace.Tops:FindFirstChild(myId)
        if not myTop then return {} end
        local myHealth=myTop:GetAttribute("Health")
        if not myHealth or myHealth<=0 then return {} end
        local ok0,myPivot=pcall(function() return myTop:GetPivot().Position end)
        if not ok0 then return {} end
        local ARENA_RADIUS=45
        local realPlayers={} local npcs={}
        for _,top in ipairs(workspace.Tops:GetChildren()) do
            if top.Name==myId then continue end
            local health=top:GetAttribute("Health")
            if not health or health<=0 then continue end
            local ok,pivot=pcall(function() return top:GetPivot().Position end)
            if not ok then continue end
            local dist=(myPivot-pivot).Magnitude
            if dist>ARENA_RADIUS then continue end
            local name=getOpponentNameFromTop(top)
            if isNPCTop(top) then
                table.insert(npcs,{top=top,name=name})
            else
                table.insert(realPlayers,{top=top,name=name})
            end
        end
        if #realPlayers>0 then return realPlayers end
        return npcs
    end

    local function buildCycleList()
        local found=getAllOpponentsInArena()
        if #found~=#cycleList then
            cycleList=found cycleIndex=1 cycleTimer=0 return true
        end
        for i,entry in ipairs(found) do
            if not cycleList[i] or cycleList[i].top~=entry.top then
                cycleList=found cycleIndex=1 cycleTimer=0 return true
            end
        end
        return false
    end

    local proximityInterval=0
    local proximityConn=RunService.Heartbeat:Connect(function(dt)
        if not viewer.autoEnabled then return end
        proximityInterval+=dt
        if proximityInterval<0.5 then return end
        proximityInterval=0

        -- If Target is set, the hookMyTop/Target watcher handles it — don't interfere
        local myTop=workspace.Tops:FindFirstChild(myId)
        local hasTarget=myTop and myTop:FindFirstChild("Target")
        local targetTop=hasTarget and hasTarget.Value
        if targetTop and targetTop.Parent then return end
        -- Only scan proximity if MY top is in a stadium fight OR has health (just launched)
        local myHealth=myTop and myTop:GetAttribute("Health")
        if not myTop or (myTop:GetAttribute("IsStadiumFight")~=true and (not myHealth or myHealth<=0)) then return end

        -- Scan arena for all spinning opponents
        local changed=buildCycleList()
        if #cycleList==0 then
            -- Nothing spinning in arena
            -- Only clear if we're not in the middle of the stopped-delay
            if lastOpponentName and not oppStoppedTime then
                lastOpponentName=nil lastOpponentTop=nil
                buildViewerGui("Waiting for beyblade...",nil)
            elseif lastOpponentName and oppStoppedTime then
                -- Waiting for delay to expire, don't touch anything
                return
            end
            return
        end

        -- Single opponent
        if #cycleList==1 then
            local entry=cycleList[1]
            if lastOpponentTop~=entry.top then
                lastOpponentName=entry.name
                lastOpponentTop=entry.top
                oppStoppedTime=nil
                buildViewerGui(entry.name,entry.top)
            end
            return
        end

        -- Multiple opponents: cycle only on timer, never mid-cycle
        cycleTimer+=0.5
        if cycleTimer>=CYCLE_INTERVAL then
            cycleTimer=0
            if cycleIndex>#cycleList then cycleIndex=1 end
            local entry=cycleList[cycleIndex]
            -- Only rebuild if actually switching to a different top
            if lastOpponentTop~=entry.top then
                lastOpponentName=entry.name
                lastOpponentTop=entry.top
                oppStoppedTime=nil
                buildViewerGui(entry.name,entry.top)
            end
            cycleIndex+=1
            if cycleIndex>#cycleList then cycleIndex=1 end
        end
    end)

    viewer.autoConn=guiScanConn
    viewer._extraConns={topsAddedConn,topsRemovedConn,proximityConn}
end

local function stopAutoOpponentViewer()
    if viewer.autoConn then viewer.autoConn:Disconnect() viewer.autoConn=nil end
    if viewer._extraConns then
        for _,c in ipairs(viewer._extraConns) do pcall(function() c:Disconnect() end) end
        viewer._extraConns=nil
    end
end

                VirtualInputManager:SendKeyEvent(false,Enum.KeyCode.F13,false,game)

        end)

        end)
        if success and result and result.data and #result.data>0 then
            local lowest=nil local lowestCount=math.huge
            for _,server in ipairs(result.data) do
                if server.id~=currentJobId and server.playing<lowestCount then lowest=server lowestCount=server.playing end

                task.wait(1) TeleportService:TeleportToPlaceInstance(placeId,lowest.id,LocalPlayer)
            else Notify("Join Lowest","No other servers found.",3) end
        else Notify("Join Lowest","Could not fetch server list.",3) end
    end)
end)

-- ── UI Toggle Keybind ─────────────────────────────────────────────────────────
local uiToggleKey=Enum.KeyCode.RightShift
local function setupUIKeybind()
    if connections.uiKeybind then connections.uiKeybind:Disconnect() end
    connections.uiKeybind=UserInputService.InputBegan:Connect(function(input,gp)
        if gp then return end
        if input.KeyCode==uiToggleKey then
            if Library.Shown~=nil then
                Library.Shown=not Library.Shown
                if Library.GUI then Library.GUI.Enabled=Library.Shown end
            elseif Library.Toggle then Library:Toggle()
            elseif Library.Window and Library.Window.GUI then
                Library.Window.GUI.Enabled=not Library.Window.GUI.Enabled
            end
        end
    end)
end
setupUIKeybind()
