local HttpService = game:GetService("HttpService")
local CONFIG_FILE = "TAT_Configs.json"

local function loadAllConfigs()
    local ok, data = pcall(function() return readfile(CONFIG_FILE) end)
    if not ok or not data or data == "" then return {} end
    local ok2, parsed = pcall(function() return HttpService:JSONDecode(data) end)
    if ok2 and type(parsed) == "table" then return parsed end
    return {}
end

local function saveAllConfigs(configs)
    local ok, encoded = pcall(function() return HttpService:JSONEncode(configs) end)
    if not ok then return false end
    local ok2 = pcall(function() writefile(CONFIG_FILE, encoded) end)
    return ok2
end

local function getMoveCost(moveName)
    if not moveName or moveName=="?" or moveName=="none" or moveName=="None" then return nil end
    local cost = MOVE_COST_EXACT[moveName]
    if cost then return cost end
    local stripped = moveName:gsub("[%s%p]",""):lower()
    for k,v in pairs(MOVE_COST_EXACT) do
        if k:gsub("[%s%p]",""):lower()==stripped then return v end
    end
    return nil
end

-- ── Shared helper functions (defined ONCE at top level) ───────────────────────
local function _topAttr(top, a)
    local v = top:GetAttribute(a)
    return v ~= nil and tostring(v) or "?"
end

local function _getPartMode(part)
    if not part then return nil end
    for _, a in ipairs({"Mode","PartMode","CurrentMode","State","ModeState","ActiveMode","Selected"}) do
        local v = part:GetAttribute(a)
        if v and tostring(v) ~= "" and tostring(v) ~= "nil" then return tostring(v) end
    end
    for _, child in ipairs(part:GetChildren()) do
        if child:IsA("StringValue") and (child.Name=="Mode" or child.Name=="PartMode" or child.Name=="CurrentMode" or child.Name=="ActiveMode") then
            if child.Value ~= "" then return child.Value end
        end
        if child:IsA("BoolValue") then
            local n = child.Name:lower()
            if n:find("on") or n:find("off") or n:find("mode") or n:find("lock") or n:find("active") then
                return child.Name .. "=" .. (child.Value and "true" or "false")
            end
        end
    end
    return nil
end

local function _resolveMode(partInstanceName, rawMode)
    if not rawMode then return nil end
    for partKey, modeMap in pairs(PART_MODE_DISPLAY) do
        if partInstanceName:lower():find(partKey:lower(), 1, true) then
            local display = modeMap[rawMode]
            if display then return display end
            for k, v in pairs(modeMap) do
                if k:lower() == rawMode:lower() then return v end
            end
            if rawMode:find("=") then
                local boolVal = rawMode:match("=(.+)$")
                if boolVal then
                    local d = modeMap[boolVal] or modeMap[boolVal:lower()]
                    if d then return d end
                end
            end
            return rawMode
        end
    end
    return nil
end

local function _partAttr(top, pName)
    local part = top:FindFirstChild(pName)
    if not part then return "?" end
    local n = part:GetAttribute("Name")
    if n and tostring(n) ~= "" and tostring(n) ~= "?" then return tostring(n) end
    local iname = part.Name
    if iname and iname ~= pName and iname ~= "" then return iname end
    for attrName, attrVal in pairs(part:GetAttributes()) do
        local v = tostring(attrVal)
        local al = attrName:lower()
        if v ~= "" and v ~= "?" and al ~= "id" and al ~= "rarity" and al ~= "health"
        and al ~= "maxhealth" and al ~= "attack" and al ~= "defense" and al ~= "stamina"
        and al ~= "mode" and al ~= "partmode" and al ~= "currentmode" then return v end
    end
    for _, child in ipairs(part:GetChildren()) do
        if child:IsA("StringValue") and (child.Name=="Name" or child.Name=="PartName") then
            if child.Value ~= "" and child.Value ~= "?" then return child.Value end
        end
    end
    return "?"
end

local function _partAttrWithMode(top, pName)
    local part = top:FindFirstChild(pName)
    if not part then return "?" end
    local baseName = _partAttr(top, pName)
    if baseName == "?" then return "?" end
    local rawMode
    local topLevelVal = top:GetAttribute(pName .. "Mode")
    if topLevelVal ~= nil then rawMode = tostring(topLevelVal) end
    if not rawMode then rawMode = _getPartMode(part) end
    if rawMode then
        local display = _resolveMode(baseName, rawMode)
        if display then return baseName .. " (" .. display .. ")" end
    end
    return baseName
end

local function randomWait(mn,mx) task.wait(mn+math.random()*(mx-mn)) end
local function maybeJump(humanoid)
    if math.random(1,3)==1 then humanoid.Jump=true task.wait(0.1) humanoid.Jump=false end
end
local function checkStuck(hrp, humanoid, target)
    local cur = hrp.Position
    if farmState.lastPos then
        if (cur-farmState.lastPos).Magnitude < 0.5 then
            farmState.stuckCount += 1
            if farmState.stuckCount >= 2 then
                humanoid.Jump=true task.wait(0.2) humanoid.Jump=false
                humanoid:MoveTo(cur+Vector3.new(math.random(-4,4),0,math.random(-4,4)))
                task.wait(0.3) humanoid:MoveTo(target) farmState.stuckCount=0
            end
        else farmState.stuckCount=0 end
    end
    farmState.lastPos = cur
end
local function walkTo(hrp, humanoid, target, timeout, blacklistKey, farmCheck)
    farmState.stuckCount=0 farmState.lastPos=nil
    humanoid:MoveTo(target)
    local t = tick()+(timeout or 20)
    local startPos = hrp.Position
    repeat
        task.wait(0.3)
        if farmCheck and not farmCheck() then return false end
        checkStuck(hrp,humanoid,target) maybeJump(humanoid)
    until (hrp.Position-target).Magnitude<8 or tick()>t
    local arrived = (hrp.Position-target).Magnitude < 8
    if not arrived and blacklistKey and (hrp.Position-startPos).Magnitude<15 then
        farmState.trashBlacklist[blacklistKey]=true
    end
    return arrived
end

-- ── Underground movement ──────────────────────────────────────────────────────
local function goUnderground(hrp)
    if moveState.isUnderground then return end
    local dest = Vector3.new(hrp.Position.X, moveState.UNDERGROUND_Y, hrp.Position.Z)
    local dur = math.max((dest-hrp.Position).Magnitude/moveState.SAFE_SPEED, 0.05)
    TweenService:Create(hrp,TweenInfo.new(dur,Enum.EasingStyle.Linear),{CFrame=CFrame.new(dest)}):Play()
    task.wait(dur) moveState.isUnderground=true
end
local function surfaceUp(hrp)
    if not moveState.isUnderground then return end
    local dest = Vector3.new(hrp.Position.X, moveState.UNDERGROUND_Y+8, hrp.Position.Z)
    local dur = math.max((dest-hrp.Position).Magnitude/moveState.SAFE_SPEED, 0.05)
    TweenService:Create(hrp,TweenInfo.new(dur,Enum.EasingStyle.Linear),{CFrame=CFrame.new(dest)}):Play()
    task.wait(dur) moveState.isUnderground=false
end
local function touchGround(hrp)
    if moveState.isTouchingGround then return end
    moveState.isTouchingGround=true
    local d1 = Vector3.new(hrp.Position.X, moveState.GROUND_Y, hrp.Position.Z)
    local t1 = TweenService:Create(hrp,TweenInfo.new(math.max((d1-hrp.Position).Magnitude/moveState.SAFE_SPEED,0.05),Enum.EasingStyle.Linear),{CFrame=CFrame.new(d1)})
    t1:Play() t1.Completed:Wait() task.wait(0.1)
    local d2 = Vector3.new(hrp.Position.X, moveState.UNDERGROUND_Y, hrp.Position.Z)
    local t2 = TweenService:Create(hrp,TweenInfo.new(math.max((d2-hrp.Position).Magnitude/moveState.SAFE_SPEED,0.05),Enum.EasingStyle.Linear),{CFrame=CFrame.new(d2)})
    t2:Play() t2.Completed:Wait()
    moveState.lastGroundTouch=tick() moveState.isTouchingGround=false
end
local function blatantTweenTo(hrp, targetPos, farmCheck)
    local function makeTween()
        local dest = Vector3.new(targetPos.X, moveState.UNDERGROUND_Y, targetPos.Z)
        local dist = (dest-hrp.Position).Magnitude
        if dist<1 then return nil end
        return TweenService:Create(hrp,TweenInfo.new(math.max(dist/moveState.SAFE_SPEED,0.05),Enum.EasingStyle.Linear),{CFrame=CFrame.new(dest)})
    end
    local tween = makeTween() if not tween then return end
    local tweenDone = false
    tween.Completed:Connect(function() tweenDone=true end) tween:Play()
    while not tweenDone do
        task.wait(0.5)
        if not farmCheck() then tween:Cancel() return end
        if tick()-moveState.lastGroundTouch>=moveState.GROUND_TOUCH_INTERVAL and not moveState.isTouchingGround then
            tween:Cancel() touchGround(hrp)
            tween=makeTween() if not tween then return end
            tweenDone=false
            tween.Completed:Connect(function() tweenDone=true end) tween:Play()
        end
    end
end
local function tweenWalkTo(hrp, target, timeout, blacklistKey, farmCheck)
    local dist = (hrp.Position - target).Magnitude
    if dist < 1 then return true end
    local duration = math.max(dist / 60, 0.05)
    local tweenDone = false
    local tween = TweenService:Create(hrp, TweenInfo.new(duration, Enum.EasingStyle.Linear), {CFrame = CFrame.new(target)})
    tween.Completed:Connect(function() tweenDone = true end)
    tween:Play()
    local t = tick() + (timeout or 30)
    repeat task.wait(0.1)
        if farmCheck and not farmCheck() then tween:Cancel() return false end
    until tweenDone or tick() > t
    local arrived = (hrp.Position - target).Magnitude < 8
    if not arrived and blacklistKey then farmState.trashBlacklist[blacklistKey] = true end
    return arrived
end
local function blatantWalkTo(hrp, humanoid, target, timeout, blacklistKey, farmCheck)
    if not moveState.isUnderground then goUnderground(hrp) end
    blatantTweenTo(hrp, target, farmCheck)
    local arrived = (Vector3.new(hrp.Position.X,0,hrp.Position.Z)-Vector3.new(target.X,0,target.Z)).Magnitude < 8
    if not arrived and blacklistKey then farmState.trashBlacklist[blacklistKey]=true end
    return arrived
end

-- ── Trash/Dumpster helpers ────────────────────────────────────────────────────
local function getTrashCount(waitForValid)
    local function read()
        local ok, label = pcall(function() return playerGui.HUD["Main HUD"].Corner.Trash.Label end)
        if not ok or not label then return 0,0 end
        local cur, max = label.Text:match("(%d+)%s*/%s*(%d+)")
        return tonumber(cur) or 0, tonumber(max) or 0
    end
    if waitForValid then
        local deadline=tick()+3 local c,m=read()
        while m==0 and tick()<deadline do task.wait(0.2) c,m=read() end
        return c,m
    end
    return read()
end
local function isDumpsterOnCooldown(part)
    local t = part:FindFirstChild("TimerGUI") return t and t.Enabled==true
end
local function getAvailableDumpster()
    local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local best, bestDist = nil, math.huge
    for _, part in ipairs(workspace.Map.Environment.Dumpsters:GetChildren()) do
        if part:IsA("Part") and not isDumpsterOnCooldown(part) then
            local prompt = part:FindFirstChild("Prompt")
            if prompt and prompt.Enabled then
                local d = (hrp.Position-part.Position).Magnitude
                if d<bestDist then best={part=part,pos=part.Position} bestDist=d end
            end
        end
    end
    return best
end
local function getNextAvailableDumpsterTime()
    local soonest = nil
    for _, part in ipairs(workspace.Map.Environment.Dumpsters:GetChildren()) do
        if part:IsA("Part") then
            local tg = part:FindFirstChild("TimerGUI")
            if tg and tg.Enabled then
                local lbl = tg:FindFirstChild("Label")
                if lbl and lbl.Text ~= "" then
                    local m, s = lbl.Text:match("(%d+):(%d+)")
                    if m and s then
                        local secs = tonumber(m)*60+tonumber(s)
                        if not soonest or secs<soonest then soonest=secs end
                    end
                end
            else return 0 end
        end
    end
    return soonest or 0
end

-- ── Farm Loop ─────────────────────────────────────────────────────────────────
local function runFarmLoop(farmCheck, blatant)
    local function getChar()
        local c = LocalPlayer.Character if not c then return nil,nil end
        return c:FindFirstChild("HumanoidRootPart"), c:FindFirstChild("Humanoid")
    end
    local lastTrashCount=-1 local lastIncreaseTime=tick() local hasPickedUp=false
    startAutoSprint(farmCheck)
    local antiGravConn = nil
    while farmCheck() do
        local hrp, humanoid = getChar()
        if not hrp or not humanoid then task.wait(1) continue end
        local cur, max = getTrashCount(true)
        if cur ~= lastTrashCount then lastTrashCount=cur lastIncreaseTime=tick() end
        if tick()-lastIncreaseTime > 10 then
            LocalPlayer.Character:BreakJoints()
            task.wait(4)
            startAutoSprint(farmCheck)
            lastIncreaseTime=tick() lastTrashCount=-1 hasPickedUp=false
            continue
        end
        if hasPickedUp and max>0 and cur>=max then
            if moveState.isUnderground then local h=getChar() if h then surfaceUp(h) end end
            local dumpster = getAvailableDumpster()
            if dumpster then
                local arrived = blatant and tweenWalkTo(hrp,dumpster.pos,30,nil,farmCheck) or walkTo(hrp,humanoid,dumpster.pos,30,nil,farmCheck)
                if arrived then
                    randomWait(0.3,0.7)
                    local prompt = dumpster.part:FindFirstChild("Prompt")
                    if prompt then
                        fireproximityprompt(prompt,2) task.wait(2)
                        farmState.trashBlacklist={} lastTrashCount=-1
                        lastIncreaseTime=tick() hasPickedUp=false
                    end
                end
            else
                local waitSecs = math.max(getNextAvailableDumpsterTime() or 60, 5)
                local waitEnd = tick()+waitSecs+2
                repeat task.wait(2) if getAvailableDumpster() then break end
                until tick()>waitEnd or not farmCheck()
                lastIncreaseTime=tick()
            end
        else
            local hrp2, humanoid2 = getChar()
            if not hrp2 or not humanoid2 then task.wait(1) continue end
            local nearest, nearestDist, nearestPrompt, nearestKey = nil, math.huge, nil, nil
            for _, paper in ipairs(workspace.Map.Environment.Trash:GetChildren()) do
                if farmState.trashBlacklist[paper.Name] then continue end
                local prompt = paper:FindFirstChild("Prompt")
                if prompt and prompt.Enabled then
                    local d = (hrp2.Position-paper.Position).Magnitude
                    if d<nearestDist then nearest=paper nearestDist=d nearestPrompt=prompt nearestKey=paper.Name end
                end
            end
            if nearest then
                local arrived
                if blatant then arrived=tweenWalkTo(hrp2,nearest.Position,20,nearestKey,farmCheck)
                else arrived=walkTo(hrp2,humanoid2,nearest.Position,20,nearestKey,farmCheck) end
                if arrived and nearestPrompt and nearestPrompt.Enabled then
                    fireproximityprompt(nearestPrompt) hasPickedUp=true lastIncreaseTime=tick() task.wait(0.3)
                elseif not arrived then lastIncreaseTime=tick() end
            else task.wait(3) end
        end
        task.wait(0.1)
    end
    stopAutoSprint()
end

-- ── Auto Launch Cycle ─────────────────────────────────────────────────────────
local function doAutoLaunchCycle()
    if launchState.isDoingCycle then return end
    launchState.isDoingCycle=true
    local character=LocalPlayer.Character
    local hrp=character and character:FindFirstChild("HumanoidRootPart")
    local humanoid=character and character:FindFirstChild("Humanoid")
    if not hrp or not humanoid then launchState.isDoingCycle=false return end
    if not launchState.savedCFrame then launchState.savedCFrame=hrp.CFrame end
    if isTopInArena() then
        local battleTimeout=tick()+180
        repeat task.wait(0.5) if not launchState.running then launchState.isDoingCycle=false return end
        until not isTopInArena() or tick()>battleTimeout
        if not launchState.running then launchState.isDoingCycle=false return end
        task.wait(1.5)
    end
    if getMyTop()~=nil then
        local pickupTimeout=tick()+30
        repeat task.wait(0.3) until topNeedsPickup() or topIsInHand() or tick()>pickupTimeout
        if topNeedsPickup() then
            pressKey(Enum.KeyCode.E) task.wait(0.8)
            if not topIsInHand() and topNeedsPickup() then
                local topPos=getTopPosition()
                if topPos then
                    farmState.stuckCount=0 farmState.lastPos=nil
                    humanoid:MoveTo(topPos)
                    local wt=tick()+25
                    repeat
                        task.wait(0.3)
                        if not launchState.running then launchState.isDoingCycle=false return end
                        local np=getTopPosition() if np then topPos=np humanoid:MoveTo(topPos) end
                        checkStuck(hrp,humanoid,topPos) maybeJump(humanoid)
                    until (hrp.Position-(getTopPosition() or topPos)).Magnitude<5 or tick()>wt
                    task.wait(0.3) pressKey(Enum.KeyCode.E) task.wait(0.8)
                end
            end
        end
    end
    local handTimeout=tick()+10
    repeat task.wait(0.3) until topIsInHand() or tick()>handTimeout
    if not topIsInHand() then
        launchState.isDoingCycle=false
        if launchState.running then task.wait(2) task.spawn(doAutoLaunchCycle) end
        return
    end
    if not launchState.running then launchState.isDoingCycle=false return end
    local standPos=launchState.savedCFrame.Position
    farmState.stuckCount=0 farmState.lastPos=nil humanoid:MoveTo(standPos)
    local returnTimeout=tick()+20
    repeat
        task.wait(0.3)
        if not launchState.running then launchState.isDoingCycle=false return end
        checkStuck(hrp,humanoid,standPos) maybeJump(humanoid)
    until (hrp.Position-standPos).Magnitude<4 or tick()>returnTimeout
    hrp.CFrame=launchState.savedCFrame task.wait(2)
    if not launchState.running then launchState.isDoingCycle=false return end
    pressKey(Enum.KeyCode.E) task.wait(1.5)
    if not launchState.running then launchState.isDoingCycle=false return end
    VirtualInputManager:SendMouseButtonEvent(0,0,0,true,game,1) task.wait(0.05)
    VirtualInputManager:SendMouseButtonEvent(0,0,0,false,game,1)
    launchState.isDoingCycle=false
    if launchState.running then task.wait(2) task.spawn(doAutoLaunchCycle) end
end

    toggleStates.AutoPerfectLaunch=value
    if connections.perfect then connections.perfect:Disconnect() connections.perfect=nil end
    if not value then return end
    local passCount=0 local wasInZone=false
    local lastTrackerX=nil local barWasVisible=false
    local fireAtX=nil local armed=false
    connections.perfect=RunService.Heartbeat:Connect(function()
        local now=tick() if now-clickState.last<clickState.COOLDOWN then return end
        local bar=getBar()
        if not bar then
            if barWasVisible then
                passCount=0 wasInZone=false lastTrackerX=nil barWasVisible=false fireAtX=nil armed=false
            end
            return
        end
        barWasVisible=true
        local goal=bar:FindFirstChild("Goal") local tracker=bar:FindFirstChild("Tracker")
        if not goal or not tracker then return end
        local tc=tracker.AbsolutePosition.X+tracker.AbsoluteSize.X/2
        local gc=goal.AbsolutePosition.X+goal.AbsoluteSize.X/2
        local halfZone=goal.AbsoluteSize.X/2
        local inZone=math.abs(tc-gc)<halfZone
        if inZone and not wasInZone then
            passCount+=1
            if passCount==1 then
                local spread=halfZone*0.40
                fireAtX=gc+(math.random()*2-1)*spread
            elseif passCount==2 then
                armed=true
            end
        end
        if not inZone and wasInZone then
            if passCount==1 then
                local spread=halfZone*0.40
                fireAtX=gc+(math.random()*2-1)*spread
            end
        end
        wasInZone=inZone
        lastTrackerX=tc
        if armed and inZone and fireAtX then
            if math.abs(tc-fireAtX)<halfZone*0.5 then
                passCount=0 fireAtX=nil armed=false clickState.last=now doClick()
            end
        end
    end)
end

        end
        local health = myTop:GetAttribute("Health") or 0
        local maxHealth = myTop:GetAttribute("MaxHealth") or 0
        if maxHealth<=0 or health<=0 then
            if autoSwitchState.wasInBattle then autoSwitchState.hasSwitched=false autoSwitchState.wasInBattle=false end
            return
        end
        autoSwitchState.wasInBattle=true
        if autoSwitchState.hasSwitched then return end
        local myRPMPct = health/maxHealth
        local oppType = resolveOpponentType(getOpponentTypeForSwitch())
        if not oppType then return end
        if autoSwitchState.blacklist[oppType] then return end
        local cfg = autoSwitchState.typeConfig[oppType]
        if not cfg or not cfg.active then return end
        if cfg.threshold >= 1.10 then
            -- 110% means instant, no RPM check needed
        elseif myRPMPct > cfg.threshold then return end
        autoSwitchState.hasSwitched=true
        pressGToState(autoSwitchState.switchTo)
    end)
end
local function stopAutoSwitchState()
    if autoSwitchConn then autoSwitchConn:Disconnect() autoSwitchConn=nil end
    autoSwitchState.hasSwitched=false autoSwitchState.wasInBattle=false
end

        local myId = tostring(LocalPlayer.UserId)

            if not autoStyleState._resetTimer then autoStyleState._resetTimer=tick() end

            autoStyleState._resetTimer=nil wasInFight=false lastDetectedOpponent=nil

            local uid = tonumber(oppTop.Name)

                    for _, child in ipairs(inst:GetChildren()) do
                        if child:IsA("TextLabel") and child.Visible then
                            local tl = (child.Text or ""):lower()
                            if tl:find(" vs ") then
                                local left, right = tl:match("^(.-)%s+vs%s+(.+)$")
                                if left and right then
                                    left=left:gsub("^%s+",""):gsub("%s+$","")
                                    right=right:gsub("^%s+",""):gsub("%s+$","")
                                    local myL=myName:lower() local myDL=myDisplay:lower()
                                    local oppL
                                    if left==myL or left==myDL or myL:find(left,1,true) or left:find(myL,1,true) then oppL=right

                                    end
                                end
                            end
                        end
                        scanLabels(child)
                    end
                end
                scanLabels(gui)
                if oppName then break end
            end
        end
        if not oppName then
            if lastDetectedOpponent~=nil then lastDetectedOpponent=nil autoStyleState.hasSwitched=false end
            return
        end

        end

    end)

-- ── Approach tracker ──────────────────────────────────────────────────────────
-- We track a rolling window of distances so we can tell if the beys
-- are genuinely closing in on each other vs just orbiting nearby.
local approach = {
    history     = {},   -- ring buffer of last N distances
    MAX_HISTORY = 6,    -- frames to look back
    NEAR_DIST   = 12,   -- studs: "they are close"
    CLOSE_DIST  = 8,    -- studs: "impact imminent"
    triggered   = false,-- so we only fire once per approach
}

local function resetApproach()
    approach.history = {}
    approach.triggered = false
end

-- Returns true the FIRST time the beys cross CLOSE_DIST while already
-- having been within NEAR_DIST for at least a few frames.
local function checkApproach(myTop, oppTop)
    if not myTop or not oppTop then resetApproach() return false end

    local ok1, myPos  = pcall(function() return myTop:GetPivot().Position end)
    local ok2, oppPos = pcall(function() return oppTop:GetPivot().Position end)
    if not ok1 or not ok2 then return false end

    local dist = (myPos - oppPos).Magnitude

    -- push to history ring
    table.insert(approach.history, dist)
    if #approach.history > approach.MAX_HISTORY then
        table.remove(approach.history, 1)
    end

    -- need at least a few samples
    if #approach.history < 3 then return false end

    -- already fired this approach cycle
    if approach.triggered then
        -- reset trigger once they separate again so next approach works
        if dist > approach.NEAR_DIST then
            approach.triggered = false
        end
        return false
    end

    -- check: distance is shrinking AND we just crossed into CLOSE_DIST
    local prev = approach.history[#approach.history - 1]
    local shrinking = dist < prev
    local closeEnough = dist <= approach.CLOSE_DIST

    -- also require we spent a couple frames in NEAR range already
    local hadNearFrames = false
    for i = 1, #approach.history - 1 do
        if approach.history[i] <= approach.NEAR_DIST then
            hadNearFrames = true
            break
        end
    end

    if closeEnough and shrinking and hadNearFrames then
        approach.triggered = true
        return true
    end

    return false
end

local function getOppTopGS()
    local oppTop = viewer.currentTop
    if oppTop and oppTop.Parent then return oppTop end
    local myTop = getMyTopGS()
    if not myTop then return nil end
    local ok0, myPivot = pcall(function() return myTop:GetPivot().Position end)
    if not ok0 then return nil end
    local closest, closestDist = nil, math.huge
    for _, top in ipairs(workspace.Tops:GetChildren()) do
        if top.Name == tostring(LocalPlayer.UserId) then continue end
        local h = top:GetAttribute("Health")
        if not h or h <= 0 then continue end
        local ok, pivot = pcall(function() return top:GetPivot().Position end)
        if ok then
            local d = (myPivot - pivot).Magnitude
            if d < 80 and d < closestDist then closestDist = d closest = top end
        end
    end
    return closest
end

-- ── Reset helper called between fights ───────────────────────────────────────
local function resetFightState()
    guardSwitch.wasInFight    = false
    guardSwitch.disabledByRPM = false
    guardSwitch.inGuard       = false
    guardSwitch.isSwitching   = false
    guardSwitch.lastCycleTime = 0
    resetApproach()
end

-- ── Stop ─────────────────────────────────────────────────────────────────────
local function stopGuardSwitch()
    if guardSwitch.conn then
        guardSwitch.conn:Disconnect()
        guardSwitch.conn = nil
    end
    resetFightState()
    task.spawn(function()
        task.wait(0.05)
        if getCurrentStateGS() == "Guarding" then
            forceState("Attacking")
        end
    end)
end

-- ── Barrage cycle ─────────────────────────────────────────────────────────────
-- Triggered when beys are about to collide.
-- Guard BEFORE hit → stay in Guard through hit → Attack after.
local function doBarrageCycle()
    if guardSwitch.isSwitching then return end
    guardSwitch.isSwitching = true
    guardSwitch.lastCycleTime = tick()   -- stamp NOW so interval resets immediately

    task.spawn(function()
        -- Enter Guard before the hit
        forceState("Guarding")
        guardSwitch.inGuard = true

        -- Hold Guard long enough for the collision to land and resolve
        -- (0.45s is the post-hit window; tweak if needed)
        task.wait(0.45)

        -- Return to Attack
        forceState("Attacking")
        guardSwitch.inGuard     = false
        guardSwitch.isSwitching = false
    end)
end

-- ── Smash cycle ───────────────────────────────────────────────────────────────
-- Guard very briefly then snap back to Attack right before actual contact.
local function doSmashCycle()
    if guardSwitch.isSwitching then return end
    guardSwitch.isSwitching = true
    guardSwitch.lastCycleTime = tick()

    task.spawn(function()
        forceState("Guarding")
        guardSwitch.inGuard = true

        task.wait(0.12)   -- very short window then snap back before impact

        forceState("Attacking")
        guardSwitch.inGuard     = false
        guardSwitch.isSwitching = false
    end)
end

-- ── Main loop ─────────────────────────────────────────────────────────────────
local function startGuardSwitch()
    stopGuardSwitch()

    guardSwitch.conn = RunService.Heartbeat:Connect(function()
        if not guardSwitch.enabled then return end

        -- ── Fight gate ────────────────────────────────────────────────────────
        local myTop = getMyTopGS()
        if not myTop then
            if guardSwitch.wasInFight then resetFightState() end
            return
        end

        local inFight = myTop:GetAttribute("IsStadiumFight") == true
        if not inFight then
            if guardSwitch.wasInFight then resetFightState() end
            return
        end
        guardSwitch.wasInFight = true

        local myHealth = myTop:GetAttribute("Health") or 0
        if myHealth <= 0 then return end

        -- ── Opponent type gate ────────────────────────────────────────────────
        if getOppTypeGS() ~= "Attack" then
            if guardSwitch.inGuard then
                task.spawn(function() forceState("Attacking") end)
                guardSwitch.inGuard = false
            end
            return
        end

        local oppTop = getOppTopGS()
        if not oppTop then return end

        -- ── RPM Deficit check ─────────────────────────────────────────────────
        local myPct   = getRPMPct(myTop)
        local oppPct  = getRPMPct(oppTop)
        local deficit = oppPct - myPct   -- positive = we are behind

        if deficit > guardSwitch.rpmThreshold then
            if not guardSwitch.disabledByRPM then
                guardSwitch.disabledByRPM = true
                if guardSwitch.inGuard then
                    task.spawn(function() forceState("Attacking") end)
                    guardSwitch.inGuard = false
                end
            end
        else
            if guardSwitch.disabledByRPM then
                guardSwitch.disabledByRPM = false
            end
        end

        if guardSwitch.disabledByRPM then return end
        if guardSwitch.isSwitching   then return end

        -- ── Interval gate ─────────────────────────────────────────────────────
        -- Must be at least switchInterval seconds since the last cycle started.
        local now = tick()
        if (now - guardSwitch.lastCycleTime) < guardSwitch.switchInterval then return end

        -- ── Approach check ────────────────────────────────────────────────────
        -- Only fire when beys are genuinely closing in on each other.
        local aboutToCollide = checkApproach(myTop, oppTop)
        if not aboutToCollide then return end

        -- ── Fire the correct cycle ────────────────────────────────────────────
        if guardSwitch.mode == "Smash" then
            doSmashCycle()
        else
            doBarrageCycle()
        end
    end)
end

local function formatTime(secs)
    secs=math.max(0,math.floor(secs))
    return string.format("%02d : %02d", math.floor(secs/60), secs%60)
end

-- ── Previous Top Snapshot System ──────────────────────────────────────────────
local function snapshotTop(top)
    if not top or not top.Parent then return nil end
    local uid = tonumber(top.Name)
    local playerName = nil
    if uid then
        local p = Players:GetPlayerByUserId(uid)
        if p then playerName = p.Name end
    end
    if not playerName then return nil end
    local function safePart(inst, partName)
        local part = inst:FindFirstChild(partName) if not part then return "?" end
        local n = part:GetAttribute("Name")
        if n and tostring(n) ~= "" then return tostring(n) end
        return part.Name ~= partName and part.Name or "?"
    end
    local function safePartWithMode(inst, partName)
        local baseName = safePart(inst, partName) if baseName == "?" then return "?" end
        local rawMode = _getPartMode(inst:FindFirstChild(partName))
        if rawMode then local display = _resolveMode(baseName, rawMode) if display then return baseName.." ("..display..")" end end
        return baseName
    end
    local maxHealth = tonumber(top:GetAttribute("MaxHealth")) or 0
    local maxRPM = maxHealth > 0 and math.floor(maxHealth * 20) or 0
    local function resolveMoveSnap(attrName)
        local val = _topAttr(top, attrName) if val == "?" then return "None" end
        return val:gsub("(%l)(%u)", "%1 %2"):gsub("(%u)(%u%l)", "%1 %2")
    end
    local move1 = resolveMoveSnap("SignatureMove_1")
    local move2 = resolveMoveSnap("SignatureMove_2")
    local move3 = resolveMoveSnap("SignatureMove_3")
    local rawType = top:GetAttribute("Type") or top:GetAttribute("Style") or "?"
    return {
        playerName = playerName,
        spirit     = _topAttr(top, "Spirit"),
        crown      = safePart(top, "Crown"),
        body       = safePartWithMode(top, "Body"),
        belt       = safePartWithMode(top, "Belt"),
        point      = safePart(top, "Point"),
        move1=move1, move2=move2, move3=move3,
        moveCost1=getMoveCost(move1), moveCost2=getMoveCost(move2), moveCost3=getMoveCost(move3),
        typeStr    = tostring(rawType),
        attack     = _topAttr(top, "Attack"),
        defense    = _topAttr(top, "Defense"),
        stamina    = _topAttr(top, "Stamina"),
        maxRPM     = maxRPM,
    }
end

local function startSnapshotTracker()
    local trackedTops = {}
    local function trackTop(top)
        if trackedTops[top] then return end
        trackedTops[top] = true
        task.wait(0.001)
        if not top or not top.Parent then trackedTops[top]=nil return end
        local uid = tonumber(top.Name)
        if not uid or not Players:GetPlayerByUserId(uid) then trackedTops[top]=nil return end
        -- Take launch snapshot as currentBuild
        local currentBuild = snapshotTop(top)
        if not currentBuild then trackedTops[top]=nil return end
        local playerName = currentBuild.playerName
        -- Each new launch: the snapshot from last launch becomes "previous", current launch becomes "current"
        if previousTopSnapshots[playerName] then
            -- Shift: last current becomes previous
            previousTopSnapshots[playerName] = previousTopSnapshots[playerName .. "_current"] or previousTopSnapshots[playerName]
        end
        -- Save this launch as the new current
        previousTopSnapshots[playerName .. "_current"] = currentBuild
        -- If no previous yet, use this launch as both
        if not previousTopSnapshots[playerName] then
            previousTopSnapshots[playerName] = currentBuild
        end
local swapping = false

        local function saveAndUpdate()
            if swapping then return end
            swapping = true
            previousTopSnapshots[playerName] = {
                playerName = currentBuild.playerName,
                spirit     = currentBuild.spirit,
                crown      = currentBuild.crown,
                body       = currentBuild.body,
                belt       = currentBuild.belt,
                point      = currentBuild.point,
                move1      = currentBuild.move1,
                move2      = currentBuild.move2,
                move3      = currentBuild.move3,
                moveCost1  = currentBuild.moveCost1,
                moveCost2  = currentBuild.moveCost2,
                moveCost3  = currentBuild.moveCost3,
                typeStr    = currentBuild.typeStr,
                attack     = currentBuild.attack,
                defense    = currentBuild.defense,
                stamina    = currentBuild.stamina,
                maxRPM     = currentBuild.maxRPM,
            }
            task.delay(0.001, function()
                if not top or not top.Parent then swapping = false return end
                local newSnap = snapshotTop(top)
                if newSnap then
                    currentBuild = newSnap
                end
                swapping = false
            end)
        end

        top.ChildRemoved:Connect(function(child)
            local partNames = {Crown=true, Body=true, Belt=true, Point=true}
            if not partNames[child.Name] then return end
            saveAndUpdate()
        end)

        top.ChildAdded:Connect(function(child)
            local partNames = {Crown=true, Body=true, Belt=true, Point=true}
            if not partNames[child.Name] then return end
            saveAndUpdate()
        end)

        local function watchPartName(part)
            if not part or not part:IsA("BasePart") then return end
            part.AttributeChanged:Connect(function(attrName)
                if attrName ~= "Name" then return end
                saveAndUpdate()
            end)
        end
        for _, child in ipairs(top:GetChildren()) do
            watchPartName(child)
        end
        top.ChildAdded:Connect(function(child) watchPartName(child) end)

        top.AttributeChanged:Connect(function(attrName)
            local watched = {Spirit=true,SignatureMove_1=true,SignatureMove_2=true,SignatureMove_3=true}
            if not watched[attrName] then return end
            saveAndUpdate()
        end)
        top.AncestryChanged:Connect(function()
            if not top.Parent then trackedTops[top]=nil end
        end)
    end
    workspace.Tops.ChildAdded:Connect(function(top)
        task.spawn(trackTop, top)
    end)
    for _, top in ipairs(workspace.Tops:GetChildren()) do
        task.spawn(trackTop, top)
    end
end

local function isRealMeteor(child) return child:IsA("Model") or child:IsA("Part") or child:IsA("BasePart") end
local function getLiveMeteorCount()
    local count=0
    for _,child in ipairs(workspace.Map.Meteors:GetChildren()) do if isRealMeteor(child) then count+=1 end end
    return count
end
local function startMeteorTimer()
    buildMeteorTimerGui()
    local waitingTaskActive=false
    local function scheduleWaitingStatus()
        if waitingTaskActive then return end waitingTaskActive=true
        task.spawn(function()
            task.wait(30) if not waitingTaskActive then return end
            if not meteorESP.enabled then waitingTaskActive=false return end
            if getLiveMeteorCount()==0 and meteorESP.overrideText=="" then
                setMeteorStatus("Waiting for meteors to spawn...", Color3.fromRGB(180,180,180))
            end
            waitingTaskActive=false
        end)
    end
    if not meteorESP.lastCycleStart then
        local savedUnix=loadLastMeteorTime()
        if savedUnix then
            local secondsSince=(os.time()-savedUnix)%meteorESP.INTERVAL
            meteorESP.lastCycleStart=tick()-secondsSince meteorESP.cycleResultShown=false
        else meteorESP.lastCycleStart=tick() meteorESP.cycleResultShown=false end
    end
    local existingCount=getLiveMeteorCount()
    if existingCount>0 then meteorESP.overrideText="Spawned! ("..existingCount..")" meteorESP.overrideColor=Color3.fromRGB(100,255,100)
    else meteorESP.overrideText="" end
    meteorESP.isPlayerCountLow=getServerPlayerCount()<meteorESP.MIN_PLAYERS
    if meteorESP.isPlayerCountLow then
        setMeteorStatus("Paused – need "..meteorESP.MIN_PLAYERS.."+ players ("..getServerPlayerCount().." now)", Color3.fromRGB(200,180,0))
    end
    updateStatusLabel()
    connections.meteorTimerAdded=workspace.Map.Meteors.ChildAdded:Connect(function(child)
        if not isRealMeteor(child) or not meteorESP.enabled then return end
        waitingTaskActive=false
        local count=getLiveMeteorCount()
        meteorESP.overrideText="Spawned! ("..count..")" meteorESP.overrideColor=Color3.fromRGB(100,255,100)
        setMeteorStatus(meteorESP.overrideText, meteorESP.overrideColor)
        meteorESP.lastCycleStart=tick() meteorESP.cycleResultShown=false saveLastMeteorTime(os.time())
    end)
    connections.meteorTimerRemoved=workspace.Map.Meteors.ChildRemoved:Connect(function(child)
        if not isRealMeteor(child) or not meteorESP.enabled then return end
        local remaining=getLiveMeteorCount()
        if remaining>0 then
            meteorESP.overrideText="Spawned! ("..remaining..")" meteorESP.overrideColor=Color3.fromRGB(100,255,100)
            setMeteorStatus(meteorESP.overrideText, meteorESP.overrideColor)
        else
            meteorESP.overrideText=""
            if not meteorESP.isPlayerCountLow then setMeteorStatus("All meteors picked up", Color3.fromRGB(255,140,0)) scheduleWaitingStatus()
            else updateStatusLabel() end
        end
    end)
    connections.meteorTimer=RunService.Heartbeat:Connect(function()
        if not meteorESP.timerGui then return end
        local playerCount=getServerPlayerCount() local lowNow=playerCount<meteorESP.MIN_PLAYERS
        if lowNow~=meteorESP.isPlayerCountLow then
            meteorESP.isPlayerCountLow=lowNow updateStatusLabel()
        end
        if meteorESP.isPlayerCountLow then
            meteorESP.timerGui.countdown.Text="-- : -- ⏸" meteorESP.timerGui.countdown.TextColor3=Color3.fromRGB(200,180,0)
        else
            if not meteorESP.lastCycleStart then return end
            local elapsed=tick()-meteorESP.lastCycleStart local remaining=meteorESP.INTERVAL-elapsed
            if remaining<=0 then
                if not meteorESP.cycleResultShown then
                    meteorESP.cycleResultShown=true
                    if getLiveMeteorCount()==0 and meteorESP.overrideText=="" then setMeteorStatus("Not Spawned", Color3.fromRGB(255,80,80)) end
                    meteorESP.lastCycleStart=tick() meteorESP.cycleResultShown=false
                end
                meteorESP.timerGui.countdown.Text="00 : 00" meteorESP.timerGui.countdown.TextColor3=Color3.fromRGB(255,60,60)
            else
                meteorESP.cycleResultShown=false meteorESP.timerGui.countdown.Text=formatTime(remaining)
                local pct=remaining/meteorESP.INTERVAL
                if pct>0.5 then meteorESP.timerGui.countdown.TextColor3=Color3.fromRGB(180,120,255)
                elseif pct>0.2 then meteorESP.timerGui.countdown.TextColor3=Color3.fromRGB(255,165,0)
                else meteorESP.timerGui.countdown.TextColor3=Color3.fromRGB(255,60,60) end
            end
        end
        if meteorESP.timerGui.status.Text~=meteorESP.statusText then meteorESP.timerGui.status.Text=meteorESP.statusText end
    end)
end
local function stopMeteorTimer()
    if connections.meteorTimer then connections.meteorTimer:Disconnect() connections.meteorTimer=nil end
    if connections.meteorTimerAdded then connections.meteorTimerAdded:Disconnect() connections.meteorTimerAdded=nil end
    if connections.meteorTimerRemoved then connections.meteorTimerRemoved:Disconnect() connections.meteorTimerRemoved=nil end
    destroyMeteorTimerGui() meteorESP.isPlayerCountLow=false meteorESP.overrideText="" meteorESP.statusText=""
end

    highlight.DepthMode=Enum.HighlightDepthMode.AlwaysOnTop
    highlight.Adornee=meteor highlight.Parent=meteor

    bb.Name="MeteorESP" bb.AlwaysOnTop=true bb.Size=UDim2.new(0,130,0,42)

    label.Text="☄ Meteor" label.Parent=bb
    local conn=RunService.Heartbeat:Connect(function()
        if not meteorESP.enabled or not core or not core.Parent then return end
        local char=LocalPlayer.Character local hrp=char and char:FindFirstChild("HumanoidRootPart")
        if hrp then label.Text="☄ Meteor\n"..math.floor((hrp.Position-core.Position).Magnitude).." studs" end

    meteorESP.removedConns[meteor]=meteor.AncestryChanged:Connect(function()
        if not meteor.Parent then
            conn:Disconnect()
            if meteorESP.removedConns[meteor] then meteorESP.removedConns[meteor]:Disconnect() meteorESP.removedConns[meteor]=nil end
        end
    end)
end

        local core=meteor:FindFirstChild("Core")
        if core then local bb=core:FindFirstChild("MeteorESP") if bb then bb:Destroy() end end
    end
    for _,conn in pairs(meteorESP.removedConns) do pcall(function() conn:Disconnect() end) end
    meteorESP.removedConns={}
end
local function startMeteorESP()
    for _,meteor in ipairs(workspace.Map.Meteors:GetChildren()) do createMeteorESP(meteor) end
    connections.meteorAdded=workspace.Map.Meteors.ChildAdded:Connect(function(meteor)
        if not meteorESP.enabled or not isRealMeteor(meteor) then return end
        task.spawn(function()
            local ok,core=pcall(function() return meteor:WaitForChild("Core",3) end)
            if not ok or not core or not meteorESP.enabled then return end
            createMeteorESP(meteor)
        end)
    end)
end
local function stopMeteorESP()
    if connections.meteorAdded then connections.meteorAdded:Disconnect() connections.meteorAdded=nil end
    removeAllMeteorESP()
end

        if not connections.meteorAdded then startMeteorESP() end
    else stopMeteorESP() stopMeteorTimer() end
end

-- Top Inspector
local function getOwnerName(top)
    local ownerVal=top:FindFirstChild("Owner")
    if ownerVal and ownerVal.Value then return ownerVal.Value.Name end
    local uid=tonumber(top.Name)
    if uid then local p=Players:GetPlayerByUserId(uid) if p then return p.Name end return "ID:"..top.Name end
    return top.Name
end
local function getRPMBarColor(pct)
    if pct>=1.0 then return Color3.fromRGB(180,80,255)
    elseif pct>=0.50 then return Color3.fromRGB(0,200,255)
    elseif pct>=0.25 then return Color3.fromRGB(255,165,0)
    else return Color3.fromRGB(255,50,50) end
end
local function getLiveRPM(top)
    local state=tostring(top:GetAttribute("State") or "")
    if not LIVE_STATES[state] then return nil end
    local health=top:GetAttribute("Health")
    if health and health>0 then return math.floor(health*20) end return 0
end
local function getLiveMaxRPM(top)
    local mh=top:GetAttribute("MaxHealth")
    if mh and mh>0 then return math.max(math.floor(mh*20),1) end
    local vc=tonumber(top:GetAttribute("VictoryCents")) or 60
    return math.max(vc*RPM_MULT,1)
end

local function buildTopBillboard(top)
    if topInspector.buildLock[top] then return end topInspector.buildLock[top]=true
    local crown=top:FindFirstChild("Crown")
    local body=top:FindFirstChild("Body")
    if not crown or not body then
        topInspector.buildLock[top]=nil
        local readyConn readyConn=top.ChildAdded:Connect(function()
            if top:FindFirstChild("Crown") and top:FindFirstChild("Body") then
                readyConn:Disconnect() task.wait(0.5)
                if top.Parent and topInspector.enabled then buildTopBillboard(top) end
            end
        end)
        return
    end
    if topInspector.bbCache[top] then
        pcall(function() topInspector.bbCache[top].rpmConn:Disconnect() end)
        pcall(function() topInspector.bbCache[top].bb:Destroy() end)
        topInspector.bbCache[top]=nil
    end
    local attachPart=crown or body or top:FindFirstChildWhichIsA("BasePart")
    if not attachPart then topInspector.buildLock[top]=nil return end
    local sc=topInspector.scale
    local BASE_W=260 local LINE_H=14 local PAD=4
    local RPM_LABEL_H=20 local RPM_BAR_H=22 local RPM_GAP=6 local STATE_ROW_H=24
    local totalH=PAD+LINE_H+RPM_GAP+RPM_LABEL_H+RPM_BAR_H+STATE_ROW_H+PAD+12
    local bb=Instance.new("BillboardGui")
    bb.Name="TopInspectorBB" bb.AlwaysOnTop=true
    bb.Size=UDim2.new(0,math.floor(BASE_W*sc),0,math.floor(totalH*sc))
    bb.StudsOffsetWorldSpace=Vector3.new(0,5,0) bb.MaxDistance=topInspector._maxDistance or 150
    bb.Adornee=attachPart bb.Parent=attachPart
    local inspHL=Instance.new("Highlight")
    inspHL.Name="TopInspectorHL" inspHL.FillColor=Color3.fromRGB(0,150,255)
    inspHL.OutlineColor=Color3.fromRGB(0,150,255) inspHL.FillTransparency=0.5
    inspHL.OutlineTransparency=0 inspHL.DepthMode=Enum.HighlightDepthMode.AlwaysOnTop
    inspHL.Adornee=top inspHL.Parent=top
    local bg=Instance.new("Frame")
    bg.BackgroundColor3=Color3.fromRGB(10,10,20)
    bg.BackgroundTransparency= 1-(topInspector._opacity or 1.0)
    bg.BorderSizePixel=0 bg.Size=UDim2.new(1,0,1,0) bg.Parent=bb
    Instance.new("UICorner",bg).CornerRadius=UDim.new(0,4)
    local yPos=PAD local baseTextSize=11
    local function makeLine(text, color)
        local lbl=Instance.new("TextLabel")
        lbl.BackgroundTransparency=1 lbl.Size=UDim2.new(1,-6,0,math.floor(LINE_H*sc))
        lbl.Position=UDim2.new(0,3,0,math.floor(yPos*sc)) lbl.Font=Enum.Font.GothamBold
        lbl.TextSize=math.floor(baseTextSize*sc) lbl.TextColor3=color
        lbl.TextStrokeTransparency=0.5 lbl.TextStrokeColor3=Color3.fromRGB(0,0,0)
        lbl.TextXAlignment=Enum.TextXAlignment.Left lbl.Text=text lbl.Parent=bb
        yPos+=LINE_H return lbl
    end
    makeLine(getOwnerName(top), Color3.fromRGB(255,255,255))
    yPos+=RPM_GAP
    local rpmLabelRow=yPos
    local rpmTitle=Instance.new("TextLabel")
    rpmTitle.BackgroundTransparency=1 rpmTitle.Size=UDim2.new(0.5,0,0,math.floor(RPM_LABEL_H*sc))
    rpmTitle.Position=UDim2.new(0,3,0,math.floor(rpmLabelRow*sc)) rpmTitle.Font=Enum.Font.GothamBold
    rpmTitle.TextSize=math.floor(15*sc) rpmTitle.TextColor3=Color3.fromRGB(200,200,200)
    rpmTitle.TextXAlignment=Enum.TextXAlignment.Left rpmTitle.Text="RPM" rpmTitle.Parent=bb
    local rpmPct=Instance.new("TextLabel")
    rpmPct.Name="RPMPct" rpmPct.BackgroundTransparency=1 rpmPct.Size=UDim2.new(0.5,-3,0,math.floor(RPM_LABEL_H*sc))
    rpmPct.Position=UDim2.new(0.5,0,0,math.floor(rpmLabelRow*sc)) rpmPct.Font=Enum.Font.GothamBold
    rpmPct.TextSize=math.floor(20*sc) rpmPct.TextColor3=Color3.fromRGB(200,200,200)
    rpmPct.TextXAlignment=Enum.TextXAlignment.Right rpmPct.Text="..." rpmPct.Parent=bb
    local barY=rpmLabelRow+RPM_LABEL_H
    local barBG=Instance.new("Frame")
    barBG.Name="RPMBarBG" barBG.BackgroundColor3=Color3.fromRGB(30,30,40) barBG.BorderSizePixel=0
    barBG.Size=UDim2.new(1,-6,0,math.floor(RPM_BAR_H*sc)) barBG.Position=UDim2.new(0,3,0,math.floor(barY*sc)) barBG.Parent=bb
    Instance.new("UICorner",barBG).CornerRadius=UDim.new(0,3)
    local barFill=Instance.new("Frame")
    barFill.Name="RPMBarFill" barFill.BackgroundColor3=Color3.fromRGB(0,200,255) barFill.BorderSizePixel=0
    barFill.Size=UDim2.new(0,0,1,0) barFill.Parent=barBG
    Instance.new("UICorner",barFill).CornerRadius=UDim.new(0,3)
    local barText=Instance.new("TextLabel")
    barText.Name="RPMBarText" barText.BackgroundTransparency=1 barText.Size=UDim2.new(1,0,1,0)
    barText.Font=Enum.Font.GothamBold barText.TextSize=math.floor(14*sc) barText.TextColor3=Color3.fromRGB(255,255,255)
    barText.TextStrokeTransparency=0.3 barText.TextStrokeColor3=Color3.fromRGB(0,0,0) barText.Text="..." barText.Parent=barBG
    local stateY=barY+RPM_BAR_H+8
    local stateRow=Instance.new("TextLabel")
    stateRow.Name="StateRow" stateRow.BackgroundTransparency=1
    stateRow.Size=UDim2.new(1,-6,0,math.floor(STATE_ROW_H*sc))
    stateRow.Position=UDim2.new(0,3,0,math.floor(stateY*sc))
    stateRow.Font=Enum.Font.GothamBold stateRow.TextSize=math.floor(13*sc)
    stateRow.TextColor3=Color3.fromRGB(200,200,200) stateRow.TextStrokeTransparency=0.5
    stateRow.TextStrokeColor3=Color3.fromRGB(0,0,0) stateRow.TextXAlignment=Enum.TextXAlignment.Center
    stateRow.Text="State: Idle" stateRow.Parent=bb
    local lastUpdate=0 local UPDATE_INTERVAL=0.1 local lastKnownRPM=0
    local zeroFrames=0 local ZERO_FRAME_THRESHOLD=8
    local rpmConn=RunService.Heartbeat:Connect(function(dt)
        lastUpdate+=dt if lastUpdate<UPDATE_INTERVAL then return end lastUpdate=0
        if not top or not top.Parent or not bb or not bb.Parent then return end
        local maxRPM=getLiveMaxRPM(top)
        local state=tostring(top:GetAttribute("State") or "")
        local isSpinning=LIVE_STATES[state]==true
        local currentRPM
        if isSpinning then
            local liveRPM=getLiveRPM(top)
            if liveRPM and liveRPM>0 then lastKnownRPM=liveRPM zeroFrames=0 currentRPM=liveRPM
            else zeroFrames+=1 currentRPM=zeroFrames>=ZERO_FRAME_THRESHOLD and 0 or lastKnownRPM end
        else zeroFrames=0 currentRPM=lastKnownRPM end
        local rawPct=maxRPM>0 and (currentRPM/maxRPM) or 0 local fillPct=math.clamp(rawPct,0,1)
        local color=getRPMBarColor(rawPct) local pctInt=math.floor(rawPct*100)
        local displayPct=rawPct>=1.0 and ("+"..math.min(pctInt,999).."%") or (pctInt.."%")
        barFill.BackgroundColor3=isSpinning and color or Color3.fromRGB(100,100,100)
        barFill.Size=UDim2.new(fillPct,0,1,0) barText.Text=string.format("%d / %d",currentRPM,maxRPM)
        rpmPct.Text=displayPct rpmPct.TextColor3=isSpinning and color or Color3.fromRGB(150,150,150)
        local stateLabel=bb:FindFirstChild("StateRow")
        if stateLabel then
            local displayState=state~="" and state or "Idle"
            stateLabel.Text="State: "..displayState
            stateLabel.TextColor3=STATE_COLORS[displayState] or Color3.fromRGB(200,200,200)
        end
        local freshPart=top:FindFirstChild("Body") or top:FindFirstChild("Crown") or top:FindFirstChildWhichIsA("BasePart")
        if freshPart and bb.Adornee~=freshPart then bb.Adornee=freshPart end
    end)
    topInspector.bbCache[top]={bb=bb,rpmConn=rpmConn,hl=inspHL} topInspector.buildLock[top]=nil
    local attrConn=top.AttributeChanged:Connect(function(attrName)
        local staticAttrs={Spirit=true,Attack=true,Defense=true,Stamina=true,SignatureMove_1=true,SignatureMove_2=true,SignatureMove_3=true,MaxHealth=true}
        if staticAttrs[attrName] then
            task.delay(0.3,function() if top and top.Parent and topInspector.enabled then buildTopBillboard(top) end end)
        end
    end)
    local function watchPartAttrs(part)
        if not part:IsA("BasePart") and not part:IsA("Model") then return end
        part.AttributeChanged:Connect(function(attrName)
            if attrName=="Name" and top and top.Parent and topInspector.enabled then
                task.delay(0.1,function() if top and top.Parent and topInspector.enabled then buildTopBillboard(top) end end)
            end
        end)
    end
    for _,child in ipairs(top:GetChildren()) do watchPartAttrs(child) end
    top.ChildAdded:Connect(function(child) watchPartAttrs(child) end)
    top.AncestryChanged:Connect(function()
        if not top.Parent then
            pcall(function() rpmConn:Disconnect() end) pcall(function() attrConn:Disconnect() end)
            if topInspector.bbCache[top] then pcall(function() topInspector.bbCache[top].bb:Destroy() end) topInspector.bbCache[top]=nil end
            topInspector.buildLock[top]=nil
        end
    end)
end

local function clearAllTopBillboards()
    for _,entry in pairs(topInspector.bbCache) do
        pcall(function() entry.rpmConn:Disconnect() end) pcall(function() entry.bb:Destroy() end)
        pcall(function() if entry.hl then entry.hl:Destroy() end end)
    end topInspector.bbCache={} topInspector.buildLock={}
    for _,top in ipairs(workspace.Tops:GetChildren()) do
        local hl=top:FindFirstChild("TopInspectorHL") if hl then hl:Destroy() end
    end
end
local function teardownTopInspectorListeners()
    for _,conn in pairs(topInspector.conns) do pcall(function() conn:Disconnect() end) end topInspector.conns={}
end
local function setupTopInspectorListeners()
    teardownTopInspectorListeners()
    topInspector.conns["Added"]=workspace.Tops.ChildAdded:Connect(function(top)
        if not topInspector.enabled then return end
        task.wait(0.2) if not topInspector.enabled then return end
        buildTopBillboard(top)
    end)
    topInspector.conns["Removed"]=workspace.Tops.ChildRemoved:Connect(function(top)
        if topInspector.bbCache[top] then
            pcall(function() topInspector.bbCache[top].rpmConn:Disconnect() end)
            pcall(function() topInspector.bbCache[top].bb:Destroy() end)
            topInspector.bbCache[top]=nil
        end topInspector.buildLock[top]=nil
    end)
end
local function onTopInspector(value)
    topInspector.enabled=value
    if value then
        setupTopInspectorListeners()
        for _,top in ipairs(workspace.Tops:GetChildren()) do task.spawn(buildTopBillboard,top) end
    else teardownTopInspectorListeners() clearAllTopBillboards() end
end

                        if sub:IsA("TextLabel") then
                            sub.TextSize=math.max(math.floor(14*sc),6)
                        end
                    end
                end
            end
        end)
    end
end})

-- Trash ESP
local function createTrashESP(part)
    -- always skip gold trash — gold trash ESP handles those
    if part:FindFirstChild("Golden") then return end
    if trashESP.bbCache[part] then return end
    local prompt=part:FindFirstChild("Prompt") if not prompt then return end
    local oldHL=part:FindFirstChild("TrashESPHighlight") if oldHL then oldHL:Destroy() end
    local highlight=Instance.new("Highlight")
    highlight.Name="TrashESPHighlight"
    highlight.FillColor=Color3.fromRGB(190,190,190)
    highlight.OutlineColor=Color3.fromRGB(240,240,240)
    highlight.FillTransparency=0.35
    highlight.OutlineTransparency=0
    highlight.DepthMode=Enum.HighlightDepthMode.AlwaysOnTop
    highlight.Adornee=part highlight.Parent=part highlight.Enabled=prompt.Enabled
    local bb=Instance.new("BillboardGui")
    bb.Name="TrashESP" bb.AlwaysOnTop=true bb.Size=UDim2.new(0,120,0,36)
    bb.StudsOffsetWorldSpace=Vector3.new(0,3.2,0) bb.Adornee=part bb.Parent=workspace bb.Enabled=prompt.Enabled
    local label=Instance.new("TextLabel")
    label.BackgroundTransparency=1 label.Size=UDim2.new(1,0,1,0) label.Font=Enum.Font.GothamBold
    label.TextSize=12 label.TextColor3=Color3.fromRGB(220,220,220)
    label.TextStrokeTransparency=0 label.TextStrokeColor3=Color3.fromRGB(0,0,0)
    label.Text="🗑️ Trash" label.Parent=bb
    trashESP.bbCache[part]={bb=bb,highlight=highlight}
    local heartbeatConn=RunService.Heartbeat:Connect(function()
        if not trashESP.enabled or not part or not part.Parent then return end
        -- if this became gold trash mid-session, clean up and bail
        if part:FindFirstChild("Golden") then
            pcall(function() bb:Destroy() end) pcall(function() highlight:Destroy() end)
            trashESP.bbCache[part]=nil return
        end
        local enabled=prompt.Enabled
        bb.Enabled=enabled highlight.Enabled=enabled
        if enabled then
            local char=LocalPlayer.Character local hrp=char and char:FindFirstChild("HumanoidRootPart")
            if hrp then label.Text="🗑️ Trash\n"..math.floor((hrp.Position-part.Position).Magnitude).." studs" end
        end
    end)
    local promptConn=prompt:GetPropertyChangedSignal("Enabled"):Connect(function()
        if part:FindFirstChild("Golden") then return end
        bb.Enabled=prompt.Enabled highlight.Enabled=prompt.Enabled
    end)
    local ancestryConn=part.AncestryChanged:Connect(function()
        if not part.Parent then
            pcall(function() bb:Destroy() end) pcall(function() highlight:Destroy() end)
            trashESP.bbCache[part]=nil
        end
    end)
    trashESP.promptConns[part]={heartbeat=heartbeatConn,prompt=promptConn,ancestry=ancestryConn}
end
local function clearTrashESP()
    for part,conns in pairs(trashESP.promptConns) do
        pcall(function() conns.heartbeat:Disconnect() end)
        pcall(function() conns.prompt:Disconnect() end)
        pcall(function() if conns.ancestry then conns.ancestry:Disconnect() end end)
    end trashESP.promptConns={}
    for _,entry in pairs(trashESP.bbCache) do
        pcall(function() entry.bb:Destroy() end) pcall(function() entry.highlight:Destroy() end)
    end trashESP.bbCache={}
end
local trashESPAddedConn=nil

    if value then
        for _,part in ipairs(workspace.Map.Environment.Trash:GetChildren()) do
            if not part:FindFirstChild("Golden") then createTrashESP(part) end

-- Gold Trash ESP
local function isGoldTrash(part) return part:FindFirstChild("Golden")~=nil end
local function removeTrashESPForGold(part)
    local oldTrashHL=part:FindFirstChild("TrashESPHighlight") if oldTrashHL then oldTrashHL:Destroy() end
    if trashESP.bbCache[part] then
        pcall(function() trashESP.bbCache[part].bb:Destroy() end)
        pcall(function() trashESP.bbCache[part].highlight:Destroy() end)
        trashESP.bbCache[part]=nil
    end
    if trashESP.promptConns[part] then
        pcall(function() trashESP.promptConns[part].heartbeat:Disconnect() end)
        pcall(function() trashESP.promptConns[part].prompt:Disconnect() end)
        pcall(function() if trashESP.promptConns[part].ancestry then trashESP.promptConns[part].ancestry:Disconnect() end end)
        trashESP.promptConns[part]=nil
    end
end
local function createGoldTrashESP(part)
    if goldTrashESP.bbCache[part] then return end
    local prompt=part:FindFirstChild("Prompt") if not prompt then return end
    -- always evict any normal trash ESP first
    removeTrashESPForGold(part)
    local oldHL=part:FindFirstChild("GoldTrashESPHighlight") if oldHL then oldHL:Destroy() end
    local highlight=Instance.new("Highlight")
    highlight.Name="GoldTrashESPHighlight"
    highlight.FillColor=Color3.fromRGB(255,210,0)
    highlight.OutlineColor=Color3.fromRGB(255,255,180)
    highlight.FillTransparency=0.3
    highlight.OutlineTransparency=0
    highlight.DepthMode=Enum.HighlightDepthMode.AlwaysOnTop
    highlight.Adornee=part highlight.Parent=part highlight.Enabled=prompt.Enabled
    local bb=Instance.new("BillboardGui")
    bb.Name="GoldTrashESP" bb.AlwaysOnTop=true bb.Size=UDim2.new(0,140,0,40)
    bb.StudsOffsetWorldSpace=Vector3.new(0,3.2,0) bb.Adornee=part bb.Parent=workspace bb.Enabled=prompt.Enabled
    local label=Instance.new("TextLabel")
    label.BackgroundTransparency=1 label.Size=UDim2.new(1,0,1,0) label.Font=Enum.Font.GothamBold
    label.TextSize=13 label.TextColor3=Color3.fromRGB(255,215,0)
    label.TextStrokeTransparency=0 label.TextStrokeColor3=Color3.fromRGB(0,0,0)
    label.Text="💰 Gold Trash" label.Parent=bb
    goldTrashESP.bbCache[part]={bb=bb,highlight=highlight}
    local heartbeatConn=RunService.Heartbeat:Connect(function()
        if not goldTrashESP.enabled or not part or not part.Parent then return end
        local enabled=prompt.Enabled
        bb.Enabled=enabled highlight.Enabled=enabled
        if enabled then
            local char=LocalPlayer.Character local hrp=char and char:FindFirstChild("HumanoidRootPart")
            if hrp then label.Text="💰 Gold Trash\n"..math.floor((hrp.Position-part.Position).Magnitude).." studs" end
        end
    end)
    local promptConn=prompt:GetPropertyChangedSignal("Enabled"):Connect(function()
        bb.Enabled=prompt.Enabled highlight.Enabled=prompt.Enabled
    end)
    local ancestryConn=part.AncestryChanged:Connect(function()
        if not part.Parent then
            pcall(function() bb:Destroy() end) pcall(function() highlight:Destroy() end)
            goldTrashESP.bbCache[part]=nil
        end
    end)
    goldTrashESP.promptConns[part]={heartbeat=heartbeatConn,prompt=promptConn,ancestry=ancestryConn}
end
local function clearGoldTrashESP()
    for part,conns in pairs(goldTrashESP.promptConns) do
        pcall(function() conns.heartbeat:Disconnect() end)
        pcall(function() conns.prompt:Disconnect() end)
        pcall(function() if conns.ancestry then conns.ancestry:Disconnect() end end)
    end goldTrashESP.promptConns={}
    for _,entry in pairs(goldTrashESP.bbCache) do
        pcall(function() entry.bb:Destroy() end) pcall(function() entry.highlight:Destroy() end)
    end goldTrashESP.bbCache={}
end
local function onGoldTrashESP(value)
    toggleStates.GoldTrashESP=value goldTrashESP.enabled=value
    if value then
        for _,part in ipairs(workspace.Map.Environment.Trash:GetChildren()) do
            if isGoldTrash(part) then
                task.wait(0.02)
                createGoldTrashESP(part)
            end
        end
        if not goldTrashESP._addedConn then
            goldTrashESP._addedConn=workspace.Map.Environment.Trash.ChildAdded:Connect(function(part)
                if not goldTrashESP.enabled then return end
                task.wait(0.05)
                if isGoldTrash(part) then createGoldTrashESP(part) end
            end)
        end
        -- also watch for Golden child being added to existing trash parts
        if not goldTrashESP._goldenConn then
            goldTrashESP._goldenConn=workspace.Map.Environment.Trash.ChildAdded:Connect(function(part)
                -- handled above
            end)
        end
    else
        clearGoldTrashESP()
        if goldTrashESP._addedConn then goldTrashESP._addedConn:Disconnect() goldTrashESP._addedConn=nil end
        if goldTrashESP._goldenConn then goldTrashESP._goldenConn:Disconnect() goldTrashESP._goldenConn=nil end
    end
end

local function getPlayerList()
    local names={} for _,p in ipairs(Players:GetPlayers()) do table.insert(names,p.Name) end table.sort(names) return names

local function getTopForPlayer(playerName)
    for _,top in ipairs(workspace.Tops:GetChildren()) do
        local uid=tonumber(top.Name)
        if uid then local p=Players:GetPlayerByUserId(uid) if p and p.Name==playerName then return top end end
        local ownerVal=top:FindFirstChild("Owner")
        if ownerVal and ownerVal.Value and ownerVal.Value.Name==playerName then return top end
    end
    return nil
end

-- Auto Opponent Viewer
local function startAutoOpponentViewer()
    if viewer.autoConn then viewer.autoConn:Disconnect() viewer.autoConn=nil end
    if viewer._extraConns then
        for _,c in ipairs(viewer._extraConns) do pcall(function() c:Disconnect() end) end
        viewer._extraConns=nil
    end
    local lastOpponentName=nil local lastOpponentTop=nil local targetWatchConn=nil
    local myId=tostring(LocalPlayer.UserId)

    local function isNPCTop(oppTop)
        if not oppTop then return false end
        local uid=tonumber(oppTop.Name)
        if uid then return Players:GetPlayerByUserId(uid)==nil end
        local ov=oppTop:FindFirstChild("Owner")
        if ov and ov.Value then return false end
        return true
    end
    local function getOpponentNameFromTop(oppTop)
        if not oppTop then return nil end
        local uid=tonumber(oppTop.Name)
        if uid then local p=Players:GetPlayerByUserId(uid) if p then return p.Name end return "NPC:"..oppTop.Name end
        local ov=oppTop:FindFirstChild("Owner") if ov and ov.Value then return ov.Value.Name end
        return oppTop.Name
    end
    local function openForOpponent(oppTop, oppName)
        local name=oppName or getOpponentNameFromTop(oppTop)
        if not name or name=="" then return end
        if name==lastOpponentName and oppTop==lastOpponentTop then return end
        if lastOpponentTop and lastOpponentTop.Parent then
            local currentHealth=lastOpponentTop:GetAttribute("Health")
            if currentHealth and currentHealth>0 then
                if not isNPCTop(lastOpponentTop) and (oppTop and isNPCTop(oppTop) or true) then return end
            end
        end
        if oppTop and isNPCTop(oppTop) then
            local myTop=workspace.Tops:FindFirstChild(myId)
            if myTop then
                local ok0, myPivot=pcall(function() return myTop:GetPivot().Position end)
                if ok0 then
                    for _, top in ipairs(workspace.Tops:GetChildren()) do
                        if top.Name==myId or isNPCTop(top) then continue end
                        local h=top:GetAttribute("Health") if not h or h<=0 then continue end
                        local ok2, pivot=pcall(function() return top:GetPivot().Position end)
                        if ok2 and (myPivot-pivot).Magnitude<45 then
                            local realName=getOpponentNameFromTop(top)
                            lastOpponentName=realName lastOpponentTop=top
                            buildViewerGui(realName,top) return
                        end
                    end
                end
            end
        end
        lastOpponentName=name lastOpponentTop=oppTop
        if oppTop and oppTop.Parent then buildViewerGui(name,oppTop)
        else buildViewerGui("Waiting for beyblade...",nil) end
    end
    local function clearOpponent()
        lastOpponentName=nil lastOpponentTop=nil
        if viewer.autoEnabled then buildViewerGui("Waiting for beyblade...",nil) end
    end
    local function hookMyTop(myTop)
        if targetWatchConn then targetWatchConn:Disconnect() targetWatchConn=nil end
        if not myTop then clearOpponent() return end
        local function handleTargetValue(targetChild)
            if not targetChild then return end
            local oppTop=targetChild.Value
            if oppTop and oppTop.Parent then openForOpponent(oppTop,nil) end
            targetChild:GetPropertyChangedSignal("Value"):Connect(function()
                local opp=targetChild.Value if opp and opp.Parent then openForOpponent(opp,nil) end
            end)
        end
        local existingTarget=myTop:FindFirstChild("Target")
        if existingTarget then handleTargetValue(existingTarget) end
        targetWatchConn=myTop.ChildAdded:Connect(function(child)
            if child.Name~="Target" then return end handleTargetValue(child)
        end)
        myTop.ChildRemoved:Connect(function(child)
            if child.Name~="Target" then return end
            task.delay(0.5,function()
                if not viewer.autoEnabled then return end
                local t2=workspace.Tops:FindFirstChild(myId)
                if not t2 or not t2:FindFirstChild("Target") then clearOpponent() end
            end)
        end)
    end
    local topsAddedConn=workspace.Tops.ChildAdded:Connect(function(top)
        if not viewer.autoEnabled then return end
        if top.Name==myId then task.wait(0.3) hookMyTop(top) end
    end)
    local topsRemovedConn=workspace.Tops.ChildRemoved:Connect(function(top)
        if not viewer.autoEnabled then return end
        if top.Name==myId then hookMyTop(nil) end
    end)
    local existingMyTop=workspace.Tops:FindFirstChild(myId)
    if existingMyTop then hookMyTop(existingMyTop) end

    local cycleList={} local cycleIndex=1 local CYCLE_INTERVAL=3 local cycleLastSwitchTime=0
    local function getAllOpponentsInArena()
        local myTop=workspace.Tops:FindFirstChild(myId) if not myTop then return {} end
        local myHealth=myTop:GetAttribute("Health") if not myHealth or myHealth<=0 then return {} end
        local ok0, myPivot=pcall(function() return myTop:GetPivot().Position end) if not ok0 then return {} end
        local realPlayers, npcs={},{}
        for _, top in ipairs(workspace.Tops:GetChildren()) do
            if top.Name==myId then continue end
            local health=top:GetAttribute("Health") if not health or health<=0 then continue end
            local ok, pivot=pcall(function() return top:GetPivot().Position end) if not ok then continue end
            if (myPivot-pivot).Magnitude>45 then continue end
            local name=getOpponentNameFromTop(top)
            if isNPCTop(top) then table.insert(npcs,{top=top,name=name})
            else table.insert(realPlayers,{top=top,name=name}) end
        end
        if #realPlayers>0 then return realPlayers end return npcs
    end
    local function buildCycleList()
        local found=getAllOpponentsInArena()
        local changed=#found~=#cycleList
        if not changed then
            for i,entry in ipairs(found) do if not cycleList[i] or cycleList[i].top~=entry.top then changed=true break end end
        end
        if changed then
            cycleList=found
            local currentStillPresent=false
            for _,entry in ipairs(found) do if entry.top==lastOpponentTop then currentStillPresent=true break end end
            if not currentStillPresent then cycleIndex=1 cycleLastSwitchTime=-999 end
        end
        return changed
    end

    local myName=LocalPlayer.Name local myDisplay=LocalPlayer.DisplayName
    local function extractVsLabel(text)
        if not text or text=="" then return nil end
        local tl=text:lower() if not tl:find(" vs ") then return nil end
        local left,right=tl:match("^(.-)%s+vs%s+(.+)$") if not left or not right then return nil end
        left=left:gsub("^%s+",""):gsub("%s+$","") right=right:gsub("^%s+",""):gsub("%s+$","")
        local myL,myDL=myName:lower(),myDisplay:lower()
        local oppL
        if left==myL or left==myDL or myL:find(left,1,true) or left:find(myL,1,true) then oppL=right
        elseif right==myL or right==myDL or myL:find(right,1,true) or right:find(myL,1,true) then oppL=left
        else return nil end
        for _,p in ipairs(Players:GetPlayers()) do
            if p.Name:lower()==oppL or p.DisplayName:lower()==oppL then return p.Name end
        end
        return oppL
    end
    local function resolveTopByName(oppL)
        local lower=oppL:lower()
        for _,p in ipairs(Players:GetPlayers()) do
            if p.Name:lower()==lower or p.DisplayName:lower()==lower then return workspace.Tops:FindFirstChild(tostring(p.UserId)) end
        end
        return nil
    end

    local proximityLastRun=0 local PROXIMITY_INTERVAL=0.5
    local proximityConn=RunService.Heartbeat:Connect(function()
        if not viewer.autoEnabled then return end
        local now=tick()
        if lastOpponentTop then
            if not lastOpponentTop.Parent then lastOpponentTop=nil lastOpponentName=nil cycleLastSwitchTime=-999 end
        end
        if now-proximityLastRun<PROXIMITY_INTERVAL then return end
        proximityLastRun=now
        local myTop=workspace.Tops:FindFirstChild(myId)
        local targetChild=myTop and myTop:FindFirstChild("Target")
        local targetTop=targetChild and targetChild.Value
        if targetTop and targetTop.Parent then
            local h=targetTop:GetAttribute("Health") if h and h>0 then return end
        end
        if not myTop then return end
        local myHealth=myTop:GetAttribute("Health") if not myHealth or myHealth<=0 then return end
        if myTop:GetAttribute("IsStadiumFight")~=true then if lastOpponentName then clearOpponent() end return end
        buildCycleList()
        if #cycleList==0 then
            if lastOpponentName then return end
            for _,gui in ipairs(playerGui:GetChildren()) do
                if not gui:IsA("ScreenGui") or not gui.Enabled then continue end
                local function collectLabels(inst, out)
                    for _,child in ipairs(inst:GetChildren()) do
                        if child:IsA("TextLabel") and child.Visible then table.insert(out,child.Text) end
                        collectLabels(child,out)
                    end
                end
                local labels={} collectLabels(gui,labels)
                for _,text in ipairs(labels) do
                    local opp=extractVsLabel(text)
                    if opp and opp~=lastOpponentName then
                        local oppTop=resolveTopByName(opp)
                        openForOpponent(oppTop, oppTop and nil or opp) return
                    end
                end
            end
            return
        end
        if #cycleList==1 then
            local entry=cycleList[1]
            if lastOpponentTop~=entry.top then
                lastOpponentName=entry.name lastOpponentTop=entry.top
                cycleLastSwitchTime=tick() buildViewerGui(entry.name,entry.top)
            end
            return
        end
        if now-cycleLastSwitchTime<CYCLE_INTERVAL then return end
        cycleLastSwitchTime=now
        if cycleIndex>#cycleList then cycleIndex=1 end
        local entry=cycleList[cycleIndex] cycleIndex+=1
        if cycleIndex>#cycleList then cycleIndex=1 end
        if lastOpponentTop~=entry.top then
            lastOpponentName=entry.name lastOpponentTop=entry.top buildViewerGui(entry.name,entry.top)
        end
    end)
    viewer.autoConn=proximityConn
    viewer._extraConns={topsAddedConn,topsRemovedConn}
end

    if viewer._extraConns then
        for _,c in ipairs(viewer._extraConns) do pcall(function() c:Disconnect() end) end
        viewer._extraConns=nil
    end
end

-- ── Instant Interact ──────────────────────────────────────────────────────────

do
local instantInteract = {
    enabled = false,
    conn    = nil,
    RANGE   = 10,        -- studs — prompt fires when trash is within this distance
    MOVING_THRESHOLD = 0.5,  -- studs/s below this = considered standing still
}

local lastPos     = nil
local lastPosTime = 0

local function isMoving()
    local char = LocalPlayer.Character
    local hrp  = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    local now = tick()
    local pos = hrp.Position
    if lastPos and (now - lastPosTime) > 0 then
        local speed = (pos - lastPos).Magnitude / (now - lastPosTime)
        lastPos     = pos
        lastPosTime = now
        return speed >= instantInteract.MOVING_THRESHOLD
    end
    lastPos     = pos
    lastPosTime = now
    return false
end

local function isTrashPart(part)
    -- must be in the Trash folder
    local parent = part.Parent
    if not parent then return false end
    return parent == workspace.Map.Environment.Trash
end

local function isGoldTrashPart(part)
    return isTrashPart(part) and part:FindFirstChild("Golden") ~= nil
end

local function startInstantInteract()
    if instantInteract.conn then
        instantInteract.conn:Disconnect()
        instantInteract.conn = nil
    end
    lastPos     = nil
    lastPosTime = 0

    instantInteract.conn = RunService.Heartbeat:Connect(function()
        if not instantInteract.enabled then return end
        if not isMoving() then return end

        local char = LocalPlayer.Character
        local hrp  = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end

        local hrpPos = hrp.Position

        for _, part in ipairs(workspace.Map.Environment.Trash:GetChildren()) do
            -- only trash and gold trash
            if not isTrashPart(part) then continue end

            local prompt = part:FindFirstChild("Prompt")
            if not prompt or not prompt.Enabled then continue end

            local dist = (hrpPos - part.Position).Magnitude
            if dist <= instantInteract.RANGE then
                fireproximityprompt(prompt)
            end
        end
    end)
end

local function stopInstantInteract()
    if instantInteract.conn then
        instantInteract.conn:Disconnect()
        instantInteract.conn = nil
    end
    lastPos     = nil
    lastPosTime = 0
end

MiscLeft:AddToggle('InstantInteract', {
    Text     = 'Instant Interact (Trash)',
    Default  = false,
    Callback = function(value)
        instantInteract.enabled = value
        if value then startInstantInteract() else stopInstantInteract() end
    end,
})

end -- instant interact do block

local WebhookRight = Tabs.Misc:AddLeftGroupbox('Webhooks')

local webhookState = { url = "", enabled = false, disconnectPinged = false }
local WEBHOOK_FILE = "TAT_Webhook.json"
local function loadWebhookState()
    local ok, data = pcall(function() return readfile(WEBHOOK_FILE) end)
    if not ok or not data or data == "" then return end
    local ok2, parsed = pcall(function() return HttpService:JSONDecode(data) end)
    if ok2 and type(parsed) == "table" then
        webhookState.url = parsed.url or ""
        webhookState.enabled = parsed.enabled or false
    end
end
local function saveWebhookState()
    pcall(function() writefile(WEBHOOK_FILE, HttpService:JSONEncode({url=webhookState.url, enabled=webhookState.enabled})) end)
end
loadWebhookState()

local function tryRequest(url, payload)
    local opts = { Url=url, Method="POST", Headers={["Content-Type"]="application/json"}, Body=payload }
    if syn and syn.request then pcall(syn.request, opts) return end
    if http and http.request then pcall(http.request, opts) return end
    if request then pcall(request, opts) return end
    if http_request then pcall(http_request, opts) return end
end

local function sendDisconnectWebhook()
    if not webhookState.url or webhookState.url == "" then return end
    local uname = LocalPlayer.Name local dname = LocalPlayer.DisplayName local uid = LocalPlayer.UserId
    local jobId = tostring(game.JobId):sub(1,8)
    local avatarUrl = string.format("https://thumbnails.roblox.com/v1/users/avatar?userIds=%d&size=420x420&format=Png&isCircular=false", uid)
    local payload = HttpService:JSONEncode({
        content = "@everyone ⚠️ DISCONNECT DETECTED!",
        embeds  = {{
            title       = "🔴 DISCONNECTED",
            description = string.format("**%s** (`@%s`) disconnected!\n\n**Server:** `%s...`", dname, uname, jobId),
            color       = 15158332,
            thumbnail   = { url = avatarUrl },
            footer      = { text = string.format("Requiem | TAT  •  UserID: %d", uid) },
            timestamp   = os.date("!%Y-%m-%dT%H:%M:%SZ"),
        }}
    })
    task.spawn(tryRequest, webhookState.url, payload)
end

LocalPlayer.OnTeleport:Connect(function(state)
    if state == Enum.TeleportState.RequestedFromServer then
        if webhookState.enabled and not webhookState.disconnectPinged then
            webhookState.disconnectPinged = true
            sendDisconnectWebhook()
        end
    end
end)

task.spawn(function()

        if not ok and webhookState.enabled and not webhookState.disconnectPinged then
            webhookState.disconnectPinged = true
            sendDisconnectWebhook()
            break
        end
    end
end)
