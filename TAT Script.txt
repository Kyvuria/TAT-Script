-- ── Underground ───────────────────────────────────────────────────────────────
local PLAYER_RADIUS = 100
local UNDERGROUND_Y = -50    -- studs below ground
local TWEEN_SPEED   = 80     -- studs per second while underground
local isUnderground = false
local savedGroundY  = nil

local function isPlayerNearby()
    local char = LocalPlayer.Character
    local hrp  = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local oh = p.Character:FindFirstChild("HumanoidRootPart")
            if oh and (hrp.Position - oh.Position).Magnitude <= PLAYER_RADIUS then
                return true
            end
        end
    end
    return false
end

local function tweenHRP(hrp, targetPos, duration)
    local info  = TweenInfo.new(math.max(duration, 0.05), Enum.EasingStyle.Linear)
    local tween = TweenService:Create(hrp, info, { CFrame = CFrame.new(targetPos) })
    tween:Play()
    tween.Completed:Wait()
end

-- Sink underground, store ground Y
local function goUnderground(hrp)
    if isUnderground then return end
    savedGroundY = hrp.Position.Y
    tweenHRP(hrp, Vector3.new(hrp.Position.X, savedGroundY + UNDERGROUND_Y, hrp.Position.Z), 0.35)
    isUnderground = true
end

-- Tween XZ continuously underground toward target (speed-based duration)
local function tweenUnderTo(hrp, targetXZ)
    local underY  = savedGroundY + UNDERGROUND_Y
    local current = Vector3.new(hrp.Position.X, underY, hrp.Position.Z)
    local dest    = Vector3.new(targetXZ.X, underY, targetXZ.Z)
    local dist    = (dest - current).Magnitude
    tweenHRP(hrp, dest, math.max(dist / TWEEN_SPEED, 0.05))
end

-- Surface back to saved ground Y
local function surfaceUp(hrp)
    if not isUnderground then return end
    tweenHRP(hrp, Vector3.new(hrp.Position.X, savedGroundY, hrp.Position.Z), 0.35)
    isUnderground = false
    savedGroundY  = nil
end

local function randomWait(mn, mx)
    task.wait(mn + math.random() * (mx - mn))
end

local function maybeJump(humanoid)
    if math.random(1,3) == 1 then humanoid.Jump = true task.wait(0.1) humanoid.Jump = false end
end

local function checkStuck(hrp, humanoid, target)
    local cur = hrp.Position
    if lastPos then
        if (cur - lastPos).Magnitude < 0.5 then
            stuckCount += 1
            if stuckCount >= 2 then
                humanoid.Jump = true task.wait(0.2) humanoid.Jump = false
                humanoid:MoveTo(cur + Vector3.new(math.random(-4,4), 0, math.random(-4,4)))
                task.wait(0.3) humanoid:MoveTo(target) stuckCount = 0
            end
        else stuckCount = 0 end
    end
    lastPos = cur
end

local function walkTo(hrp, humanoid, target, timeout, blacklistKey, farmCheck)
    stuckCount = 0 lastPos = nil
    humanoid:MoveTo(target)
    local t = tick() + (timeout or 20)
    local startPos = hrp.Position
    repeat
        task.wait(0.3)
        if not farmCheck() then return false end
        checkStuck(hrp, humanoid, target) maybeJump(humanoid)
    until (hrp.Position - target).Magnitude < 8 or tick() > t
    local arrived = (hrp.Position - target).Magnitude < 8
    if not arrived and blacklistKey and (hrp.Position - startPos).Magnitude < 15 then
        trashBlacklist[blacklistKey] = true
    end
    return arrived
end

-- ── Blatant walk with continuous underground tweening ────────────────────────
local function blatantWalkTo(hrp, humanoid, target, timeout, blacklistKey, farmCheck)
    if isPlayerNearby() then
        -- Go underground if not already
        if not isUnderground then
            goUnderground(hrp)
        end

        -- Tween underground directly to trash XZ
        tweenUnderTo(hrp, target)

        -- Stay underground, keep tweening toward target until no players nearby
        local waitEnd = tick() + (timeout or 20)
        while isPlayerNearby() and tick() < waitEnd and farmCheck() do
            -- Keep nudging toward target while waiting underground
            tweenUnderTo(hrp, target)
            task.wait(0.2)
        end

        if not farmCheck() then return false end

        -- Players gone — surface at current XZ
        surfaceUp(hrp)
        return true
    else
        -- No players nearby — surface if underground then walk normally
        if isUnderground then surfaceUp(hrp) end
        return walkTo(hrp, humanoid, target, timeout, blacklistKey, farmCheck)
    end
end

local function getTrashCount(waitForValid)
    local function read()
        local ok, label = pcall(function() return playerGui.HUD["Main HUD"].Corner.Trash.Label end)
        if not ok or not label then return 0, 0 end
        local cur, max = label.Text:match("(%d+)%s*/%s*(%d+)")
        return tonumber(cur) or 0, tonumber(max) or 0
    end
    if waitForValid then
        local deadline = tick() + 3
        local c, m = read()
        while m == 0 and tick() < deadline do task.wait(0.2) c, m = read() end
        return c, m
    end
    return read()
end

local function isDumpsterOnCooldown(part)
    local t = part:FindFirstChild("TimerGUI")
    return t and t.Enabled == true
end

local function getAvailableDumpster()
    local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local best, bestDist = nil, math.huge
    for _, part in ipairs(workspace.Map.Environment.Dumpsters:GetChildren()) do
        if part:IsA("Part") and not isDumpsterOnCooldown(part) then
            local prompt = part:FindFirstChild("Prompt")
            if prompt and prompt.Enabled then
                local d = (hrp.Position - part.Position).Magnitude
                if d < bestDist then best = { part = part, pos = part.Position } bestDist = d end
            end
        end
    end
    return best
end

local function getNextAvailableDumpsterTime()
    local soonest = nil
    for _, part in ipairs(workspace.Map.Environment.Dumpsters:GetChildren()) do
        if part:IsA("Part") then
            local tg = part:FindFirstChild("TimerGUI")
            if tg and tg.Enabled then
                local lbl = tg:FindFirstChild("Label")
                if lbl and lbl.Text ~= "" then
                    local m, s = lbl.Text:match("(%d+):(%d+)")
                    if m and s then
                        local secs = tonumber(m)*60 + tonumber(s)
                        if not soonest or secs < soonest then soonest = secs end
                    end
                end
            else return 0 end
        end
    end
    return soonest or 0
end

-- ── Farm loop ─────────────────────────────────────────────────────────────────
local function runFarmLoop(farmCheck, blatant)
    local function getChar()
        local c = LocalPlayer.Character
        if not c then return nil, nil end
        return c:FindFirstChild("HumanoidRootPart"), c:FindFirstChild("Humanoid")
    end

    local lastTrashCount = -1
    local lastIncreaseTime = tick()
    local hasPickedUpAtLeastOne = false

    startAutoSprint(farmCheck)
    if blatant then startNoclip() end

    while farmCheck() do
        local hrp, humanoid = getChar()
        if not hrp or not humanoid then task.wait(1) continue end

        local cur, max = getTrashCount(true)
        if cur ~= lastTrashCount then lastTrashCount = cur lastIncreaseTime = tick() end

        if tick() - lastIncreaseTime > 10 then
            Rayfield:Notify({ Title = "Trash Farm", Content = "Stuck! Resetting...", Duration = 3 })
            if isUnderground then surfaceUp(hrp) end
            LocalPlayer.Character:BreakJoints()
            task.wait(4)
            isUnderground = false savedGroundY = nil
            startAutoSprint(farmCheck)
            if blatant then startNoclip() end
            lastIncreaseTime = tick() lastTrashCount = -1 hasPickedUpAtLeastOne = false
            continue
        end

        if hasPickedUpAtLeastOne and max > 0 and cur >= max then
            -- Always surface before going to dumpster
            if isUnderground then
                local h = getChar()
                if h then surfaceUp(h) end
            end
            local dumpster = getAvailableDumpster()
            if dumpster then
                local arrived = walkTo(hrp, humanoid, dumpster.pos, 30, nil, farmCheck)
                if arrived then
                    randomWait(0.3, 0.7)
                    local prompt = dumpster.part:FindFirstChild("Prompt")
                    if prompt then
                        fireproximityprompt(prompt, 2)
                        task.wait(2)
                        trashBlacklist = {} lastTrashCount = -1
                        lastIncreaseTime = tick() hasPickedUpAtLeastOne = false
                    end
                end
            else
                local waitSecs = math.max(getNextAvailableDumpsterTime() or 60, 5)
                Rayfield:Notify({ Title = "Trash Farm", Content = "Dumpsters on cooldown! Waiting " .. math.ceil(waitSecs) .. "s", Duration = 5 })
                local waitEnd = tick() + waitSecs + 2
                repeat
                    task.wait(2)
                    if getAvailableDumpster() then break end
                until tick() > waitEnd or not farmCheck()
                lastIncreaseTime = tick()
            end
        else
            local hrp2, humanoid2 = getChar()
            if not hrp2 or not humanoid2 then task.wait(1) continue end

            local nearest, nearestDist, nearestPrompt, nearestKey = nil, math.huge, nil, nil
            for _, paper in ipairs(workspace.Map.Environment.Trash:GetChildren()) do
                if trashBlacklist[paper.Name] then continue end
                local prompt = paper:FindFirstChild("Prompt")
                if prompt and prompt.Enabled then
                    local d = (hrp2.Position - paper.Position).Magnitude
                    if d < nearestDist then
                        nearest = paper nearestDist = d
                        nearestPrompt = prompt nearestKey = paper.Name
                    end
                end
            end

            if nearest then
                local arrived
                if blatant then
                    arrived = blatantWalkTo(hrp2, humanoid2, nearest.Position, 20, nearestKey, farmCheck)
                else
                    arrived = walkTo(hrp2, humanoid2, nearest.Position, 20, nearestKey, farmCheck)
                end
                if arrived and nearestPrompt and nearestPrompt.Enabled then
                    fireproximityprompt(nearestPrompt)
                    hasPickedUpAtLeastOne = true
                    lastIncreaseTime = tick()
                    task.wait(0.3)
                elseif not arrived then
                    lastIncreaseTime = tick()
                end
            else
                task.wait(3)
            end
        end

        task.wait(0.1)
    end

    -- Cleanup
    if isUnderground then
        local h = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if h then surfaceUp(h) end
    end
    isUnderground = false savedGroundY = nil
    stopAutoSprint()
    if blatant then stopNoclip() end
end

local isBurgerBuying = false

local function doBurgerRun(manual)
    if isBurgerBuying then return end
    isBurgerBuying = true isManualRun = manual or false manualCancelled = false
    local character = LocalPlayer.Character
    local hrp = character and character:FindFirstChild("HumanoidRootPart")
    local humanoid = character and character:FindFirstChild("Humanoid")
    if not hrp or not humanoid then isBurgerBuying = false return end
    local originalPos = hrp.Position
    local standPos = Vector3.new(573.88, 118, -157.43)
    local keyConnection
    if isManualRun then
        keyConnection = UserInputService.InputBegan:Connect(function(input, gp)
            if gp then return end
            if input.UserInputType == Enum.UserInputType.Keyboard then
                manualCancelled = true humanoid:MoveTo(hrp.Position)
            end
        end)
    end
    stuckCount = 0 lastPos = nil
    humanoid:MoveTo(standPos)
    local timeout = tick() + 12
    repeat
        task.wait(0.3)
        if manualCancelled then break end
        checkStuck(hrp, humanoid, standPos) maybeJump(humanoid)
    until (hrp.Position - standPos).Magnitude < 3 or tick() > timeout or manualCancelled
    if not manualCancelled then
        randomWait(0.4, 0.9)
        local ok, vendor = pcall(function() return workspace.Map["Food Vendors"]:GetChildren()[11] end)
        if ok and vendor then
            local prompt = vendor.Counter["Prompt Host"].Prompt
            if prompt then fireproximityprompt(prompt) end
        end
        if not isManualRun then
            randomWait(2.8, 3.4)
            local returnPos = originalPos + Vector3.new(math.random(-2,2), 0, math.random(-2,2))
            stuckCount = 0 lastPos = nil
            humanoid:MoveTo(returnPos)
            local returnTimeout = tick() + 12
            repeat
                task.wait(0.3) checkStuck(hrp, humanoid, returnPos) maybeJump(humanoid)
            until (hrp.Position - returnPos).Magnitude < 3 or tick() > returnTimeout
        end
    end
    if keyConnection then keyConnection:Disconnect() end
    isBurgerBuying = false isManualRun = false manualCancelled = false
end

local meteorESPEnabled = false
local meteorAddedConnection = nil
local meteorRemovedConnections = {}

local function createMeteorESP(meteor)
    local core = meteor:FindFirstChild("Core") if not core then return end
    if core:FindFirstChild("MeteorESP") then return end
    local bb = Instance.new("BillboardGui")
    bb.Name = "MeteorESP" bb.AlwaysOnTop = true
    bb.Size = UDim2.new(0,120,0,40) bb.StudsOffset = Vector3.new(0,4,0)
    bb.Adornee = core bb.Parent = core
    local frame = Instance.new("Frame")
    frame.BackgroundTransparency = 1 frame.Size = UDim2.new(1,0,1,0) frame.Parent = bb
    local label = Instance.new("TextLabel")
    label.BackgroundTransparency = 1 label.Size = UDim2.new(1,0,1,0)
    label.Font = Enum.Font.GothamBold label.TextSize = 14
    label.TextColor3 = Color3.fromRGB(180,120,255)
    label.TextStrokeTransparency = 0 label.TextStrokeColor3 = Color3.fromRGB(0,0,0)
    label.Text = "☄ Meteor" label.Parent = frame
    local conn = RunService.Heartbeat:Connect(function()
        if not meteorESPEnabled or not core or not core.Parent then return end
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then label.Text = "☄ Meteor\n" .. math.floor((hrp.Position - core.Position).Magnitude) .. " studs" end
    end)
    local key = meteor:GetFullName()
    meteorRemovedConnections[key] = meteor.AncestryChanged:Connect(function()
        if not meteor.Parent then
            conn:Disconnect()
            if meteorRemovedConnections[key] then meteorRemovedConnections[key]:Disconnect() meteorRemovedConnections[key] = nil end
        end
    end)
end

local function removeAllMeteorESP()
    for _, meteor in ipairs(workspace.Map.Meteors:GetChildren()) do
        local core = meteor:FindFirstChild("Core")
        if core then local bb = core:FindFirstChild("MeteorESP") if bb then bb:Destroy() end end
    end
    for _, conn in pairs(meteorRemovedConnections) do conn:Disconnect() end
    meteorRemovedConnections = {}
end

local function startMeteorESP()
    for _, meteor in ipairs(workspace.Map.Meteors:GetChildren()) do createMeteorESP(meteor) end
    meteorAddedConnection = workspace.Map.Meteors.ChildAdded:Connect(function(meteor)
        if meteorESPEnabled then task.wait(0.2) createMeteorESP(meteor) end
    end)
end

local function stopMeteorESP()
    if meteorAddedConnection then meteorAddedConnection:Disconnect() meteorAddedConnection = nil end
    removeAllMeteorESP()
end

local function fileExists(path)
    local ok, r = pcall(function() return isfile(path) end) return ok and r
end
local function loadAllConfigs()
    if not fileExists(CONFIG_FILE) then return {} end
    local ok, data = pcall(function() return readfile(CONFIG_FILE) end)
    if not ok or not data or data == "" then return {} end
    local ok2, parsed = pcall(function() return HttpService:JSONDecode(data) end)
    if ok2 and type(parsed) == "table" then return parsed end
    return {}
end
local function saveAllConfigs(configs)
    local ok, encoded = pcall(function() return HttpService:JSONEncode(configs) end)
    if not ok then Rayfield:Notify({ Title = "Config", Content = "Encode failed!", Duration = 4 }) return false end
    local ok2 = pcall(function() writefile(CONFIG_FILE, encoded) end)
    if not ok2 then Rayfield:Notify({ Title = "Config", Content = "Write failed!", Duration = 5 }) return false end
    local ok3, verify = pcall(function() return readfile(CONFIG_FILE) end)
    if not ok3 or verify ~= encoded then Rayfield:Notify({ Title = "Config", Content = "Verify failed!", Duration = 5 }) return false end
    return true
end
local function getCurrentFlags()
    return {
        AutoPerfectLaunch    = toggleStates.AutoPerfectLaunch,
        AutoBuyBurger        = toggleStates.AutoBuyBurger,
        AutoTrashFarm        = toggleStates.AutoTrashFarm,
        AutoTrashFarmBlatant = toggleStates.AutoTrashFarmBlatant,
        MeteorESP            = toggleStates.MeteorESP,
    }
end
local function applyFlags(flags)
    for flag, value in pairs(flags) do
        local cb = toggleCallbacks[flag]
        if cb then
            pcall(function() if Rayfield.Flags[flag] then Rayfield.Flags[flag]:Set(value) end end)
            if value ~= toggleStates[flag] then cb(value) end
        end
    end
end
local function getConfigNames()
    local configs = loadAllConfigs() local names = {}
    for name in pairs(configs) do table.insert(names, name) end
    if #names == 0 then table.insert(names, "No Configs Saved") end
    return names
end

local ConfigTab = Window:CreateTab("Config", 4483362458)
ConfigTab:CreateSection("Saved Configs")
local selectedConfig = ""
local configDropdown = ConfigTab:CreateDropdown({
    Name = "Select Config", Options = getConfigNames(), CurrentOption = {"No Configs Saved"},
    Flag = "SelectedConfig", MultipleOptions = false,
    Callback = function(s) selectedConfig = type(s) == "table" and s[1] or s end,
})
ConfigTab:CreateButton({ Name = "Load Selected Config", Callback = function()
    if selectedConfig == "" or selectedConfig == "No Configs Saved" then
        Rayfield:Notify({ Title = "Config", Content = "Select a config first!", Duration = 3 }) return
    end
    local cfg = loadAllConfigs()[selectedConfig]
    if cfg and cfg.flags then
        applyFlags(cfg.flags)
        local summary = "" for flag, val in pairs(cfg.flags) do if val then summary = summary .. flag .. " " end end
        Rayfield:Notify({ Title = "Config Loaded: " .. selectedConfig, Content = summary ~= "" and ("ON: " .. summary) or "All off", Duration = 5 })
    else Rayfield:Notify({ Title = "Config", Content = "Not found!", Duration = 3 }) end
end })
ConfigTab:CreateButton({ Name = "Delete Selected Config", Callback = function()
    if selectedConfig == "" or selectedConfig == "No Configs Saved" then
        Rayfield:Notify({ Title = "Config", Content = "Select a config to delete!", Duration = 3 }) return
    end
    local configs = loadAllConfigs()
    if configs[selectedConfig] then
        configs[selectedConfig] = nil saveAllConfigs(configs)
        Rayfield:Notify({ Title = "Config", Content = "Deleted: " .. selectedConfig, Duration = 3 })
        selectedConfig = "" configDropdown:Refresh(getConfigNames(), {"No Configs Saved"})
    else Rayfield:Notify({ Title = "Config", Content = "Not found!", Duration = 3 }) end
end })
ConfigTab:CreateSection("Create Config")
local configNameInput = ""
ConfigTab:CreateInput({ Name = "Config Name", PlaceholderText = "Enter config name...", RemoveTextAfterFocusLost = false,
    Callback = function(text) configNameInput = text end })
local autoLoadOnSave = false
ConfigTab:CreateToggle({ Name = "Auto Load on Execute", CurrentValue = false, Flag = "ConfigAutoLoad",
    Callback = function(value) autoLoadOnSave = value end })
ConfigTab:CreateButton({ Name = "Save Config", Callback = function()
    local name = configNameInput
    if not name or name == "" then Rayfield:Notify({ Title = "Config", Content = "Enter a name first!", Duration = 3 }) return end
    local configs = loadAllConfigs()
    if autoLoadOnSave then for _, v in pairs(configs) do v.autoLoad = false end end
    local flags = getCurrentFlags()
    configs[name] = { flags = flags, autoLoad = autoLoadOnSave }
    if saveAllConfigs(configs) then
        local summary = "" for flag, val in pairs(flags) do if val then summary = summary .. flag .. " " end end
        Rayfield:Notify({ Title = "Saved: " .. name, Content = (summary ~= "" and ("ON: " .. summary) or "All off") .. (autoLoadOnSave and " | Auto Load ON" or ""), Duration = 5 })
        configDropdown:Refresh(getConfigNames(), {name}) selectedConfig = name
    end
end })

task.spawn(function()
    task.wait(2)
    local configs = loadAllConfigs() local names = getConfigNames()
    configDropdown:Refresh(names, {names[1]})
    selectedConfig = names[1] ~= "No Configs Saved" and names[1] or ""
    for name, cfg in pairs(configs) do
        if cfg.autoLoad and cfg.flags then
            task.wait(0.5) applyFlags(cfg.flags)
            Rayfield:Notify({ Title = "Config", Content = "Auto loaded: " .. name, Duration = 4 }) break
        end
    end
end)

Rayfield:Notify({ Title = "Loaded", Content = "Totally Accurate Tops script loaded.", Duration = 3 })
