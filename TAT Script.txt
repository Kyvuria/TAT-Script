-- ── Pre-load config BEFORE building UI so CurrentValue is correct from the start ──
local HttpService = game:GetService("HttpService")
local CONFIG_FILE = "TAT_Configs.json"

local function loadAllConfigs()
    local ok, data = pcall(function() return readfile(CONFIG_FILE) end)
    if not ok or not data or data == "" then return {} end
    local ok2, parsed = pcall(function() return HttpService:JSONDecode(data) end)
    if ok2 and type(parsed) == "table" then return parsed end
    return {}
end

local function saveAllConfigs(configs)
    local ok, encoded = pcall(function() return HttpService:JSONEncode(configs) end)
    if not ok then return false end
    local ok2 = pcall(function() writefile(CONFIG_FILE, encoded) end)
    return ok2
end

-- ── Meteor time persistence ───────────────────────────────────────────────────
local METEOR_TIME_FILE = "TAT_MeteorTime.json"

local function loadLastMeteorTime()
    local ok, data = pcall(function() return readfile(METEOR_TIME_FILE) end)
    if not ok or not data or data == "" then return nil end
    local ok2, parsed = pcall(function() return HttpService:JSONDecode(data) end)
    if ok2 and type(parsed) == "table" and parsed.lastSpawn then
        return parsed.lastSpawn
    end
    return nil
end

local function saveLastMeteorTime(t)
    pcall(function()
        writefile(METEOR_TIME_FILE, HttpService:JSONEncode({ lastSpawn = t }))
    end)
end

local function randomWait(mn, mx) task.wait(mn + math.random() * (mx - mn)) end

local function maybeJump(humanoid)
    if math.random(1,3) == 1 then humanoid.Jump = true task.wait(0.1) humanoid.Jump = false end
end

local function checkStuck(hrp, humanoid, target)
    local cur = hrp.Position
    if lastPos then
        if (cur - lastPos).Magnitude < 0.5 then
            stuckCount += 1
            if stuckCount >= 2 then
                humanoid.Jump = true task.wait(0.2) humanoid.Jump = false
                humanoid:MoveTo(cur + Vector3.new(math.random(-4,4), 0, math.random(-4,4)))
                task.wait(0.3) humanoid:MoveTo(target) stuckCount = 0
            end
        else stuckCount = 0 end
    end
    lastPos = cur
end

local function walkTo(hrp, humanoid, target, timeout, blacklistKey, farmCheck)
    stuckCount = 0 lastPos = nil
    humanoid:MoveTo(target)
    local t = tick() + (timeout or 20)
    local startPos = hrp.Position
    repeat
        task.wait(0.3)
        if farmCheck and not farmCheck() then return false end
        checkStuck(hrp, humanoid, target) maybeJump(humanoid)
    until (hrp.Position - target).Magnitude < 8 or tick() > t
    local arrived = (hrp.Position - target).Magnitude < 8
    if not arrived and blacklistKey and (hrp.Position - startPos).Magnitude < 15 then
        trashBlacklist[blacklistKey] = true
    end
    return arrived
end

local function goUnderground(hrp)
    if isUnderground then return end
    local dest = Vector3.new(hrp.Position.X, UNDERGROUND_Y, hrp.Position.Z)
    local dur = math.max((dest - hrp.Position).Magnitude / SAFE_SPEED, 0.05)
    local tween = TweenService:Create(hrp, TweenInfo.new(dur, Enum.EasingStyle.Linear), {CFrame = CFrame.new(dest)})
    tween:Play() tween.Completed:Wait() isUnderground = true
end

local function surfaceUp(hrp)
    if not isUnderground then return end
    local dest = Vector3.new(hrp.Position.X, UNDERGROUND_Y + 8, hrp.Position.Z)
    local dur = math.max((dest - hrp.Position).Magnitude / SAFE_SPEED, 0.05)
    local tween = TweenService:Create(hrp, TweenInfo.new(dur, Enum.EasingStyle.Linear), {CFrame = CFrame.new(dest)})
    tween:Play() tween.Completed:Wait() isUnderground = false
end

local function touchGround(hrp)
    if isTouchingGround then return end
    isTouchingGround = true
    local downDest = Vector3.new(hrp.Position.X, GROUND_Y, hrp.Position.Z)
    local downTween = TweenService:Create(hrp, TweenInfo.new(math.max((downDest-hrp.Position).Magnitude/SAFE_SPEED,0.05), Enum.EasingStyle.Linear), {CFrame=CFrame.new(downDest)})
    downTween:Play() downTween.Completed:Wait() task.wait(0.1)
    local upDest = Vector3.new(hrp.Position.X, UNDERGROUND_Y, hrp.Position.Z)
    local upTween = TweenService:Create(hrp, TweenInfo.new(math.max((upDest-hrp.Position).Magnitude/SAFE_SPEED,0.05), Enum.EasingStyle.Linear), {CFrame=CFrame.new(upDest)})
    upTween:Play() upTween.Completed:Wait()
    lastGroundTouch = tick() isTouchingGround = false
end

local function blatantTweenTo(hrp, targetPos, farmCheck)
    local function makeTween()
        local dest = Vector3.new(targetPos.X, UNDERGROUND_Y, targetPos.Z)
        local dist = (dest - hrp.Position).Magnitude
        if dist < 1 then return nil end
        return TweenService:Create(hrp, TweenInfo.new(math.max(dist/SAFE_SPEED,0.05), Enum.EasingStyle.Linear), {CFrame=CFrame.new(dest)})
    end
    local tween = makeTween() if not tween then return end
    local tweenDone = false
    tween.Completed:Connect(function() tweenDone = true end) tween:Play()
    while not tweenDone do
        task.wait(0.5)
        if not farmCheck() then tween:Cancel() return end
        if tick() - lastGroundTouch >= GROUND_TOUCH_INTERVAL and not isTouchingGround then
            tween:Cancel() touchGround(hrp)
            tween = makeTween() if not tween then return end
            tweenDone = false
            tween.Completed:Connect(function() tweenDone = true end) tween:Play()
        end
    end
end

local function blatantWalkTo(hrp, humanoid, target, timeout, blacklistKey, farmCheck)
    if not isUnderground then goUnderground(hrp) end
    blatantTweenTo(hrp, target, farmCheck)
    local arrived = (Vector3.new(hrp.Position.X,0,hrp.Position.Z) - Vector3.new(target.X,0,target.Z)).Magnitude < 8
    if not arrived and blacklistKey then trashBlacklist[blacklistKey] = true end
    return arrived
end

local function getTrashCount(waitForValid)
    local function read()
        local ok, label = pcall(function() return playerGui.HUD["Main HUD"].Corner.Trash.Label end)
        if not ok or not label then return 0, 0 end
        local cur, max = label.Text:match("(%d+)%s*/%s*(%d+)")
        return tonumber(cur) or 0, tonumber(max) or 0
    end
    if waitForValid then
        local deadline = tick() + 3
        local c, m = read()
        while m == 0 and tick() < deadline do task.wait(0.2) c, m = read() end
        return c, m
    end
    return read()
end

local function isDumpsterOnCooldown(part)
    local t = part:FindFirstChild("TimerGUI")
    return t and t.Enabled == true
end

local function getAvailableDumpster()
    local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local best, bestDist = nil, math.huge
    for _, part in ipairs(workspace.Map.Environment.Dumpsters:GetChildren()) do
        if part:IsA("Part") and not isDumpsterOnCooldown(part) then
            local prompt = part:FindFirstChild("Prompt")
            if prompt and prompt.Enabled then
                local d = (hrp.Position - part.Position).Magnitude
                if d < bestDist then best = {part=part, pos=part.Position} bestDist = d end
            end
        end
    end
    return best
end

local function getNextAvailableDumpsterTime()
    local soonest = nil
    for _, part in ipairs(workspace.Map.Environment.Dumpsters:GetChildren()) do
        if part:IsA("Part") then
            local tg = part:FindFirstChild("TimerGUI")
            if tg and tg.Enabled then
                local lbl = tg:FindFirstChild("Label")
                if lbl and lbl.Text ~= "" then
                    local m, s = lbl.Text:match("(%d+):(%d+)")
                    if m and s then
                        local secs = tonumber(m)*60 + tonumber(s)
                        if not soonest or secs < soonest then soonest = secs end
                    end
                end
            else return 0 end
        end
    end
    return soonest or 0
end

local function runFarmLoop(farmCheck, blatant)
    local function getChar()
        local c = LocalPlayer.Character
        if not c then return nil, nil end
        return c:FindFirstChild("HumanoidRootPart"), c:FindFirstChild("Humanoid")
    end
    local lastTrashCount = -1
    local lastIncreaseTime = tick()
    local hasPickedUpAtLeastOne = false
    startAutoSprint(farmCheck)
    if blatant then startNoclip() lastGroundTouch = tick() end
    while farmCheck() do
        local hrp, humanoid = getChar()
        if not hrp or not humanoid then task.wait(1) continue end
        local cur, max = getTrashCount(true)
        if cur ~= lastTrashCount then lastTrashCount = cur lastIncreaseTime = tick() end
        if tick() - lastIncreaseTime > 10 then
            Notify("Trash Farm", "Stuck! Resetting...", 3)
            if isUnderground then surfaceUp(hrp) end
            LocalPlayer.Character:BreakJoints()
            task.wait(4) isUnderground = false
            startAutoSprint(farmCheck)
            if blatant then startNoclip() lastGroundTouch = tick() end
            lastIncreaseTime = tick() lastTrashCount = -1 hasPickedUpAtLeastOne = false
            continue
        end
        if hasPickedUpAtLeastOne and max > 0 and cur >= max then
            if isUnderground then local h = getChar() if h then surfaceUp(h) end end
            local dumpster = getAvailableDumpster()
            if dumpster then
                local arrived = walkTo(hrp, humanoid, dumpster.pos, 30, nil, farmCheck)
                if arrived then
                    randomWait(0.3, 0.7)
                    local prompt = dumpster.part:FindFirstChild("Prompt")
                    if prompt then
                        fireproximityprompt(prompt, 2) task.wait(2)
                        trashBlacklist = {} lastTrashCount = -1
                        lastIncreaseTime = tick() hasPickedUpAtLeastOne = false
                    end
                end
            else
                local waitSecs = math.max(getNextAvailableDumpsterTime() or 60, 5)
                Notify("Trash Farm", "Dumpsters on cooldown! Waiting " .. math.ceil(waitSecs) .. "s", 5)
                local waitEnd = tick() + waitSecs + 2
                repeat task.wait(2) if getAvailableDumpster() then break end
                until tick() > waitEnd or not farmCheck()
                lastIncreaseTime = tick()
            end
        else
            local hrp2, humanoid2 = getChar()
            if not hrp2 or not humanoid2 then task.wait(1) continue end
            local nearest, nearestDist, nearestPrompt, nearestKey = nil, math.huge, nil, nil
            for _, paper in ipairs(workspace.Map.Environment.Trash:GetChildren()) do
                if trashBlacklist[paper.Name] then continue end
                local prompt = paper:FindFirstChild("Prompt")
                if prompt and prompt.Enabled then
                    local d = (hrp2.Position - paper.Position).Magnitude
                    if d < nearestDist then
                        nearest = paper nearestDist = d nearestPrompt = prompt nearestKey = paper.Name
                    end
                end
            end
            if nearest then
                local arrived
                if blatant then
                    arrived = blatantWalkTo(hrp2, humanoid2, nearest.Position, 20, nearestKey, farmCheck)
                else
                    arrived = walkTo(hrp2, humanoid2, nearest.Position, 20, nearestKey, farmCheck)
                end
                if arrived and nearestPrompt and nearestPrompt.Enabled then
                    fireproximityprompt(nearestPrompt)
                    hasPickedUpAtLeastOne = true lastIncreaseTime = tick() task.wait(0.3)
                elseif not arrived then lastIncreaseTime = tick() end
            else task.wait(3) end
        end
        task.wait(0.1)
    end
    stopAutoSprint()
    if blatant then
        stopNoclip()
        local h = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if h and isUnderground then surfaceUp(h) end
        isUnderground = false
    end
end

local savedLaunchCFrame = nil

local function doAutoLaunchCycle()
    if isDoingLaunchCycle then return end
    isDoingLaunchCycle = true
    local character = LocalPlayer.Character
    local hrp = character and character:FindFirstChild("HumanoidRootPart")
    local humanoid = character and character:FindFirstChild("Humanoid")
    if not hrp or not humanoid then isDoingLaunchCycle = false return end
    if not savedLaunchCFrame then savedLaunchCFrame = hrp.CFrame end
    if isTopInArena() then
        local battleTimeout = tick() + 180
        repeat
            task.wait(0.5)
            if not autoLaunchRunning then isDoingLaunchCycle = false return end
        until not isTopInArena() or tick() > battleTimeout
        if not autoLaunchRunning then isDoingLaunchCycle = false return end
        task.wait(1.5)
    end
    if getMyTop() ~= nil then
        local pickupTimeout = tick() + 30
        repeat task.wait(0.3) until topNeedsPickup() or topIsInHand() or tick() > pickupTimeout
        if topNeedsPickup() then
            local topPos = getTopPosition()
            if topPos then
                stuckCount = 0 lastPos = nil
                humanoid:MoveTo(topPos)
                local wt = tick() + 25
                repeat
                    task.wait(0.3)
                    if not autoLaunchRunning then isDoingLaunchCycle = false return end
                    local np = getTopPosition()
                    if np then topPos = np humanoid:MoveTo(topPos) end
                    checkStuck(hrp, humanoid, topPos) maybeJump(humanoid)
                until (hrp.Position - (getTopPosition() or topPos)).Magnitude < 5 or tick() > wt
                task.wait(0.3) pressKey(Enum.KeyCode.E) task.wait(0.8)
            end
        end
    end
    local handTimeout = tick() + 10
    repeat task.wait(0.3) until topIsInHand() or tick() > handTimeout
    if not topIsInHand() then
        Notify("Auto Launch", "Pickup failed, retrying...", 3)
        isDoingLaunchCycle = false
        if autoLaunchRunning then task.wait(2) task.spawn(doAutoLaunchCycle) end
        return
    end
    if not autoLaunchRunning then isDoingLaunchCycle = false return end
    local standPos = savedLaunchCFrame.Position
    stuckCount = 0 lastPos = nil
    humanoid:MoveTo(standPos)
    local returnTimeout = tick() + 20
    repeat
        task.wait(0.3)
        if not autoLaunchRunning then isDoingLaunchCycle = false return end
        checkStuck(hrp, humanoid, standPos) maybeJump(humanoid)
    until (hrp.Position - standPos).Magnitude < 4 or tick() > returnTimeout
    hrp.CFrame = savedLaunchCFrame
    task.wait(2)
    if not autoLaunchRunning then isDoingLaunchCycle = false return end
    pressKey(Enum.KeyCode.E) task.wait(1.5)
    if not autoLaunchRunning then isDoingLaunchCycle = false return end
    VirtualInputManager:SendMouseButtonEvent(0,0,0,true,game,1) task.wait(0.05)
    VirtualInputManager:SendMouseButtonEvent(0,0,0,false,game,1)
    isDoingLaunchCycle = false
    if autoLaunchRunning then task.wait(2) task.spawn(doAutoLaunchCycle) end
end

local isBurgerBuying = false

local function doBurgerRun(manual)
    if isBurgerBuying then return end
    isBurgerBuying = true isManualRun = manual or false manualCancelled = false
    local character = LocalPlayer.Character
    local hrp = character and character:FindFirstChild("HumanoidRootPart")
    local humanoid = character and character:FindFirstChild("Humanoid")
    if not hrp or not humanoid then isBurgerBuying = false return end
    local originalPos = hrp.Position
    local standPos = Vector3.new(573.88, 118, -157.43)
    local keyConnection
    if isManualRun then
        keyConnection = UserInputService.InputBegan:Connect(function(input, gp)
            if gp then return end
            if input.UserInputType == Enum.UserInputType.Keyboard then
                manualCancelled = true humanoid:MoveTo(hrp.Position)
            end
        end)
    end
    stuckCount = 0 lastPos = nil
    humanoid:MoveTo(standPos)
    local timeout = tick() + 12
    repeat
        task.wait(0.3) if manualCancelled then break end
        checkStuck(hrp, humanoid, standPos) maybeJump(humanoid)
    until (hrp.Position - standPos).Magnitude < 3 or tick() > timeout or manualCancelled
    if not manualCancelled then
        randomWait(0.4, 0.9)
        local bought = false
        local ok, vendor = pcall(function() return workspace.Map["Food Vendors"]:GetChildren()[11] end)
        if ok and vendor then
            local prompt = vendor.Counter["Prompt Host"].Prompt
            if prompt and not bought then bought = true fireproximityprompt(prompt) end
        end
        if not isManualRun then
            randomWait(2.8, 3.4)
            local returnPos = originalPos + Vector3.new(math.random(-2,2), 0, math.random(-2,2))
            stuckCount = 0 lastPos = nil
            humanoid:MoveTo(returnPos)
            local returnTimeout = tick() + 12
            repeat task.wait(0.3) checkStuck(hrp, humanoid, returnPos) maybeJump(humanoid)
            until (hrp.Position - returnPos).Magnitude < 3 or tick() > returnTimeout
        end
    end
    if keyConnection then keyConnection:Disconnect() end
    isBurgerBuying = false isManualRun = false manualCancelled = false
end

local function buildACBypassTimerGui()
    local old = playerGui:FindFirstChild("ACBypassTimerGui") if old then old:Destroy() end
    local sg = Instance.new("ScreenGui")
    sg.Name = "ACBypassTimerGui" sg.ResetOnSpawn = false sg.DisplayOrder = 1000 sg.Parent = playerGui
    local frame = Instance.new("Frame")
    frame.BackgroundColor3 = Color3.fromRGB(10,10,20) frame.BackgroundTransparency = 0.3
    frame.BorderSizePixel = 0 frame.Size = UDim2.new(0,300,0,90)
    frame.Position = UDim2.new(0.5,-150,0,12) frame.Parent = sg
    local corner = Instance.new("UICorner") corner.CornerRadius = UDim.new(0,8) corner.Parent = frame
    -- Drag support
    local dragHandle = Instance.new("TextButton")
    dragHandle.Name = "DragHandle" dragHandle.BackgroundTransparency = 1
    dragHandle.Size = UDim2.new(1,0,0,34) dragHandle.Position = UDim2.new(0,0,0,0)
    dragHandle.Text = "" dragHandle.Parent = frame
    local dragging = false local dragStart = nil local startPos = nil
    dragHandle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true dragStart = input.Position startPos = frame.Position
        end
    end)
    dragHandle.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
    local titleLabel = Instance.new("TextLabel")

local function runACBypassTimer()
    buildACBypassTimerGui()
    -- Freeze character by zeroing speed only — NO anchoring (anchoring causes physics launch on release)
    local char = LocalPlayer.Character
    local humanoid = char and char:FindFirstChild("Humanoid")
    if humanoid then
        humanoid.WalkSpeed = 0
        humanoid.UseJumpPower = true
        humanoid.JumpPower = 0
    end
    -- Keep speed zeroed every frame so the server can't override it during the 10s
    local freezeConn = RunService.Heartbeat:Connect(function()
        local c = LocalPlayer.Character
        local h = c and c:FindFirstChild("Humanoid")
        if h then h.WalkSpeed = 0 h.JumpPower = 0 end
    end)

    local startTick = tick()
    if acBypassTimerConnection then acBypassTimerConnection:Disconnect() end
    acBypassTimerConnection = RunService.Heartbeat:Connect(function()
        if not acBypassTimerGui then return end
        local elapsed = tick() - startTick
        local left = math.max(0, 10 - elapsed)
        acBypassTimerGui.countdown.Text = tostring(math.ceil(left))
        local pct = left / 10
        if pct > 0.6 then acBypassTimerGui.countdown.TextColor3 = Color3.fromRGB(100,200,255)
        elseif pct > 0.3 then acBypassTimerGui.countdown.TextColor3 = Color3.fromRGB(255,165,0)
        else acBypassTimerGui.countdown.TextColor3 = Color3.fromRGB(255,60,60) end
        if left <= 0 then
            acBypassTimerConnection:Disconnect() acBypassTimerConnection = nil
            freezeConn:Disconnect()
            -- Restore movement
            local c2 = LocalPlayer.Character
            local h2 = c2 and c2:FindFirstChild("Humanoid")
            if h2 then h2.WalkSpeed = currentWSValue h2.UseJumpPower = true h2.JumpPower = currentJPValue end
            acBypassTimerGui.countdown.Text = "GO!"
            acBypassTimerGui.countdown.TextColor3 = Color3.fromRGB(100,255,100)
            acBypassTimerGui.status.Text = "Safe to move now"
            acBypassTimerGui.status.TextColor3 = Color3.fromRGB(100,255,100)
            task.wait(1.5)
            destroyACBypassTimerGui()
        end
    end)
end

local function formatTime(secs)
    secs = math.max(0, math.floor(secs))
    return string.format("%02d : %02d", math.floor(secs/60), secs%60)
end

local function isRealMeteor(child)
    return child:IsA("Model") or child:IsA("Part") or child:IsA("BasePart")
end

local function getLiveMeteorCount()
    local count = 0
    for _, child in ipairs(workspace.Map.Meteors:GetChildren()) do
        if isRealMeteor(child) then count += 1 end
    end
    return count
end

local function startMeteorTimer()
    buildMeteorTimerGui()
    local waitingTaskActive = false
    local function scheduleWaitingStatus()
        if waitingTaskActive then return end
        waitingTaskActive = true
        task.spawn(function()
            task.wait(30)
            if not waitingTaskActive then return end
            if not meteorESPEnabled then waitingTaskActive = false return end
            if getLiveMeteorCount() == 0 and meteorOverrideText == "" then
                setMeteorStatus("Waiting for meteors to spawn...", Color3.fromRGB(180,180,180))
            end
            waitingTaskActive = false
        end)
    end
    if not lastMeteorCycleStart then
        local savedUnix = loadLastMeteorTime()
        if savedUnix then
            local secondsSince = (os.time() - savedUnix) % METEOR_INTERVAL
            lastMeteorCycleStart = tick() - secondsSince
            meteorCycleResultShown = false
            local remaining = math.floor(METEOR_INTERVAL - secondsSince)
            Notify("Meteor Timer", "Synced! ~" .. math.floor(remaining/60) .. "m " .. (remaining%60) .. "s remaining.", 5)
        else
            lastMeteorCycleStart = tick()
            meteorCycleResultShown = false
        end
    end
    local existingCount = getLiveMeteorCount()
    if existingCount > 0 then
        meteorOverrideText  = "Spawned! (" .. existingCount .. ")"
        meteorOverrideColor = Color3.fromRGB(100,255,100)
    else
        meteorOverrideText = ""
    end
    isPlayerCountLow = getServerPlayerCount() < MIN_PLAYERS_REQUIRED
    if isPlayerCountLow then
        setMeteorStatus("Paused – need " .. MIN_PLAYERS_REQUIRED .. "+ players (" .. getServerPlayerCount() .. " now)", Color3.fromRGB(200,180,0))
    end
    updateStatusLabel()
    meteorTimerAddedConn = workspace.Map.Meteors.ChildAdded:Connect(function(child)
        if not isRealMeteor(child) then return end
        task.wait(0.2) if not meteorESPEnabled then return end
        waitingTaskActive = false
        local count = getLiveMeteorCount()
        meteorOverrideText  = "Spawned! (" .. count .. ")"
        meteorOverrideColor = Color3.fromRGB(100,255,100)
        setMeteorStatus(meteorOverrideText, meteorOverrideColor)
        lastMeteorCycleStart = tick() meteorCycleResultShown = false
        saveLastMeteorTime(os.time())
    end)
    meteorTimerRemovedConn = workspace.Map.Meteors.ChildRemoved:Connect(function(child)
        if not isRealMeteor(child) then return end
        task.wait(0.2) if not meteorESPEnabled then return end
        local remaining = getLiveMeteorCount()
        if remaining > 0 then
            meteorOverrideText  = "Spawned! (" .. remaining .. ")"
            meteorOverrideColor = Color3.fromRGB(100,255,100)
            setMeteorStatus(meteorOverrideText, meteorOverrideColor)
        else
            meteorOverrideText = ""
            if not isPlayerCountLow then
                setMeteorStatus("All meteors picked up", Color3.fromRGB(255,140,0))
                scheduleWaitingStatus()
            else updateStatusLabel() end
        end
    end)
    meteorTimerConnection = RunService.Heartbeat:Connect(function()
        if not meteorTimerGui then return end
        local playerCount = getServerPlayerCount()
        local lowNow = playerCount < MIN_PLAYERS_REQUIRED
        if lowNow ~= isPlayerCountLow then
            isPlayerCountLow = lowNow
            if lowNow then Notify("Meteor Timer", "Paused – only " .. playerCount .. " players.", 4)
            else Notify("Meteor Timer", "Resumed – " .. playerCount .. " players.", 4) end
            updateStatusLabel()
        end
        if isPlayerCountLow then
            meteorTimerGui.countdown.Text = "-- : -- ⏸"
            meteorTimerGui.countdown.TextColor3 = Color3.fromRGB(200,180,0)
        else
            if not lastMeteorCycleStart then return end
            local elapsed   = tick() - lastMeteorCycleStart
            local remaining = METEOR_INTERVAL - elapsed
            if remaining <= 0 then
                if not meteorCycleResultShown then
                    meteorCycleResultShown = true
                    if getLiveMeteorCount() == 0 and meteorOverrideText == "" then
                        setMeteorStatus("Not Spawned", Color3.fromRGB(255,80,80))
                    end
                    lastMeteorCycleStart = tick() meteorCycleResultShown = false
                end
                meteorTimerGui.countdown.Text = "00 : 00"
                meteorTimerGui.countdown.TextColor3 = Color3.fromRGB(255,60,60)
            else
                meteorCycleResultShown = false
                meteorTimerGui.countdown.Text = formatTime(remaining)
                local pct = remaining / METEOR_INTERVAL
                if pct > 0.5 then meteorTimerGui.countdown.TextColor3 = Color3.fromRGB(180,120,255)
                elseif pct > 0.2 then meteorTimerGui.countdown.TextColor3 = Color3.fromRGB(255,165,0)
                else meteorTimerGui.countdown.TextColor3 = Color3.fromRGB(255,60,60) end
            end
        end
        if meteorTimerGui.status.Text ~= meteorStatusText then meteorTimerGui.status.Text = meteorStatusText end
    end)
end

local function stopMeteorTimer()
    if meteorTimerConnection  then meteorTimerConnection:Disconnect()  meteorTimerConnection  = nil end
    if meteorTimerAddedConn   then meteorTimerAddedConn:Disconnect()   meteorTimerAddedConn   = nil end
    if meteorTimerRemovedConn then meteorTimerRemovedConn:Disconnect() meteorTimerRemovedConn = nil end
    destroyMeteorTimerGui()
    isPlayerCountLow = false meteorOverrideText = "" meteorStatusText = ""
end

-- ── Top Inspector (right side of ESP tab) ────────────────────────────────────
local topInspectorEnabled = false
local topInspectorConnections = {}
local topBBCache = {}
local topBuildLock = {}
local topInspectorScale = 1.0  -- UI scale multiplier

local function getOwnerName(top)
    local ownerVal = top:FindFirstChild("Owner")
    if ownerVal and ownerVal.Value then return ownerVal.Value.Name end
    local uid = tonumber(top.Name)
    if uid then
        local p = Players:GetPlayerByUserId(uid)
        if p then return p.Name end
        return "ID:"..top.Name
    end
    return top.Name
end

local function getPartName(top, partName)
    local part = top:FindFirstChild(partName) if not part then return "?" end
    local n = part:GetAttribute("Name")
    if n and tostring(n) ~= "" then return tostring(n) end
    return "?"
end

local function getRPMBarColor(pct)
    if pct >= 1.0 then return Color3.fromRGB(180,80,255)
    elseif pct >= 0.50 then return Color3.fromRGB(0,200,255)
    elseif pct >= 0.25 then return Color3.fromRGB(255,165,0)
    else return Color3.fromRGB(255,50,50) end
end

local RPM_MULT = 150
local LIVE_STATES = { Attacking = true, Guarding = true, Idle = true }

local function getLiveRPM(top)
    local state = tostring(top:GetAttribute("State") or "")
    if not LIVE_STATES[state] then return nil end
    local health = top:GetAttribute("Health")
    if health and health > 0 then return math.floor(health * 20) end
    return 0
end

local function getLiveMaxRPM(top)
    local mh = top:GetAttribute("MaxHealth")
    if mh and mh > 0 then return math.max(math.floor(mh * 20), 1) end
    local vc = tonumber(top:GetAttribute("VictoryCents")) or 60
    return math.max(vc * RPM_MULT, 1)
end

local function buildTopBillboard(top)
    if topBuildLock[top] then return end
    topBuildLock[top] = true
    if top:GetAttribute("TaggedNPC") then topBuildLock[top] = nil return end
    if topBBCache[top] then
        pcall(function() topBBCache[top].rpmConn:Disconnect() end)
        pcall(function() topBBCache[top].bb:Destroy() end)
        topBBCache[top] = nil
    end
    local attachPart = top:FindFirstChild("Body") or top:FindFirstChild("Crown") or top:FindFirstChildWhichIsA("BasePart")
    if not attachPart then topBuildLock[top] = nil return end
    local attrs = top:GetAttributes()
    local crownName = getPartName(top, "Crown") local bodyName = getPartName(top, "Body")
    local beltName = getPartName(top, "Belt") local pointName = getPartName(top, "Point")
    local spirit = attrs["Spirit"] or "?" local sig1 = attrs["SignatureMove_1"] or "?"
    local sig2 = attrs["SignatureMove_2"] or "?" local sig3 = attrs["SignatureMove_3"] or "?"
    local atk = attrs["Attack"] or 0 local def = attrs["Defense"] or 0 local sta = attrs["Stamina"] or 0

    local sc = topInspectorScale
    local BASE_W = 260 local LINE_H = 14 local PAD = 4 local NUM_LINES = 10
    local RPM_LABEL_H = 20 local RPM_BAR_H = 22 local RPM_GAP = 6
    local totalH = PAD + (LINE_H * NUM_LINES) + RPM_GAP + RPM_LABEL_H + RPM_BAR_H + PAD
    local bb = Instance.new("BillboardGui")
    bb.Name = "TopInspectorBB" bb.AlwaysOnTop = true
    bb.Size = UDim2.new(0, math.floor(BASE_W * sc), 0, math.floor(totalH * sc))
    bb.StudsOffset = Vector3.new(0,5,0) bb.MaxDistance = 0 bb.Adornee = attachPart bb.Parent = attachPart
    local bg = Instance.new("Frame")
    bg.BackgroundColor3 = Color3.fromRGB(10,10,20) bg.BackgroundTransparency = 0.3
    bg.BorderSizePixel = 0 bg.Size = UDim2.new(1,0,1,0) bg.Parent = bb
    local corner = Instance.new("UICorner") corner.CornerRadius = UDim.new(0,4) corner.Parent = bg
    local yPos = PAD
    local baseTextSize = 11
    local function makeLine(text, color, rich)
        local lbl = Instance.new("TextLabel")
        lbl.BackgroundTransparency = 1 lbl.Size = UDim2.new(1,-6,0, math.floor(LINE_H * sc))
        lbl.Position = UDim2.new(0,3,0, math.floor(yPos * sc)) lbl.Font = Enum.Font.GothamBold
        lbl.TextSize = math.floor(baseTextSize * sc)
        lbl.TextColor3 = color lbl.TextStrokeTransparency = 0.5 lbl.TextStrokeColor3 = Color3.fromRGB(0,0,0)
        lbl.TextXAlignment = Enum.TextXAlignment.Left lbl.RichText = rich or false lbl.Text = text lbl.Parent = bb
        yPos += LINE_H return lbl
    end
    makeLine(getOwnerName(top), Color3.fromRGB(255,255,255))
    makeLine("Spirit: "..spirit, Color3.fromRGB(150,100,255))
    makeLine("Crown: "..crownName, Color3.fromRGB(255,215,0))
    makeLine("Body: "..bodyName, Color3.fromRGB(160,220,160))
    makeLine("Belt: "..beltName, Color3.fromRGB(205,133,63))
    makeLine("Point: "..pointName, Color3.fromRGB(220,220,220))
    makeLine("Moves: "..sig1, Color3.fromRGB(100,200,255))
    makeLine("  "..sig2, Color3.fromRGB(100,200,255))
    makeLine("  "..sig3, Color3.fromRGB(100,200,255))
    makeLine('<font color="#FF4444">ATK:'..atk..'</font>  <font color="#44DD44">DEF:'..def..'</font>  <font color="#FFCC00">STA:'..sta..'</font>', Color3.fromRGB(255,255,255), true)
    yPos += RPM_GAP
    local rpmLabelRow = yPos
    local rpmTitle = Instance.new("TextLabel")
    rpmTitle.BackgroundTransparency = 1 rpmTitle.Size = UDim2.new(0.5,0,0, math.floor(RPM_LABEL_H * sc))
    rpmTitle.Position = UDim2.new(0,3,0, math.floor(rpmLabelRow * sc)) rpmTitle.Font = Enum.Font.GothamBold
    rpmTitle.TextSize = math.floor(15 * sc)
    rpmTitle.TextColor3 = Color3.fromRGB(200,200,200) rpmTitle.TextStrokeTransparency = 0.5
    rpmTitle.TextStrokeColor3 = Color3.fromRGB(0,0,0) rpmTitle.TextXAlignment = Enum.TextXAlignment.Left
    rpmTitle.Text = "RPM" rpmTitle.Parent = bb
    local rpmPct = Instance.new("TextLabel")
    rpmPct.Name = "RPMPct" rpmPct.BackgroundTransparency = 1 rpmPct.Size = UDim2.new(0.5,-3,0, math.floor(RPM_LABEL_H * sc))
    rpmPct.Position = UDim2.new(0.5,0,0, math.floor(rpmLabelRow * sc)) rpmPct.Font = Enum.Font.GothamBold
    rpmPct.TextSize = math.floor(20 * sc)  -- Bigger % label
    rpmPct.TextColor3 = Color3.fromRGB(200,200,200) rpmPct.TextStrokeTransparency = 0.5
    rpmPct.TextStrokeColor3 = Color3.fromRGB(0,0,0) rpmPct.TextXAlignment = Enum.TextXAlignment.Right
    rpmPct.Text = "..." rpmPct.Parent = bb
    local barY = rpmLabelRow + RPM_LABEL_H
    local barBG = Instance.new("Frame")
    barBG.Name = "RPMBarBG" barBG.BackgroundColor3 = Color3.fromRGB(30,30,40) barBG.BorderSizePixel = 0
    barBG.Size = UDim2.new(1,-6,0, math.floor(RPM_BAR_H * sc)) barBG.Position = UDim2.new(0,3,0, math.floor(barY * sc)) barBG.Parent = bb
    local bgCorner = Instance.new("UICorner") bgCorner.CornerRadius = UDim.new(0,3) bgCorner.Parent = barBG
    local barFill = Instance.new("Frame")
    barFill.Name = "RPMBarFill" barFill.BackgroundColor3 = Color3.fromRGB(0,200,255) barFill.BorderSizePixel = 0
    barFill.Size = UDim2.new(0,0,1,0) barFill.Parent = barBG
    local fillCorner = Instance.new("UICorner") fillCorner.CornerRadius = UDim.new(0,3) fillCorner.Parent = barFill
    local barText = Instance.new("TextLabel")
    barText.Name = "RPMBarText" barText.BackgroundTransparency = 1 barText.Size = UDim2.new(1,0,1,0)
    barText.Font = Enum.Font.GothamBold barText.TextSize = math.floor(14 * sc)
    barText.TextColor3 = Color3.fromRGB(255,255,255)
    barText.TextStrokeTransparency = 0.3 barText.TextStrokeColor3 = Color3.fromRGB(0,0,0)
    barText.Text = "..." barText.Parent = barBG
    local lastUpdate = 0 local UPDATE_INTERVAL = 0.1 local lastKnownRPM = 0
    local zeroFrames = 0 local ZERO_FRAME_THRESHOLD = 8
    local rpmConn = RunService.Heartbeat:Connect(function(dt)
        lastUpdate += dt if lastUpdate < UPDATE_INTERVAL then return end lastUpdate = 0
        if not top or not top.Parent or not bb or not bb.Parent then return end
        local maxRPM = getLiveMaxRPM(top)
        local state = tostring(top:GetAttribute("State") or "")
        local isSpinning = LIVE_STATES[state] == true
        local currentRPM
        if isSpinning then
            local liveRPM = getLiveRPM(top)
            if liveRPM and liveRPM > 0 then lastKnownRPM = liveRPM zeroFrames = 0 currentRPM = liveRPM
            else zeroFrames += 1 currentRPM = zeroFrames >= ZERO_FRAME_THRESHOLD and 0 or lastKnownRPM end
        else zeroFrames = 0 currentRPM = lastKnownRPM end
        local rawPct = maxRPM > 0 and (currentRPM / maxRPM) or 0
        local fillPct = math.clamp(rawPct, 0, 1)
        local color = getRPMBarColor(rawPct)
        local pctInt = math.floor(rawPct * 100)
        local displayPct = rawPct >= 1.0 and ("+"..math.min(pctInt,999).."%") or (pctInt.."%")
        barFill.BackgroundColor3 = isSpinning and color or Color3.fromRGB(100,100,100)
        barFill.Size = UDim2.new(fillPct,0,1,0)
        barText.Text = string.format("%d / %d", currentRPM, maxRPM)
        rpmPct.Text = displayPct rpmPct.TextColor3 = isSpinning and color or Color3.fromRGB(150,150,150)
        local freshPart = top:FindFirstChild("Body") or top:FindFirstChild("Crown") or top:FindFirstChildWhichIsA("BasePart")
        if freshPart and bb.Adornee ~= freshPart then bb.Adornee = freshPart end
    end)
    topBBCache[top] = { bb = bb, rpmConn = rpmConn }
    topBuildLock[top] = nil
    local attrConn = top.AttributeChanged:Connect(function(attrName)
        local staticAttrs = { Spirit=true, Attack=true, Defense=true, Stamina=true, SignatureMove_1=true, SignatureMove_2=true, SignatureMove_3=true, MaxHealth=true }
        if staticAttrs[attrName] then
            task.delay(0.5, function() if top and top.Parent and topInspectorEnabled then buildTopBillboard(top) end end)
        end
    end)
    top.AncestryChanged:Connect(function()
        if not top.Parent then
            pcall(function() rpmConn:Disconnect() end) pcall(function() attrConn:Disconnect() end)
            if topBBCache[top] then pcall(function() topBBCache[top].bb:Destroy() end) topBBCache[top] = nil end
            topBuildLock[top] = nil
        end
    end)
end

local function clearAllTopBillboards()
    for _, entry in pairs(topBBCache) do
        pcall(function() entry.rpmConn:Disconnect() end) pcall(function() entry.bb:Destroy() end)
    end
    topBBCache = {} topBuildLock = {}
end

local function teardownTopInspectorListeners()
    for _, conn in pairs(topInspectorConnections) do pcall(function() conn:Disconnect() end) end
    topInspectorConnections = {}
end

local function setupTopInspectorListeners()
    teardownTopInspectorListeners()
    topInspectorConnections["Added"] = workspace.Tops.ChildAdded:Connect(function(top)
        if not topInspectorEnabled then return end
        task.wait(0.5) if not topInspectorEnabled then return end
        if top:GetAttribute("TaggedNPC") then return end
        buildTopBillboard(top)
    end)
    topInspectorConnections["Removed"] = workspace.Tops.ChildRemoved:Connect(function(top)
        if topBBCache[top] then
            pcall(function() topBBCache[top].rpmConn:Disconnect() end)
            pcall(function() topBBCache[top].bb:Destroy() end)
            topBBCache[top] = nil
        end
        topBuildLock[top] = nil
    end)
end

local function refreshTopInspector()
    clearAllTopBillboards() if not topInspectorEnabled then return end
    setupTopInspectorListeners()
    for _, top in ipairs(workspace.Tops:GetChildren()) do
        if not top:GetAttribute("TaggedNPC") then task.spawn(buildTopBillboard, top) end
    end
end

local function onTopInspector(value)
    topInspectorEnabled = value
    if value then
        setupTopInspectorListeners()
        for _, top in ipairs(workspace.Tops:GetChildren()) do
            if not top:GetAttribute("TaggedNPC") then task.spawn(buildTopBillboard, top) end
        end
        Notify("Top Inspector", "Enabled!", 3)
    else
        teardownTopInspectorListeners() clearAllTopBillboards()
        Notify("Top Inspector", "Disabled.", 3)
    end
end
ESPRight:AddToggle('TopInspector', { Text = 'Top Inspector', Default = false, Callback = onTopInspector })
ESPRight:AddButton('Refresh Tops', function()
    refreshTopInspector() Notify("Top Inspector", "Refreshed!", 2)
end)

-- Top Inspector UI Scale slider
ESPRight:AddSlider('TopInspectorScale', {
    Text     = 'Inspector UI Scale',
    Min      = 50, Max = 200, Default = 100,
    Rounding = 0,
    Suffix   = '%',
    Callback = function(value)
        topInspectorScale = value / 100
        if topInspectorEnabled then
            -- Resize existing billboards in-place to avoid flicker (no destroy/rebuild)
            local sc = topInspectorScale
            local BASE_W = 260 local LINE_H = 14 local PAD = 4 local NUM_LINES = 10
            local RPM_LABEL_H = 20 local RPM_BAR_H = 22 local RPM_GAP = 6
            local totalH = PAD + (LINE_H * NUM_LINES) + RPM_GAP + RPM_LABEL_H + RPM_BAR_H + PAD
            for _, entry in pairs(topBBCache) do
                pcall(function()
                    local bb = entry.bb
                    if not bb or not bb.Parent then return end
                    bb.Size = UDim2.new(0, math.floor(BASE_W * sc), 0, math.floor(totalH * sc))
                    -- Resize all child labels and frames
                    local yPos = PAD
                    local lineIdx = 0
                    for _, child in ipairs(bb:GetChildren()) do
                        if child:IsA("TextLabel") and child.Name ~= "RPMPct" then
                            if lineIdx < NUM_LINES then
                                child.Size = UDim2.new(1,-6,0, math.floor(LINE_H * sc))
                                child.Position = UDim2.new(0,3,0, math.floor(yPos * sc))
                                child.TextSize = math.floor(11 * sc)
                                yPos += LINE_H lineIdx += 1
                            elseif child.Name == "RPMPct" then
                                -- handled below
                            end
                        elseif child:IsA("TextLabel") and child.Name == "RPMPct" then
                            local rpmLabelRow = PAD + LINE_H * NUM_LINES + RPM_GAP
                            child.Size = UDim2.new(0.5,-3,0, math.floor(RPM_LABEL_H * sc))
                            child.Position = UDim2.new(0.5,0,0, math.floor(rpmLabelRow * sc))
                            child.TextSize = math.floor(20 * sc)
                        elseif child:IsA("Frame") and child.Name == "RPMBarBG" then
                            local barY = PAD + LINE_H * NUM_LINES + RPM_GAP + RPM_LABEL_H
                            child.Size = UDim2.new(1,-6,0, math.floor(RPM_BAR_H * sc))
                            child.Position = UDim2.new(0,3,0, math.floor(barY * sc))
                            local barText = child:FindFirstChild("RPMBarText")
                            if barText then barText.TextSize = math.floor(14 * sc) end
                        end
                    end
                    -- Also resize the RPM title label (first TextLabel named nothing special at rpmLabelRow)
                    local rpmLabelRow = PAD + LINE_H * NUM_LINES + RPM_GAP
                    for _, child in ipairs(bb:GetChildren()) do
                        if child:IsA("TextLabel") and child.Text == "RPM" then
                            child.Size = UDim2.new(0.5,0,0, math.floor(RPM_LABEL_H * sc))
                            child.Position = UDim2.new(0,3,0, math.floor(rpmLabelRow * sc))
                            child.TextSize = math.floor(15 * sc)
                        end
                    end
                end)
            end
        end
    end,
})

local function onAntiAFK(value)
    toggleStates.AntiAFK = value
    if value then
        antiAFKConnection = RunService.Heartbeat:Connect(function()
            if tick() % 60 < 0.05 then
                VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.F13, false, game) task.wait(0.05)
                VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.F13, false, game)
            end
        end)
        pcall(function()
            idledConnection = LocalPlayer.Idled:Connect(function()
                VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.F13, false, game) task.wait(0.05)
                VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.F13, false, game)
            end)
        end)
        Notify("Anti AFK", "Enabled!", 3)
    else
        if antiAFKConnection then antiAFKConnection:Disconnect() antiAFKConnection = nil end
        if idledConnection then idledConnection:Disconnect() idledConnection = nil end
        Notify("Anti AFK", "Disabled.", 3)
    end
end

            return HttpService:JSONDecode(game:HttpGet("https://games.roblox.com/v1/games/"..placeId.."/servers/Public?sortOrder=Asc&limit=100"))
        end)
        if success and result and result.data and #result.data > 0 then
            local lowest = nil local lowestCount = math.huge
            for _, server in ipairs(result.data) do
                if server.id ~= currentJobId and server.playing < lowestCount then

                Notify("Join Lowest", "Joining server with " .. lowest.playing .. " player(s)...", 4)
                task.wait(1)
                TeleportService:TeleportToPlaceInstance(placeId, lowest.id, LocalPlayer)
            else
                Notify("Join Lowest", "No other servers found.", 3)
            end
        else
            Notify("Join Lowest", "Could not fetch server list.", 3)
        end
    end)
end)

local function getCurrentFlags()
    return {
        AutoLaunchCycle=toggleStates.AutoLaunchCycle, AutoPerfectLaunch=toggleStates.AutoPerfectLaunch,
        AutoBuyBurger=toggleStates.AutoBuyBurger, AutoTrashFarm=toggleStates.AutoTrashFarm,
        AutoTrashFarmBlatant=toggleStates.AutoTrashFarmBlatant, MeteorESP=toggleStates.MeteorESP,
        PlayerESP=toggleStates.PlayerESP, GoldTrashESP=toggleStates.GoldTrashESP,
        AutoStateKeeper=toggleStates.AutoStateKeeper, AntiAFK=toggleStates.AntiAFK,
    }
end

local function applyFlags(flags)
    for flag, value in pairs(flags) do
        local cb = toggleCallbacks[flag]
        if cb then toggleStates[flag] = value pcall(cb, value) end
    end
end

local function getConfigNames()
    local configs = loadAllConfigs()
    local names = {}
    for name in pairs(configs) do table.insert(names, name) end
    table.sort(names)
    if #names == 0 then table.insert(names, "No Configs Saved") end
    return names
end

local ConfigLeft  = Tabs.Config:AddLeftGroupbox('Saved Configs')
local ConfigRight = Tabs.Config:AddRightGroupbox('Create Config')

local selectedConfig = ""
local configNames    = getConfigNames()

local configDropdown = ConfigLeft:AddDropdown('SelectedConfig', {
    Text    = 'Select Config',
    Values  = configNames,
    Default = 1,
    Callback = function(value)
        selectedConfig = value ~= "No Configs Saved" and value or ""
    end,
})
selectedConfig = configNames[1] ~= "No Configs Saved" and configNames[1] or ""

local function refreshDropdown()
    local names = getConfigNames()
    local current = selectedConfig ~= "" and selectedConfig or names[1]
    local found = false
    for _, n in ipairs(names) do if n == current then found = true break end end
    if not found then current = names[1] selectedConfig = current ~= "No Configs Saved" and current or "" end
    pcall(function() configDropdown:SetValues(names) configDropdown:SetValue(current) end)
end

ConfigLeft:AddButton('Load Selected Config', function()
    if selectedConfig == "" or selectedConfig == "No Configs Saved" then
        Notify("Config", "Select a config first!", 3) return
    end
    local cfg = loadAllConfigs()[selectedConfig]
    if cfg and cfg.flags then
        applyFlags(cfg.flags)
        local summary = ""
        for flag, val in pairs(cfg.flags) do if val then summary = summary..flag.." " end end
        Notify("Config Loaded: "..selectedConfig, summary~="" and "ON: "..summary or "All off", 5)
    else Notify("Config", "Config not found!", 3) end
end)

ConfigLeft:AddButton('Delete Selected Config', function()
    if selectedConfig == "" or selectedConfig == "No Configs Saved" then
        Notify("Config", "Select a config to delete!", 3) return
    end
    local configs = loadAllConfigs()
    if configs[selectedConfig] then
        configs[selectedConfig] = nil
        if saveAllConfigs(configs) then
            Notify("Config", "Deleted: "..selectedConfig, 3)
            selectedConfig = "" refreshDropdown()
        end
    else Notify("Config", "Not found!", 3) end
end)
